## 第八章：数据结构

用之前检查区间开闭、k 的起点、空集合返回值和多测清空方式。

| 题目需求 | 入口 | 下标/区间 | 使用提醒 |

### 并查集（连通性与连通块统计）

<a id="dsu"></a>

维护连通性、连通块点数/边数及是否存在自环。节点为 1..n；`merge(x,y)` 每调用一次都计入一条边，自环指 x=y，不等同于一般环。路径压缩配合按大小合并，单次操作均摊 $\mathcal{O}(\alpha (n))$，空间 $\mathcal{O}(n)$；不包含带权关系或回滚功能。

```c++
struct DSU {
    vector<int> fa, p, e, f;

    DSU(int n) {
        fa.resize(n + 1);
        iota(fa.begin(), fa.end(), 0);
        p.resize(n + 1, 1);
        e.resize(n + 1);
        f.resize(n + 1);
    }
    int get(int x) {
        while (x != fa[x]) x = fa[x] = fa[fa[x]];
        return x;
    }
    bool merge(int x, int y) { // 加入一条边，并尝试合并连通块
        if (x == y) f[get(x)] = 1;
        x = get(x), y = get(y);
        e[x]++;
        if (x == y) return false;
        if (p[x] < p[y]) swap(x, y); // 小集合并到大集合
        fa[y] = x;
        f[x] |= f[y], p[x] += p[y], e[x] += e[y];
        return true;
    }
    bool same(int x, int y) {
        return get(x) == get(y);
    }
    bool F(int x) { // 判断连通块内是否存在自环
        return f[get(x)];
    }
    int size(int x) { // 输出连通块中点的数量
        return p[get(x)];
    }
    int E(int x) { // 输出连通块中边的数量
        return e[get(x)];
    }
};
```

调用：`DSU d(5); d.merge(1,2);` 后 `d.same(1,2)` 为 true，`d.size(1)==2`、`d.E(1)==1`。重新构造 `DSU(n)` 清空。

### 带权并查集：差值与异或关系

<a id="potential-dsu"></a>

两个版本独立复制。点编号 1..n；按大小合并和路径压缩，单次均摊 $\mathcal{O}(\alpha (n))$，空间 $\mathcal{O}(n)$。`merge` 返回“本次关系是否与已有关系一致”，false 时不加入矛盾关系；即便不合并两个根，关系一致也返回 true。`diff(x,y)` 直接返回差值/异或值，调用前须检查 `find(x)==find(y)`，不连通时不可查询；重新构造清空。

差值版：`merge(x,y,w)` 表示 value[y]-value[x]=w；`d[x]` 表示 value[x]-value[fa[x]]。输入、存储的势能差和查询答案须在 ll 范围内；组合中间量使用 __int128，调用方须保证结果不超出 ll；原范围断言已注释。它用于维护等式关系，不直接解决不等式差分约束。

```c++
struct WeightedDSU {
    vector<int> fa, sz;
    vector<ll> d;
    WeightedDSU(int n) : fa(n + 1), sz(n + 1, 1), d(n + 1) {
        iota(fa.begin(), fa.end(), 0);
    }
    static ll narrow(i128 x) {
        /* assert(LLONG_MIN <= x && x <= LLONG_MAX) */;
        return (ll)x;
    }
    int find(int x) {
        if (fa[x] == x) return x;
        int p = fa[x];
        fa[x] = find(p);
        if (p != fa[x]) d[x] = narrow((i128)d[x] + d[p]);
        return fa[x];
    }
    bool merge(int x, int y, ll w) {
        int a = find(x), b = find(y);
        if (a == b) return (i128)d[y] - d[x] == w;
        i128 delta = (i128)w + d[x] - d[y];
        if (sz[a] < sz[b]) {
            swap(a, b);
            delta = -delta;
        }
        fa[b] = a;
        d[b] = narrow(delta);
        sz[a] += sz[b];
        return true;
    }
    ll diff(int x, int y) {
        int a = find(x), b = find(y);
        /* assert(a == b) */;
        return narrow((i128)d[y] - d[x]);
    }
};
```

调用：`WeightedDSU d(3); d.merge(1,2,5); d.merge(2,3,-2);` 后 `d.diff(1,3)==3`，再调用 `d.merge(1,3,4)` 返回 false。

异或版：`merge(x,y,w)` 表示 $\mathrm{value}[x]\oplus\mathrm{value}[y]=w$，支持 uint64_t 的全部 64 位。只取 w=0/1 时可维护同类/异类、奇偶关系。

```c++
struct XorDSU {
    vector<int> fa, sz;
    vector<ull> d;
    XorDSU(int n) : fa(n + 1), sz(n + 1, 1), d(n + 1) {
        iota(fa.begin(), fa.end(), 0);
    }
    int find(int x) {
        if (fa[x] == x) return x;
        int p = fa[x];
        fa[x] = find(p);
        d[x] ^= d[p];
        return fa[x];
    }
    bool merge(int x, int y, ull w) {
        int a = find(x), b = find(y);
        if (a == b) return (d[x] ^ d[y]) == w;
        ull delta = w ^ d[x] ^ d[y];
        if (sz[a] < sz[b]) swap(a, b); // 异或关系反向时不取负
        fa[b] = a;
        d[b] = delta;
        sz[a] += sz[b];
        return true;
    }
    ull diff(int x, int y) {
        int a = find(x), b = find(y);
        /* assert(a == b) */;
        return d[x] ^ d[y];
    }
};
```

调用：`XorDSU d(3); d.merge(1,2,1); d.merge(2,3,1);` 后 `d.diff(1,3)==0`。

### 可回滚并查集与离线动态连通性

<a id="rollback-dsu"></a>

1-based；不使用路径压缩，只按大小合并。find/merge 最坏 $\mathcal{O}(\log n)$，snapshot 为 $\mathcal{O}(1)$，rollback 的时间为撤销的成功合并次数。空间 $\mathcal{O}(n)$；未成功合并不压入历史，恢复到快照按历史长度判断。快照仅在当前历史分支有效，不能拿已撤销分支的快照恢复未来状态。

```c++
// comps：当前连通分量数
struct RollbackDSU {
    struct Change {
        int son, fa, old_sz;
    }; // 子根、父根、父根合并前大小
    vector<int> fa, sz;
    vector<Change> his; // 成功合并的修改历史
    int comps;
    RollbackDSU(int n) : fa(n + 1), sz(n + 1, 1), comps(n) {
        iota(fa.begin(), fa.end(), 0);
    }
    int find(int x) const {
        while (fa[x] != x) x = fa[x];
        return x;
    }
    bool same(int x, int y) const {
        return find(x) == find(y);
    }
    int size(int x) const {
        return sz[find(x)];
    }
    int snapshot() const {
        return (int)(his.size());
    }
    bool merge(int x, int y) {
        x = find(x);
        y = find(y);
        if (x == y) return false;
        if (sz[x] < sz[y]) swap(x, y);
        his.push_back({y, x, sz[x]});
        fa[y] = x;
        sz[x] += sz[y];
        --comps;
        return true;
    }
    void rollback(int snapshot) {
        /* assert(0 <= snapshot && snapshot <= (int)(his.size())) */;
        while ((int)(his.size()) > snapshot) {
            auto c = his.back();
            his.pop_back();
            fa[c.son] = c.son;
            sz[c.fa] = c.old_sz;
            ++comps;
        }
    }
};
```

调用：`RollbackDSU d(3); int s=d.snapshot(); d.merge(1,2); d.rollback(s);` 后 1、2 不连通。

#### 时间线段树完整示例

依赖上面的 RollbackDSU。输入 n、q，随后 q 行：`+ u v` 加入一条无向边，`- u v` 删除一条当前存在的边，`? u v` 查询连通性并输出 YES/NO。允许重边和自环；删除按同一端点对最近加入且未删除的副本配对。初始无边，调用方须保证删除的边当前存在；原合法性断言已注释。

每条边的有效时间是 [加入时刻,删除时刻)，分配到时间线段树的 $\mathcal{O}(\log q)$ 个节点。DFS 进入节点时合并，离开时回滚。时间 $\mathcal{O}(n+q \log (q+1) \log (n+1))$，空间 $\mathcal{O}(n+q \log (q+1))$；q=0 可直接退出。多测时整段逻辑应重新构造。

```c++
// addTime：将生效时间区间加入时间线段树
signed main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, q;
    cin >> n >> q;
    /* assert(n > 0 && q >= 0) */;
    if (!q) return 0;
    using Edge = pair<int, int>;
    vector<vector<Edge>> tree(4 * q);
    vector<tuple<char, int, int>> ops(q);
    map<Edge, vector<int>> start;
    auto addTime = [&](auto &&self, int p, int l, int r, int ql, int qr, Edge e) -> void {
        if (ql > r || qr < l) return;
        if (ql <= l && r <= qr) {
            tree[p].push_back(e);
            return;
        }
        int mid = (l + r) / 2;
        self(self, p * 2, l, mid, ql, qr, e);
        self(self, p * 2 + 1, mid + 1, r, ql, qr, e);
    };
    for (int t = 0; t < q; ++t) {
        char op;
        int u, v;
        cin >> op >> u >> v;
        /* assert(1 <= u && u <= n && 1 <= v && v <= n) */;
        if (u > v) swap(u, v);
        ops[t] = {op, u, v};
        Edge e{u, v};
        if (op == '+') start[e].push_back(t);
        else if (op == '-') {
            auto &s = start[e]; /* assert(!s.empty()) */;
            addTime(addTime, 1, 0, q - 1, s.back(), t - 1, e);
            s.pop_back();
        } else /* assert(op == '?') */;
    }
    for (const auto &[e, times] : start)
        for (int l : times) addTime(addTime, 1, 0, q - 1, l, q - 1, e);
    RollbackDSU d(n);
    auto dfs = [&](auto &&self, int p, int l, int r) -> void {
        int s = d.snapshot();
        for (auto [u, v] : tree[p]) d.merge(u, v);
        if (l == r) {
            auto [op, u, v] = ops[l];
            if (op == '?') cout << (d.same(u, v) ? "YES" : "NO") << '\n';
        } else {
            int mid = (l + r) / 2;
            self(self, p * 2, l, mid);
            self(self, p * 2 + 1, mid + 1, r);
        }
        d.rollback(s);
    };
    dfs(dfs, 1, 0, q - 1);
}
```

### 树状数组

<a id="bit"></a>

#### 单点加、区间和

1-based，闭区间 [l,r]；更新位置必须为 1..n，不能传 0。单次更新/查询 $\mathcal{O}(\log n)$，空间 $\mathcal{O}(n)$，初始全 0；累加结果须在 ll 范围内。调用：`BIT bit(5); bit.add(2,3);` 后 `bit.rangeAsk(1,3)==3`；重新构造清空。

```c++
struct BIT {
    int n;
    vector<ll> w;
    BIT(int n) : n(n), w(n + 1) {}
    void add(int x, ll v) { // a[x] += v，1 <= x <= n
        for (; x <= n; x += x & -x) w[x] += v;
    }
    ll ask(int x) const { // 前缀 [1,x]，ask(0)=0
        ll s = 0;
        for (; x; x -= x & -x) s += w[x];
        return s;
    }
    ll rangeAsk(int l, int r) const {
        return ask(r) - ask(l - 1);
    }
};
```

#### 0-based 树状数组：单点加、前缀和与找位置

<a id="fenwick-zero-based"></a>

`FT(n)` 初始化 n 个 0，n≥0；依赖缺省源的 `ll` 和 `sz(x)`。`add(p,x)` 将 a[p] 加 x，要求 0≤p<n；`query(p)` 返回半开区间 [0,p) 的和，要求 0≤p≤n。区间 [l,r) 的和为 `query(r)-query(l)`。初始化时间/空间 $\mathcal{O}(n)$，单次修改、查询、查找 $\mathcal{O}(\log (n+1))$；重新构造清空。长度须在 int 范围内，元素、节点和及所有运算中间量须在 ll 范围内。

`lower_bound(sum)` 找最小下标 p，使闭区间 [0,p] 的和≥sum；调用时要求每个元素非负。sum≤0 返回 -1（空前缀已满足），否则无解返回 n；n=0 时同样适用。普通修改/求和允许负数，负增量也可用于删除频次，但查找时必须保证各频次仍非负。

```c++
struct FT { // 0-based
    vector<ll> s;
    FT(int n) : s(n) {}
    void add(int p, ll x) { // a[p] += x
        for (; p < sz(s); p |= p + 1) s[p] += x;
    }
    ll query(int p) { // [0,p) 的和，query(0)=0
        ll res = 0;
        for (; p > 0; p &= p - 1) res += s[p - 1];
        return res;
    }
    int lower_bound(ll sum) { // 首个使 [0,p] 的和 >= sum 的位置；元素须非负
        if (sum <= 0) return -1;
        int p = 0, pw = 1;
        while (pw <= sz(s) / 2) pw <<= 1; // 按 n 确定最高试探位
        for (; pw; pw >>= 1)
            if (pw <= sz(s) - p && s[p + pw - 1] < sum) p += pw, sum -= s[p - 1];
        return p; // 无解为 n；n=0 且 sum>0 时返回 0
    }
};
```

调用：`FT ft(5); ft.add(0,2); ft.add(3,4);` 后数组为 [2,0,0,4,0]，`ft.query(4)==6`，`ft.query(4)-ft.query(1)==4`；`ft.lower_bound(2)==0`、`ft.lower_bound(3)==3`、`ft.lower_bound(7)==5`、`ft.lower_bound(0)==-1`。若将元素视为频次，`lower_bound(k)` 返回第 k 个元素所在的 0-based 下标。

#### 区间加、区间和

1-based，闭区间 [l,r]；要求 1≤l≤r≤n，初始全 0。使用两个树状数组维护差分，区间加和区间求和均为 $\mathcal{O}(\log n)$，空间 $\mathcal{O}(n)$。调用：`BIT bit(5); bit.rangeAdd(2,4,3);` 后 `bit.rangeSum(1,5)==9`；重新构造清空。

```c++
struct BIT {
    vector<ll> a, b;
    int n;

    BIT(int n) : n(n), a(n + 1), b(n + 1) {}
    void rangeAdd(int l, int r, ll val) { // 区间修改
        auto add = [&](int pos, ll val) {
            for (int i = pos; i <= n; i += i & -i) {
                a[i] += val;
                b[i] += pos * val;
            }
        };
        add(l, val), add(r + 1, -val);
    }
    ll rangeSum(int l, int r) { // 区间和查询
        auto sum = [&](int x) {
            ll ans = 0;
            for (int i = x; i; i -= i & -i) ans += (x + 1) * a[i] - b[i];
            return ans;
        };
        return sum(r) - sum(l - 1);
    }
};
```

#### 逆序对扩展

可直接使用：传入带占位元素的 1-based 数组，`get()` 返回严格逆序对数量，支持重复值及重复调用。时间 $\mathcal{O}(n \log n)$，空间 $\mathcal{O}(n)$。

> 性质：元素互异时，交换两个不同位置的元素会改变逆序数的奇偶性；有重复值时不保证成立。

```c++
struct BIT {
    int n;
    vector<int> w, chk; // chk 为传入的待处理数组
    BIT(const vector<int> &in) : n(in.size() - 1), w(in.size()), chk(in) {}
    void add(int x, int v) {
        for (; x <= n; x += x & -x) w[x] += v;
    }
    ll rangeAsk(int l, int r) {
        auto ask = [&](int x) {
            ll ans = 0;
            for (; x > 0; x -= x & -x) ans += w[x];
            return ans;
        };
        return ask(r) - ask(l - 1);
    }
    ll get() {
        fill(w.begin(), w.end(), 0);
        vector<array<int, 2>> alls;
        for (int i = 1; i <= n; i++) alls.push_back({chk[i], i});
        sort(alls.begin(), alls.end());
        ll ans = 0;
        for (auto [val, idx] : alls) {
            ans += rangeAsk(idx + 1, n);
            add(idx, 1);
        }
        return ans;
    }
};
```

#### 排名扩展

按值域下标 1..n 维护频次，单次操作 $\mathcal{O}(\log n)$，空间 $\mathcal{O}(n)$。频次必须非负；删除的元素须存在，kth 要求 1≤k≤总频次，前驱/后继查询要求答案存在。`get(x)` 可接收 1..n+1。实际值含 0、负数或值域很大时应先离散化，或改用平衡树。

```c++
struct BIT {
    int n;
    vector<int> w;
    BIT(int n) : n(n), w(n + 1) {}
    void add(int x, int v) {
        for (; x <= n; x += x & -x) w[x] += v;
    }
    int kth(int x) { // 查找第 k 小的值
        int ans = 0;
        for (int i = __lg(n); i >= 0; i--) {
            int val = ans + (1 << i);
            if (val < n && w[val] < x) {
                x -= w[val];
                ans = val;
            }
        }
        return ans + 1;
    }
    int get(int x) { // 查找 x 的排名
        int ans = 1;
        for (x--; x; x -= x & -x) ans += w[x];
        return ans;
    }
    int pre(int x) {
        return kth(get(x) - 1);
    } // 查找 x 的前驱
    int suf(int x) {
        return kth(get(x + 1));
    } // 查找 x 的后继
};
const int N = 10000000; // 可以用于在线处理平衡二叉树的全部要求
signed main() {
    BIT bit(N + 1); // 示例要求原值为 1..N；预知全部更新值时可先离散化
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) {
        int op, x;
        cin >> op >> x;
        if (op == 1) bit.add(x, 1); // 插入 x
        else if (op == 2) bit.add(x, -1); // 删除任意一个 x
        else if (op == 3) cout << bit.get(x) << "\n"; // 查询 x 的排名
        else if (op == 4) cout << bit.kth(x) << "\n"; // 查询排名为 x 的数
        else if (op == 5) cout << bit.pre(x) << "\n"; // 求小于 x 的最大值（前驱）
        else if (op == 6) cout << bit.suf(x) << "\n"; // 求大于 x 的最小值（后继）
    }
}
```

#### 统计区间不同数字的数量（离线查询）

<a id="distinct-range"></a>

依赖本章“单点加、区间和”BIT，1-based 闭区间。使用 `pre` 记录每个值最后出现的位置，按右端点离线处理。n 个元素、q 次询问的时间为 $\mathcal{O}((n+q)\log (n+1)+q \log (q+1))$，空间 $\mathcal{O}(n+q)$；`map` 可换成离散化后的数组减小常数。示例输出每个区间的不同值数量。[例题链接](https://www.luogu.com.cn/problem/P1972)。

```c++
signed main() {
    int n;
    cin >> n;
    vector<int> in(n + 1);
    for (int i = 1; i <= n; i++) cin >> in[i];

    int q;
    cin >> q;
    vector<array<int, 3>> query;
    for (int i = 0; i < q; i++) {
        int l, r;
        cin >> l >> r;
        query.push_back({r, l, i});
    }
    sort(query.begin(), query.end());

    vector<ll> ans(q);
    map<int, int> pre;
    int st = 1;
    BIT bit(n);
    for (auto [r, l, id] : query) {
        for (int i = st; i <= r; i++, st++) {
            if (pre.count(in[i])) { // 消除此前操作的影响
                bit.add(pre[in[i]], -1);
            }
            bit.add(i, 1);
            pre[in[i]] = i; // 更新操作
        }
        ans[id] = bit.rangeAsk(l, r);
    }

    for (ll x : ans) cout << x << endl;
}
```

### 普通分块：区间加与小于 x 的数量

<a id="sqrt-blocks"></a>

1-based 闭区间，输入带下标 0 占位，n>0。每块保存未加懒标记的原值和有序副本；整块加只改 tag，散块修改后重建。建表 $\mathcal{O}(n \log B)$，区间加 $\mathcal{O}(B \log B+n/B)$，区间计数 $\mathcal{O}(B+(n/B)\log B)$，空间 $\mathcal{O}(n)$。默认 $B\approx\sqrt n$，可通过构造参数调整。所有原值、标记与实际值须在 ll 范围内；比较 x-tag 用 __int128 避免边界溢出。init 重置多测。

```c++
// bs：指定块长，0 表示自动；bn：块数；srt：各块有序副本
struct SqrtBlocks {
    int n, B, bn;
    vector<ll> a, tag;
    vector<vector<ll>> srt;
    SqrtBlocks(const vector<ll> &in, int bs = 0) {
        init(in, bs);
    }
    void init(const vector<ll> &in, int bs = 0) {
        n = (int)(in.size()) - 1; /* assert(n > 0 && bs >= 0) */;
        B = bs ? bs : max<int>(1, (int)(sqrt(n)));
        bn = (n - 1) / B + 1;
        a = in;
        tag.assign(bn, 0);
        srt.assign(bn, {});
        for (int b = 0; b < bn; ++b) rebuild(b);
    }
    int left(int b) const {
        return b * B + 1;
    }
    int right(int b) const {
        return (int)(min(1LL * n, (b + 1LL) * B));
    }
    void rebuild(int b) {
        srt[b].assign(a.begin() + left(b), a.begin() + right(b) + 1);
        sort(srt[b].begin(), srt[b].end());
    }
    void range_add(int l, int r, ll d) {
        /* assert(1 <= l && l <= r && r <= n) */;
        for (int b = (l - 1) / B; b <= (r - 1) / B; ++b) {
            int L = max(l, left(b)), R = min(r, right(b));
            if (L == left(b) && R == right(b)) tag[b] += d;
            else {
                for (int i = L; i <= R; ++i) a[i] += d;
                rebuild(b);
            }
        }
    }
    int count_less(int l, int r, ll x) const {
        /* assert(1 <= l && l <= r && r <= n) */;
        int ans = 0;
        for (int b = (l - 1) / B; b <= (r - 1) / B; ++b) {
            int L = max(l, left(b)), R = min(r, right(b));
            if (L == left(b) && R == right(b)) {
                i128 bound = (i128)x - tag[b];
                ans += (int)(lower_bound(srt[b].begin(), srt[b].end(), bound,
                                         [](ll value, i128 limit) { return (i128)value < limit; }) -
                             srt[b].begin());
            } else
                for (int i = L; i <= R; ++i) ans += (i128)a[i] + tag[b] < x;
        }
        return ans;
    }
    ll at(int i) const {
        /* assert(1 <= i && i <= n) */;
        return a[i] + tag[(i - 1) / B];
    }
};
```

### 扫描线与 CDQ 离线数点

<a id="offline-counting"></a>

#### 二维扫描线：闭矩形内点数

静态点集，允许重复点；每个矩形 [x1,x2]×[y1,y2] 为闭区间。按 x 扫描，用内部 BIT 维护 y 的频次。将矩形拆为 x≤x2 与 x<x1 两个事件，避免对 LLONG_MIN 做减一；同 x 的严格事件先处理。返回值与询问输入顺序一致，空点集/空询问可用。n 点、q 个矩形，时间 $\mathcal{O}((n+q)\log (n+q+1))$、空间 $\mathcal{O}(n+q)$。坐标支持整个 ll 范围，不依赖外部 BIT。

```c++
// rectCnt：离线二维矩形数点
vector<ll> rectCnt(vector<array<ll, 2>> pts, const vector<array<ll, 4>> &rects) {
    struct Event {
        ll x, y1, y2;
        bool inclusive;
        int id, sign;
    };
    vector<Event> evt; // 由矩形询问拆出的扫描线事件
    for (int i = 0; i < (int)(rects.size()); ++i) {
        auto [x1, y1, x2, y2] = rects[i]; /* assert(x1 <= x2 && y1 <= y2) */;
        evt.push_back({x1, y1, y2, false, i, -1});
        evt.push_back({x2, y1, y2, true, i, 1});
    }
    sort(pts.begin(), pts.end());
    vector<ll> ys;
    for (auto p : pts) ys.push_back(p[1]);
    sort(ys.begin(), ys.end());
    ys.erase(unique(ys.begin(), ys.end()), ys.end());
    sort(evt.begin(), evt.end(),
         [](const Event &a, const Event &b) { return tie(a.x, a.inclusive) < tie(b.x, b.inclusive); });
    vector<int> bit(ys.size() + 1);
    auto add = [&](int i) {
        for (; i < (int)(bit.size()); i += i & -i) ++bit[i];
    };
    auto sum = [&](int i) {
        int ans = 0;
        for (; i; i -= i & -i) ans += bit[i];
        return ans;
    };
    vector<ll> ans(rects.size());
    int pos = 0;
    for (const auto &e : evt) {
        while (pos < (int)(pts.size()) && (pts[pos][0] < e.x || (e.inclusive && pts[pos][0] == e.x))) {
            add((int)(lower_bound(ys.begin(), ys.end(), pts[pos][1]) - ys.begin()) + 1);
            ++pos;
        }
        int lo = (int)(lower_bound(ys.begin(), ys.end(), e.y1) - ys.begin());
        int hi = (int)(upper_bound(ys.begin(), ys.end(), e.y2) - ys.begin());
        ans[e.id] += e.sign * ll(sum(hi) - sum(lo));
    }
    return ans;
}
```

#### CDQ：三维偏序（含重复点）

输入普通 0-based 点数组，返回每个点被多少个其他点支配：统计 j≠i 且 x[j]≤x[i]、y[j]≤y[i]、z[j]≤z[i]。全相同的 m 个点，每个答案为 m-1。先按三维排序合并重复点，再 CDQ；递归结束后该段按 y 有序。BIT 的贡献在每层合并结束前撤销，可重复调用。坐标为 ll，点数小于 INT_MAX；时间 $\mathcal{O}(n \log ^{2}(n+1))$、空间 $\mathcal{O}(n)$，递归深度 $\mathcal{O}(\log (n+1))$，空数组返回空。

```c++
// domCnt：CDQ 三维偏序计数
vector<int> domCnt(const vector<array<ll, 3>> &input) {
    int n = (int)(input.size());
    if (!n) return {};
    vector<int> ord(n);
    iota(ord.begin(), ord.end(), 0); // 三维排序后的原下标
    sort(ord.begin(), ord.end(), [&](int i, int j) { return input[i] < input[j]; });
    struct Point {
        ll x, y, z;
        int rank = 0, count = 0, id = 0, ans = 0;
    };
    vector<Point> a;
    vector<int> group(n);
    vector<ll> zs;
    for (int i : ord) {
        auto [x, y, z] = input[i];
        if (a.empty() || tie(a.back().x, a.back().y, a.back().z) != tie(x, y, z)) {
            a.push_back({x, y, z, 0, 0, (int)(a.size()), 0});
            zs.push_back(z);
        }
        ++a.back().count;
        group[i] = a.back().id;
    }
    sort(zs.begin(), zs.end());
    zs.erase(unique(zs.begin(), zs.end()), zs.end());
    for (auto &p : a) p.rank = (int)(lower_bound(zs.begin(), zs.end(), p.z) - zs.begin()) + 1;
    vector<int> bit(zs.size() + 1);
    vector<Point> tmp(a.size());
    auto add = [&](int i, int d) {
        for (; i < (int)(bit.size()); i += i & -i) bit[i] += d;
    };
    auto sum = [&](int i) {
        int s = 0;
        for (; i; i -= i & -i) s += bit[i];
        return s;
    };
    auto cdq = [&](auto &&self, int l, int r) -> void {
        if (r - l <= 1) return;
        int m = (l + r) / 2;
        self(self, l, m);
        self(self, m, r);
        int i = l;
        for (int j = m; j < r; ++j) {
            while (i < m && a[i].y <= a[j].y) {
                add(a[i].rank, a[i].count);
                ++i;
            }
            a[j].ans += sum(a[j].rank);
        }
        for (int k = l; k < i; ++k) add(a[k].rank, -a[k].count);
        merge(a.begin() + l, a.begin() + m, a.begin() + m, a.begin() + r, tmp.begin() + l,
              [](const Point &p, const Point &q) { return p.y < q.y; });
        copy(tmp.begin() + l, tmp.begin() + r, a.begin() + l);
    };
    cdq(cdq, 0, (int)(a.size()));
    vector<int> value(a.size()), ans(n);
    for (auto p : a) value[p.id] = p.ans + p.count - 1;
    for (int i = 0; i < n; ++i) ans[i] = value[group[i]];
    return ans;
}
```

调用：`domCnt({{1,1,1},{1,1,1},{2,2,2}})` 返回 `{1,1,2}`。若题目要求输出“答案恰为 k 的点数”，对返回数组再统计频次；不要把排除自身改成减掉全部重复点。

### 二维树状数组

#### 半动态矩阵和（单点修改）

1-based 闭矩形，初始全 0。两种模式二选一：① pointAdd 单点加＋matrixSum 矩形和，此时 pointSum(x,y) 是二维前缀和；② matrixAdd 矩形加＋pointSum 单点值，此时内部保存二维差分，不能再用 matrixSum 求矩形和。更新坐标不能为 0，查询前缀允许坐标为 0。

初始化时间、空间 $\mathcal{O}(nm)$，单次更新/查询 $\mathcal{O}(\log (n+1)\log (m+1))$；逐点载入已有矩阵另需 $\mathcal{O}(nm \log (n+1)\log (m+1))$，不是线性建树。T 建议用 ll，中间运算须不溢出；重新构造清空。

调用：`BIT_2D<ll> b(3,4); b.pointAdd(2,3,5);` 后 `b.matrixSum(1,1,3,4)==5`。差分模式另建对象：`BIT_2D<ll> d(3,4); d.matrixAdd(1,2,3,4,7);` 后 `d.pointSum(2,3)==7`。

```c++
template <class T> struct BIT_2D {
    int n, m;
    vector<vector<T>> w;

    BIT_2D(int n, int m) : n(n), m(m) {
        w.resize(n + 1, vector<T>(m + 1));
    }
    void pointAdd(int x, int y, T k) {
        for (int i = x; i <= n; i += i & -i)
            for (int j = y; j <= m; j += j & -j) w[i][j] += k;
    }
    void matrixAdd(int x, int y, int X, int Y, T k) { // 区块修改：二维差分
        X++, Y++;
        pointAdd(x, y, k), pointAdd(X, y, -k);
        pointAdd(X, Y, k), pointAdd(x, Y, -k);
    }
    T pointSum(int x, int y) {
        T ans = T();
        for (int i = x; i; i -= i & -i)
            for (int j = y; j; j -= j & -j) ans += w[i][j];
        return ans;
    }
    T matrixSum(int x, int y, int X, int Y) { // 区块查询：二维前缀和
        x--, y--;
        return pointSum(X, Y) - pointSum(x, Y) - pointSum(X, y) + pointSum(x, y);
    }
};
```

#### 动态矩阵和（区间修改）

1-based 闭矩形，支持矩形加、矩形求和；初始全 0，合法范围为 1≤x≤X≤n、1≤y≤Y≤m。四张 BIT 分别维护差分及其乘 x-1、y-1、(x-1)(y-1) 的贡献；初始化时间、空间 $\mathcal{O}(nm)$，更新/查询 $\mathcal{O}(\log (n+1)\log (m+1))$，常数比单张 BIT 大。T 建议为 ll，权值、坐标乘积及求和中间量须在 T 范围内；重新构造清空。

调用：`BIT_2D<ll> b(3,4); b.matrixAdd(1,2,3,4,2);` 后 `b.matrixSum(1,1,3,4)==18`。

```c++
template <class T> struct BIT_2D {
    int n, m;
    vector<vector<T>> b1, b2, b3, b4;

    BIT_2D(int n, int m) : n(n), m(m) {
        b1.resize(n + 1, vector<T>(m + 1));
        b2.resize(n + 1, vector<T>(m + 1));
        b3.resize(n + 1, vector<T>(m + 1));
        b4.resize(n + 1, vector<T>(m + 1));
    }
    void matrixAdd(int x, int y, int X, int Y, T k) { // 区块修改：二维差分
        X++, Y++;
        auto add = [&](int x, int y, T k) {
            for (int i = x; i <= n; i += i & -i)
                for (int j = y; j <= m; j += j & -j) {
                    b1[i][j] += k;
                    b2[i][j] += k * (x - 1);
                    b3[i][j] += k * (y - 1);
                    b4[i][j] += k * (x - 1) * (y - 1);
                }
        };
        add(x, y, k), add(X, y, -k);
        add(X, Y, k), add(x, Y, -k);
    }
    T matrixSum(int x, int y, int X, int Y) { // 区块查询：二维前缀和
        x--, y--;
        auto ask = [&](int x, int y) {
            T ans = T();
            for (int i = x; i; i -= i & -i)
                for (int j = y; j; j -= j & -j) {
                    ans += T(x) * y * b1[i][j];
                    ans -= y * b2[i][j];
                    ans -= x * b3[i][j];
                    ans += b4[i][j];
                }
            return ans;
        };
        return ask(X, Y) - ask(x, Y) - ask(X, y) + ask(x, y);
    }
};
```

#### 离线离散化二维 BIT（稀疏坐标）

<a id="sparse-bit-2d"></a>

先收集所有可能被修改的坐标对，构造后可按任意在线顺序单点加、闭矩形求和；查询边界不必预先出现。只允许修改预登记的精确坐标对，不能中途加入新坐标；值初始为 0，重复登记自动去重。坐标为 ll，支持负数和 ll 极值；权值、BIT 中间和及四项容斥中间结果须在 ll 范围内。空坐标集合可查询，结果为 0。

M 个不同修改位置，预处理 $\mathcal{O}(M \log ^{2}(M+1))$，更新/查询 $\mathcal{O}(\log ^{2}(M+1))$，空间 $\mathcal{O}(M \log (M+1))$。clearVal 清零权值、保留坐标，init 重新登记并清零。相比稠密二维 BIT，空间取决于修改点数，不取决于坐标范围乘积。

```c++
// clearVal：清空二维 BIT 数值，保留坐标表
struct SparseBIT2D {
    vector<pair<ll, ll>> pts; // 可修改的已登记坐标点
    vector<ll> xs;
    vector<vector<ll>> ys, bit;
    SparseBIT2D(vector<pair<ll, ll>> pos = {}) {
        init(move(pos));
    }
    void init(vector<pair<ll, ll>> pos) {
        pts = move(pos);
        sort(pts.begin(), pts.end());
        pts.erase(unique(pts.begin(), pts.end()), pts.end());
        xs.clear();
        for (auto [x, y] : pts) xs.push_back(x);
        xs.erase(unique(xs.begin(), xs.end()), xs.end());
        ys.assign(xs.size() + 1, {});
        bit.assign(xs.size() + 1, {});
        for (auto [x, y] : pts) {
            int p = (int)(lower_bound(xs.begin(), xs.end(), x) - xs.begin()) + 1;
            for (int i = p; i < (int)(ys.size()); i += i & -i) ys[i].push_back(y);
        }
        for (int i = 1; i < (int)(ys.size()); ++i) {
            auto &v = ys[i];
            sort(v.begin(), v.end());
            v.erase(unique(v.begin(), v.end()), v.end());
            bit[i].assign(v.size() + 1, 0);
        }
    }
    void clearVal() {
        for (auto &v : bit) fill(v.begin(), v.end(), 0);
    }
    void add(ll x, ll y, ll delta) {
        /* assert(binary_search(pts.begin(), pts.end(), pair<ll, ll>{x, y})) */;
        int p = (int)(lower_bound(xs.begin(), xs.end(), x) - xs.begin()) + 1;
        for (int i = p; i < (int)(ys.size()); i += i & -i) {
            int q = (int)(lower_bound(ys[i].begin(), ys[i].end(), y) - ys[i].begin()) + 1;
            for (int j = q; j < (int)(bit[i].size()); j += j & -j) bit[i][j] += delta;
        }
    }
    // incX/incY：前缀是否包含恰好等于 x/y 的边界
    ll prefix(ll x, ll y, bool incX = true, bool incY = true) const {
        int p =
            (int)((incX ? upper_bound(xs.begin(), xs.end(), x) : lower_bound(xs.begin(), xs.end(), x)) - xs.begin());
        ll ans = 0;
        for (int i = p; i; i -= i & -i) {
            int q =
                (int)((incY ? upper_bound(ys[i].begin(), ys[i].end(), y) : lower_bound(ys[i].begin(), ys[i].end(), y)) -
                      ys[i].begin());
            for (int j = q; j; j -= j & -j) ans += bit[i][j];
        }
        return ans;
    }
    ll rectangle(ll x1, ll y1, ll x2, ll y2) const {
        /* assert(x1 <= x2 && y1 <= y2) */;
        return prefix(x2, y2) - prefix(x1, y2, false, true) - prefix(x2, y1, true, false) +
               prefix(x1, y1, false, false);
    }
};
```

调用：`SparseBIT2D bit({{1000000000,-7},{2,3}}); bit.add(2,3,5); bit.add(1000000000,-7,2);` 后 `bit.rectangle(0,-10,1000000000,3)==7`。删点可加负权，但必须保证题目要求的频次非负；结构本身允许负权和。

### 线段树

#### 不取模的区间赋值、加法、求和与找位置

<a id="range-sum-segtree"></a>

独立模板，维护 `a[1..n]`，`a[0]` 不使用，n>0。`rangeSet/rangeAdd/query` 分别进行区间赋值、加法与求和；`firstPos(l,r,x)` 返回 [l,r] 内第一个值不小于 x 的位置；`firstPre(l,r,k)` 返回最小 p，使 `a[l]+...+a[p]>=k`，无解均返回 -1。建树 $\mathcal{O}(n)$，其余操作 $\mathcal{O}(\log n)$，空间 $\mathcal{O}(n)$。

赋值可以是 0 或负数，但调用 `firstPre` 时必须保证 [l,r] 内当前元素均非负，使前缀和单调。值、区间和、`区间长度*修改值` 及累计标记均须在 ll 范围内。`init` 可重建并清空旧标记，所有操作均要求闭区间合法且非空。

标记顺序例子：初始 5，先赋值 2 再加 3 得 5；先加 3 再赋值 2 得 2。后来的赋值覆盖旧标记，后来的加法累加到现有标记。

```c++
struct SegSum {
    struct Node {
        ll sum = 0, mx = LLONG_MIN, add = 0, setv = 0;
        bool hasSet = false;
    };
    int n = 0;
    vector<Node> t;
    SegSum() = default;
    SegSum(const vector<ll> &a) {
        init(a);
    }
    void init(const vector<ll> &a) {
        n = (int)a.size() - 1; /* assert(n > 0) */;
        t.assign(4 * n + 4, {});
        build(1, 1, n, a);
    }
    void build(int p, int l, int r, const vector<ll> &a) {
        if (l == r) {
            t[p].sum = t[p].mx = a[l];
            return;
        }
        int m = (l + r) / 2;
        build(p * 2, l, m, a);
        build(p * 2 + 1, m + 1, r, a);
        pull(p);
    }
    void pull(int p) {
        t[p].sum = t[p * 2].sum + t[p * 2 + 1].sum;
        t[p].mx = max(t[p * 2].mx, t[p * 2 + 1].mx);
    }
    void applySet(int p, int l, int r, ll v) {
        t[p].sum = v * (r - l + 1);
        t[p].mx = v;
        t[p].setv = v;
        t[p].add = 0;
        t[p].hasSet = true;
    }
    void applyAdd(int p, int l, int r, ll v) {
        t[p].sum += v * (r - l + 1);
        t[p].mx += v;
        if (t[p].hasSet) t[p].setv += v;
        else t[p].add += v;
    }
    void push(int p, int l, int r) {
        int m = (l + r) / 2;
        if (t[p].hasSet) {
            applySet(p * 2, l, m, t[p].setv);
            applySet(p * 2 + 1, m + 1, r, t[p].setv);
            t[p].hasSet = false;
        }
        if (t[p].add) {
            applyAdd(p * 2, l, m, t[p].add);
            applyAdd(p * 2 + 1, m + 1, r, t[p].add);
            t[p].add = 0;
        }
    }
    void rangeSet(int l, int r, ll v) {
        /* assert(1 <= l && l <= r && r <= n) */;
        rangeSet(1, 1, n, l, r, v);
    }
    void rangeSet(int p, int l, int r, int ql, int qr, ll v) {
        if (ql <= l && r <= qr) {
            applySet(p, l, r, v);
            return;
        }
        push(p, l, r);
        int m = (l + r) / 2;
        if (ql <= m) rangeSet(p * 2, l, m, ql, qr, v);
        if (qr > m) rangeSet(p * 2 + 1, m + 1, r, ql, qr, v);
        pull(p);
    }
    void rangeAdd(int l, int r, ll v) {
        /* assert(1 <= l && l <= r && r <= n) */;
        rangeAdd(1, 1, n, l, r, v);
    }
    void rangeAdd(int p, int l, int r, int ql, int qr, ll v) {
        if (ql <= l && r <= qr) {
            applyAdd(p, l, r, v);
            return;
        }
        push(p, l, r);
        int m = (l + r) / 2;
        if (ql <= m) rangeAdd(p * 2, l, m, ql, qr, v);
        if (qr > m) rangeAdd(p * 2 + 1, m + 1, r, ql, qr, v);
        pull(p);
    }
    ll query(int l, int r) {
        /* assert(1 <= l && l <= r && r <= n) */;
        return query(1, 1, n, l, r);
    }
    ll query(int p, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr) return t[p].sum;
        push(p, l, r);
        int m = (l + r) / 2;
        ll ans = 0;
        if (ql <= m) ans += query(p * 2, l, m, ql, qr);
        if (qr > m) ans += query(p * 2 + 1, m + 1, r, ql, qr);
        return ans;
    }
    int firstPos(int l, int r, ll x) {
        /* assert(1 <= l && l <= r && r <= n) */;
        return firstPos(1, 1, n, l, r, x);
    }
    int firstPos(int p, int l, int r, int ql, int qr, ll x) {
        if (r < ql || qr < l || t[p].mx < x) return -1;
        if (l == r) return l;
        push(p, l, r);
        int m = (l + r) / 2;
        int ans = firstPos(p * 2, l, m, ql, qr, x);
        if (ans == -1) ans = firstPos(p * 2 + 1, m + 1, r, ql, qr, x);
        return ans;
    }
    int firstPre(int l, int r, ll k) {
        /* assert(1 <= l && l <= r && r <= n) */;
        ll sum = 0;
        return firstPre(1, 1, n, l, r, k, sum);
    }
    int firstPre(int p, int l, int r, int ql, int qr, ll k, ll &sum) {
        if (r < ql || qr < l) return -1;
        if (ql <= l && r <= qr && sum + t[p].sum < k) {
            sum += t[p].sum;
            return -1;
        }
        if (l == r) return l;
        push(p, l, r);
        int m = (l + r) / 2;
        int ans = firstPre(p * 2, l, m, ql, qr, k, sum);
        if (ans == -1) ans = firstPre(p * 2 + 1, m + 1, r, ql, qr, k, sum);
        return ans;
    }
};
```

调用：`vector<ll> a{0,1,2,3}; SegSum seg(a); seg.rangeSet(1,2,4); seg.rangeAdd(2,3,2);` 后数组为 [4,6,5]，`seg.query(1,3)==15`、`seg.firstPos(1,3,5)==2`、`seg.firstPre(1,3,10)==2`。单点赋值可写作 `seg.rangeSet(p,p,x)`。

#### 最大子段和

<a id="max-subarray"></a>

独立模板，维护 `a[1..n]`，`a[0]` 不使用，n>0。支持单点赋值和区间最大子段和查询：建树 $\mathcal{O}(n)$，修改/查询 $\mathcal{O}(\log n)$，空间 $\mathcal{O}(n)$。`Node` 的 `sum/pref/suff/best` 分别为整段和、非空最大前缀和、非空最大后缀和、非空最大子段和，因此全负数组返回最大元素。所有计算须在 ll 范围内；若题目允许空子段，对最终 `best` 与 0 取 max。

```c++
struct SegSubarray {
    struct Node {
        ll sum, pref, suff, best;
    };
    int n = 0;
    vector<Node> t;
    SegSubarray() = default;
    SegSubarray(const vector<ll> &a) {
        init(a);
    }
    static Node merge(const Node &a, const Node &b) {
        Node c;
        c.sum = a.sum + b.sum;
        c.pref = max(a.pref, a.sum + b.pref);
        c.suff = max(b.suff, b.sum + a.suff);
        c.best = max({a.best, b.best, a.suff + b.pref});
        return c;
    }
    void init(const vector<ll> &a) {
        n = (int)a.size() - 1; /* assert(n > 0) */;
        t.resize(4 * n + 4);
        build(1, 1, n, a);
    }
    void build(int p, int l, int r, const vector<ll> &a) {
        if (l == r) {
            t[p] = {a[l], a[l], a[l], a[l]};
            return;
        }
        int m = (l + r) / 2;
        build(p * 2, l, m, a);
        build(p * 2 + 1, m + 1, r, a);
        t[p] = merge(t[p * 2], t[p * 2 + 1]);
    }
    void pointSet(int x, ll v) {
        /* assert(1 <= x && x <= n) */;
        pointSet(1, 1, n, x, v);
    }
    void pointSet(int p, int l, int r, int x, ll v) {
        if (l == r) {
            t[p] = {v, v, v, v};
            return;
        }
        int m = (l + r) / 2;
        if (x <= m) pointSet(p * 2, l, m, x, v);
        else pointSet(p * 2 + 1, m + 1, r, x, v);
        t[p] = merge(t[p * 2], t[p * 2 + 1]);
    }
    Node query(int l, int r) {
        /* assert(1 <= l && l <= r && r <= n) */;
        return query(1, 1, n, l, r);
    }
    Node query(int p, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr) return t[p];
        int m = (l + r) / 2;
        if (qr <= m) return query(p * 2, l, m, ql, qr);
        if (ql > m) return query(p * 2 + 1, m + 1, r, ql, qr);
        return merge(query(p * 2, l, m, ql, qr), query(p * 2 + 1, m + 1, r, ql, qr));
    }
};
```

调用：`vector<ll> a{0,-3,2,-1}; SegSubarray seg(a);` 此时 `seg.query(1,3).best==2`；`seg.pointSet(1,4)` 后答案为 6。

#### 迭代式线段树：单点赋值、区间最大值

0-based，闭区间 [l,r]，n>0，初始全 0。单点赋值和区间最大值查询均为 $\mathcal{O}(\log n)$，空间 $\mathcal{O}(n)$。这是线段树，不使用树状数组的 lowbit 操作。

```c++
struct SegMax {
    vector<ll> w;
    int n;
    SegMax(int n) : n(n), w(2 * n) {
        /* assert(n > 0) */;
    }

    void modify(int pos, ll val) {
        /* assert(0 <= pos && pos < n) */;
        for (w[pos += n] = val; pos > 1; pos /= 2) w[pos / 2] = max(w[pos], w[pos ^ 1]);
    }

    ll rangeMax(int l, int r) {
        /* assert(0 <= l && l <= r && r < n) */;
        r++; // 使用开区间实现
        ll res = numeric_limits<ll>::lowest();
        for (l += n, r += n; l < r; l /= 2, r /= 2) {
            if (l % 2) res = max(res, w[l++]);
            if (r % 2) res = max(res, w[--r]);
        }
        return res;
    }
};
```

#### 区间加法修改、区间最小值查询

1-based，闭区间 [l,r]，n>0；同时维护区间和与最小值。建树 $\mathcal{O}(n)$，区间加/求和/最小值查询 $\mathcal{O}(\log n)$，空间 $\mathcal{O}(n)$。可赋予负增量；建议 T=ll，并保证所有中间结果不溢出。

双最值切换：把下方带“单最值”的行注释掉，启用紧邻的“双最值”备选行，再取消整个 rmqMax 函数的行注释。rmq1 为最小值、rmq2 为最大值；原 `rmq(l,r)` 仍查最小值，`rmqMax(l,r)` 查最大值，`ask(l,r)` 仍查区间和。rmq 内的 min 与初值 max() 保持不动；若保留 debug，也切换对应输出行。

两种最值均支持区间加，复杂度不变，节点只多一个 T 字段。切换后例如 `Segt<ll> s(vector<ll>{0,3,-1,2}); s.modify(1,2,4);` 得 [7,3,2]，`s.rmq(1,3)==2`、`s.rmqMax(1,3)==7`、`s.ask(1,3)==12`。

调用：`Segt<ll> s(vector<ll>{0,3,1,2}); s.modify(1,2,4);` 后 `s.ask(1,3)==14`、`s.rmq(1,3)==2`；重新构造清空。

```c++
template <class T> struct Segt {
    struct node {
        int l, r;
        T w, rmq, lazy;
    };
    vector<T> w;
    vector<node> t;

    Segt(int n) : Segt(vector<T>(n + 1)) {}
    Segt(const vector<T> &in) : w(in), t(4 * in.size()) {
        build(1, in.size() - 1);
    }

#define GL (k << 1)
#define GR (k << 1 | 1)

    void build(int l, int r, int k = 1) {
        if (l == r) {
            t[k] = {l, r, w[l], w[l], 0};
            return;
        }
        t[k] = {l, r, 0, 0, 0};
        int m = (l + r) / 2;
        build(l, m, GL);
        build(m + 1, r, GR);
        pushup(k);
    }
    void pushdown(node &p, T lazy) {
        p.w += (p.r - p.l + 1) * lazy;
        p.rmq += lazy;
        p.lazy += lazy;
    }
    void pushdown(int k) {
        if (t[k].lazy == 0) return;
        pushdown(t[GL], t[k].lazy);
        pushdown(t[GR], t[k].lazy);
        t[k].lazy = 0;
    }
    void pushup(int k) {
        auto pushup = [&](node &p, node &l, node &r) {
            p.w = l.w + r.w;
            p.rmq = min(l.rmq, r.rmq);
        };
        pushup(t[k], t[GL], t[GR]);
    }
    void modify(int l, int r, T val, int k = 1) {
        if (l <= t[k].l && t[k].r <= r) {
            pushdown(t[k], val);
            return;
        }
        pushdown(k);
        int mid = (t[k].l + t[k].r) / 2;
        if (l <= mid) modify(l, r, val, GL);
        if (mid < r) modify(l, r, val, GR);
        pushup(k);
    }
    T rmq(int l, int r, int k = 1) {
        if (l <= t[k].l && t[k].r <= r) return t[k].rmq;
        pushdown(k);
        int m = (t[k].l + t[k].r) / 2;
        T ans = numeric_limits<T>::max();
        if (l <= m) ans = min(ans, rmq(l, r, GL));
        if (m < r) ans = min(ans, rmq(l, r, GR));
        return ans;
    }
    T ask(int l, int r, int k = 1) {
        if (l <= t[k].l && t[k].r <= r) return t[k].w;
        pushdown(k);
        int m = (t[k].l + t[k].r) / 2;
        T ans = 0;
        if (l <= m) ans += ask(l, r, GL);
        if (m < r) ans += ask(l, r, GR);
        return ans;
    }

#undef GL
#undef GR
};
```

#### 仿射懒标记：区间加、乘、赋值与求和（取模）

<a id="affine-segtree"></a>

主力区间和模板：统一施加 x→x*mul+add，分别用 (mul,add)=(1,v)、(v,0)、(0,v) 表示加、乘、赋值。公开接口 `upd(l,r,mul,add)` 与该顺序一致；旧版顺序为 add、mul，已有直接调用处需要迁移。1-based，闭区间 [l,r]，n>0、mod>=0；mod=0 表示不取模，mod>0 表示取模。T 使用 int 或 ll，输入可为负数。建树 $\mathcal{O}(n)$，更新/查询 $\mathcal{O}(\log n)$，空间 $\mathcal{O}(n)$。依赖 GNU/Clang 的 `__int128_t` 计算乘加中间值；不取模时，节点和、懒标记及每次加法/乘法返回的结果都必须在 T 范围内，i128 中间计算不能防止转回 T 时超范围。只维护区间和。

调用：`Segt<ll> seg(vector<ll>{0,1,2,3},1000); seg.upd(1,3,2,5);` 后数组为 [7,9,11]，`seg.query_sum(1,3)==27`。`seg.init(n,mod)` 可重置为全零并重新指定模式。

不取模：`Segt<ll> seg(vector<ll>{0,1,2,3},0); seg.range_add(1,3,3); seg.range_mul(1,2,-2); seg.range_set(2,3,5);` 后数组为 [-8,5,5]，`seg.query_sum(1,3)==2`；已有 1-based 数组 a 时使用 `Segt<ll> seg(a,0);`。

```c++
// upd：对闭区间施加更新
template <class T> struct Segt {
    struct node {
        int l, r;
        T w, add, mul; // 区间和、懒加、懒乘
    };

    int n = 0;
    T mod = 1; // 0：不取模；正数：模数
    vector<T> a; // 1-indexed
    vector<node> t;

    Segt() {}
    Segt(int n_, T mod_) {
        init(n_, mod_);
    }
    Segt(const vector<T> &in, T mod_) { // 约定 in 为 1-indexed
        n = (int)in.size() - 1;
        mod = mod_;
        /* assert(n > 0 && mod >= 0) */;
        a = in;
        t.resize(4 * n + 5);
        build(1, n);
    }

    void init(int n_, T mod_) {
        /* assert(n_ > 0 && mod_ >= 0) */;
        n = n_;
        mod = mod_;
        if (n <= 0) return;
        a.assign(n + 1, T{});
        t.resize((n << 2) + 5);
        build(1, n); // 默认全 0
    }
    inline T norm(T x) const {
        if (!mod) return x;
        x %= mod;
        if (x < 0) x += mod;
        return x;
    }

    T add_mod(T x, T y) const {
        i128 z = i128(x) + y;
        return mod ? T(z % mod) : T(z);
    }
    T mul_mod(T x, T y) const {
        i128 z = i128(x) * y;
        return mod ? T(z % mod) : T(z);
    }

#define GL (k << 1)
#define GR (k << 1 | 1)

    void pushup(int k) {
        t[k].w = add_mod(t[GL].w, t[GR].w);
    }
    void apply(node &p, T addv, T mulv) { // 对节点整段施加：x = x * mul + add
        T len = (T)(p.r - p.l + 1); // addv、mulv 已按当前模式规范化
        p.w = add_mod(mul_mod(p.w, mulv), mul_mod(norm(len), addv));
        p.add = add_mod(mul_mod(p.add, mulv), addv);
        p.mul = mul_mod(p.mul, mulv);
    }
    void pushdown(int k) {
        if (t[k].add == T{} && t[k].mul == T{1}) return;
        apply(t[GL], t[k].add, t[k].mul);
        apply(t[GR], t[k].add, t[k].mul);
        t[k].add = T{};
        t[k].mul = T{1};
    }
    void build(int l, int r, int k = 1) {
        t[k].l = l;
        t[k].r = r;
        t[k].add = T{};
        t[k].mul = T{1};
        if (l == r) {
            t[k].w = norm(a[l]);
            return;
        }
        int mid = (l + r) >> 1;
        build(l, mid, GL);
        build(mid + 1, r, GR);
        pushup(k);
    }
    void upd(int l, int r, T mulv, T addv, int k = 1) {
        if (k == 1) {
            mulv = norm(mulv);
            addv = norm(addv);
        }
        if (l <= t[k].l && t[k].r <= r) {
            apply(t[k], addv, mulv);
            return;
        }
        pushdown(k);
        int mid = (t[k].l + t[k].r) >> 1;
        if (l <= mid) upd(l, r, mulv, addv, GL);
        if (mid < r) upd(l, r, mulv, addv, GR);
        pushup(k);
    }
    void range_add(int l, int r, T val, int k = 1) {
        upd(l, r, T{1}, val, k);
    }
    void range_mul(int l, int r, T val, int k = 1) {
        upd(l, r, val, T{}, k);
    }
    void range_set(int l, int r, T val, int k = 1) {
        upd(l, r, T{}, val, k);
    }
    T query_sum(int l, int r, int k = 1) { // 区间和查询
        if (l <= t[k].l && t[k].r <= r) return t[k].w;
        pushdown(k);
        int mid = (t[k].l + t[k].r) >> 1;
        T ans = T{};
        if (l <= mid) ans = add_mod(ans, query_sum(l, r, GL));
        if (mid < r) ans = add_mod(ans, query_sum(l, r, GR));
        return ans;
    }
    void point_set(int pos, T val, int k = 1) {
        range_set(pos, pos, val, k);
    }

#undef GL
#undef GR
};
```

#### 区间赋值/推平

维护未取模的区间和及最小值，支持赋值为任意 T 可表示的数（包括 0、-1）。用独立 `has_set` 标记区分“无赋值”。1-based，闭区间 [l,r]，n>0；建树 $\mathcal{O}(n)$，赋值/查询 $\mathcal{O}(\log n)$，空间 $\mathcal{O}(n)$，建议 T=ll 并保证区间和不溢出。

调用：`Segt<ll> s(vector<ll>{0,3,1,2}); s.modify(1,2,4);` 后 `s.ask(1,3)==10`、`s.rmq(1,3)==2`；重新构造清空。

```c++
template <class T> struct Segt {
    struct node {
        int l, r;
        T w, rmq, lazy;
        bool has_set;
    };
    vector<T> w;
    vector<node> t;

    Segt() = default;
    Segt(int n) : Segt(vector<T>(n + 1)) {}
    Segt(const vector<T> &in) : w(in), t(4 * in.size()) {
        int n = (int)in.size() - 1;
        if (n > 0) build(1, n);
    }

#define GL (k << 1)
#define GR (k << 1 | 1)

    void build(int l, int r, int k = 1) {
        if (l == r) {
            t[k] = {l, r, w[l], w[l], T{}, false};
            return;
        }
        t[k] = {l, r, T{}, T{}, T{}, false};
        int mid = (l + r) / 2;
        build(l, mid, GL);
        build(mid + 1, r, GR);
        pushup(k);
    }
    void pushdown(node &p, T lazy) { // 对整段施加赋值
        p.w = (p.r - p.l + 1) * lazy;
        p.rmq = lazy;
        p.lazy = lazy;
        p.has_set = true;
    }
    void pushdown(int k) {
        if (!t[k].has_set) return;
        pushdown(t[GL], t[k].lazy);
        pushdown(t[GR], t[k].lazy);
        t[k].has_set = false;
    }
    void pushup(int k) {
        t[k].w = t[GL].w + t[GR].w;
        t[k].rmq = min(t[GL].rmq, t[GR].rmq);
    }
    void modify(int l, int r, T val, int k = 1) { // 区间修改
        if (l > r) return;
        if (l <= t[k].l && t[k].r <= r) {
            pushdown(t[k], val);
            return;
        }
        pushdown(k);
        int mid = (t[k].l + t[k].r) / 2;
        if (l <= mid) modify(l, r, val, GL);
        if (mid < r) modify(l, r, val, GR);
        pushup(k);
    }
    T rmq(int l, int r, int k = 1) { // 区间询问最小值
        if (l <= t[k].l && t[k].r <= r) return t[k].rmq;
        pushdown(k);
        int mid = (t[k].l + t[k].r) / 2;
        T ans = numeric_limits<T>::max(); // RMQ -> 为 max 时需要修改为 ::lowest()
        if (l <= mid) ans = min(ans, rmq(l, r, GL)); // RMQ -> min/max
        if (mid < r) ans = min(ans, rmq(l, r, GR)); // RMQ -> min/max
        return ans;
    }
    T ask(int l, int r, int k = 1) { // 区间询问
        if (l <= t[k].l && t[k].r <= r) return t[k].w;
        pushdown(k);
        int mid = (t[k].l + t[k].r) / 2;
        T ans = 0;
        if (l <= mid) ans += ask(l, r, GL);
        if (mid < r) ans += ask(l, r, GR);
        return ans;
    }

#undef GL
#undef GR
};
```

#### 区间取模

<a id="mod-segtree"></a>

独立模板，维护非负数组的“单点赋值＋区间取模＋区间求和”，对应 [CF438D](https://codeforces.com/contest/438/problem/D)。输入 a[1..n]、a[0] 占位，n≥1；1-based 非空闭区间，模数 x>0，初值和单点赋值均非负；sum/mx 为区间和/最大值，和须在 ll 范围内。无懒标记，mx<x 时整段剪枝。

建树 $\mathcal{O}(n)$、空间 $\mathcal{O}(n)$，单点赋值/区间和 $\mathcal{O}(\log (n+1))$；**单次区间取模最坏 $\mathcal{O}(n)$**。设 q 为总操作数、p 为单点赋值次数、A 为初值和所有赋值的最大值，总时间 $\mathcal{O}((n+p)\log (A+1)\log (n+1)+q \log (n+1)+n)$。依据是每次生效的取模 x≤v 都让 v%x<v/2；单点赋值会重新增加势能，不能把每次取模都写成 $\mathcal{O}(\log n)$。

```c++
struct SegMod {
    struct Node {
        ll sum = 0, mx = 0;
    };
    int n;
    vector<Node> t;
    SegMod(const vector<ll> &a) : n((int)a.size() - 1), t(4 * a.size()) {
        build(1, 1, n, a);
    }
    void pull(int p) {
        t[p] = {t[p * 2].sum + t[p * 2 + 1].sum, max(t[p * 2].mx, t[p * 2 + 1].mx)};
    }
    void build(int p, int l, int r, const vector<ll> &a) {
        if (l == r) {
            t[p] = {a[l], a[l]};
            return;
        }
        int m = (l + r) / 2;
        build(p * 2, l, m, a);
        build(p * 2 + 1, m + 1, r, a);
        pull(p);
    }
    void set(int p, int l, int r, int x, ll v) {
        if (l == r) {
            t[p] = {v, v};
            return;
        }
        int m = (l + r) / 2;
        if (x <= m) set(p * 2, l, m, x, v);
        else set(p * 2 + 1, m + 1, r, x, v);
        pull(p);
    }
    void set(int x, ll v) {
        set(1, 1, n, x, v);
    }
    void mod(int p, int l, int r, int ql, int qr, ll x) {
        if (qr < l || r < ql || t[p].mx < x) return;
        if (l == r) {
            t[p].sum %= x;
            t[p].mx = t[p].sum;
            return;
        }
        int m = (l + r) / 2;
        mod(p * 2, l, m, ql, qr, x);
        mod(p * 2 + 1, m + 1, r, ql, qr, x);
        pull(p);
    }
    void mod(int l, int r, ll x) {
        mod(1, 1, n, l, r, x);
    }
    ll ask(int p, int l, int r, int ql, int qr) const {
        if (qr < l || r < ql) return 0;
        if (ql <= l && r <= qr) return t[p].sum;
        int m = (l + r) / 2;
        return ask(p * 2, l, m, ql, qr) + ask(p * 2 + 1, m + 1, r, ql, qr);
    }
    ll ask(int l, int r) const {
        return ask(1, 1, n, l, r);
    }
};
```

调用：`SegMod s(vector<ll>{0,7,2,9}); s.mod(1,3,4);` 后数组为 [3,2,1]；`s.set(2,8);` 后 `s.ask(1,3)==12`。模 1 会清零，模数大于最大值不修改；重新构造清空。

#### 区间异或修改

1-based，闭区间 [l,r]，n>0；输入值及异或掩码须在 $[0,2^{B})$ 内，默认 B=20。建树 $\mathcal{O}(nB)$，修改/查询 $\mathcal{O}(B \log n)$，空间 $\mathcal{O}(nB)$。该模板独立提供位数常量，无需外部 N 或孩子编号宏。

原题需要维护”区间异或修改+区间求和“ [See](https://codeforces.com/contest/242/problem/E) 。

```c++
struct Segt {
    static constexpr int B = 20; // 按题目值域调整位数
    struct node {
        int l, r;
        int w[B], lazy; // 注意这里为了方便计算，w 只需要存位
    };
    vector<node> t;

    Segt(const vector<int> &in) {
        int n = in.size() - 1;
        t.resize(n * 4 + 1);
        auto build = [&](auto self, int l, int r, int k = 1) {
            t[k] = {l, r}; // 前置赋值
            if (l == r) {
                for (int i = 0; i < B; i++) t[k].w[i] = in[l] >> i & 1;
                return;
            }
            int mid = (l + r) / 2;
            self(self, l, mid, (k << 1));
            self(self, mid + 1, r, (k << 1 | 1));
            pushup(k);
        };
        build(build, 1, n);
    }
    void pushdown(node &p, int lazy) { /* 【在此更新下递函数】 */
        int len = p.r - p.l + 1;
        for (int i = 0; i < B; i++)
            if (lazy >> i & 1) { // 即 p.w = (p.r - p.l + 1) - p.w;
                p.w[i] = len - p.w[i];
            }
        p.lazy ^= lazy;
    }
    void pushdown(int k) { // 【不需要动】
        if (t[k].lazy == 0) return;
        pushdown(t[(k << 1)], t[k].lazy);
        pushdown(t[(k << 1 | 1)], t[k].lazy);
        t[k].lazy = 0;
    }
    void pushup(int k) {
        auto pushup = [&](node &p, node &l, node &r) { /* 【在此更新上传函数】 */
                                                       for (int i = 0; i < B; i++) {
                                                           p.w[i] = l.w[i] + r.w[i]; // 即 p.w = l.w + r.w;
                                                       }
        };
        pushup(t[k], t[(k << 1)], t[(k << 1 | 1)]);
    }
    void modify(int l, int r, int val, int k = 1) { // 区间修改
        if (l <= t[k].l && t[k].r <= r) {
            pushdown(t[k], val);
            return;
        }
        pushdown(k);
        int mid = (t[k].l + t[k].r) / 2;
        if (l <= mid) modify(l, r, val, (k << 1));
        if (mid < r) modify(l, r, val, (k << 1 | 1));
        pushup(k);
    }
    ll ask(int l, int r, int k = 1) { // 区间求和
        if (l <= t[k].l && t[k].r <= r) {
            ll ans = 0;
            for (int i = 0; i < B; i++) ans += t[k].w[i] * (1LL << i);
            return ans;
        }
        pushdown(k);
        int mid = (t[k].l + t[k].r) / 2;
        ll ans = 0;
        if (l <= mid) ans += ask(l, r, (k << 1));
        if (mid < r) ans += ask(l, r, (k << 1 | 1));
        return ans;
    }
};
```

#### 拆位运算

备选完整示例，与上一节解决同一问题；1-based、闭区间，固定维护低 20 位，输入非负且小于 $2^{20}$。建树 $\mathcal{O}(20n)$，操作 $\mathcal{O}(20 \log n)$，空间 $\mathcal{O}(20n)$。

原题同上。使用若干棵线段树维护每一位的值，区间异或转变为区间翻转。

```c++
template <class T> struct Segt_ {
    struct node {
        int l, r;
        T w;
        bool lazy; // 注意懒标记用布尔型足以
    };
    vector<T> w;
    vector<node> t;

    Segt_() {}
    void init(const vector<int> &in) {
        int n = in.size() - 1;
        w.assign(in.begin(), in.end());
        t.resize(n * 4 + 1);
        build(1, n);
    }
    void pushdown(node &p, bool lazy = 1) { // 【在此更新下递函数】
        p.w = (p.r - p.l + 1) - p.w;
        p.lazy ^= lazy;
    }
    void pushup(node &p, node &l, node &r) { // 【在此更新上传函数】
        p.w = l.w + r.w;
    }
    void pushdown(int k) { // 【不需要动】
        if (t[k].lazy == 0) return;
        pushdown(t[(k << 1)]), pushdown(t[(k << 1 | 1)]); // 注意这里不再需要传入第二个参数
        t[k].lazy = 0;
    }
    void pushup(int k) {
        pushup(t[k], t[(k << 1)], t[(k << 1 | 1)]);
    } // 【不需要动】
    void build(int l, int r, int k = 1) {
        if (l == r) {
            t[k] = {l, r, w[l], 0}; // 注意懒标记初始为 0
            return;
        }
        t[k] = {l, r};
        int mid = (l + r) / 2;
        build(l, mid, (k << 1));
        build(mid + 1, r, (k << 1 | 1));
        pushup(k);
    }
    void reverse(int l, int r, int k = 1) { // 区间翻转
        if (l <= t[k].l && t[k].r <= r) {
            pushdown(t[k], 1);
            return;
        }
        pushdown(k);
        int mid = (t[k].l + t[k].r) / 2;
        if (l <= mid) reverse(l, r, (k << 1));
        if (mid < r) reverse(l, r, (k << 1 | 1));
        pushup(k);
    }
    T ask(int l, int r, int k = 1) { // 区间求和
        if (l <= t[k].l && t[k].r <= r) return t[k].w;
        pushdown(k);
        int mid = (t[k].l + t[k].r) / 2;
        T ans = 0;
        if (l <= mid) ans += ask(l, r, (k << 1));
        if (mid < r) ans += ask(l, r, (k << 1 | 1));
        return ans;
    }
};
signed main() {
    int n;
    cin >> n;
    vector<vector<int>> in(20, vector<int>(n + 1));
    Segt_<ll> segt[20]; // 拆位建线段树
    for (int i = 1, x; i <= n; i++) {
        cin >> x;
        for (int bit = 0; bit < 20; bit++) in[bit][i] = x >> bit & 1;
    }
    for (int i = 0; i < 20; i++) segt[i].init(in[i]);

    int m, op;
    for (cin >> m; m; m--) {
        cin >> op;
        if (op == 1) {
            int l, r;
            ll ans = 0;
            cin >> l >> r;
            for (int i = 0; i < 20; i++) ans += segt[i].ask(l, r) * (1LL << i);
            cout << ans << "\n";
        } else {
            int l, r, val;
            cin >> l >> r >> val;
            for (int i = 0; i < 20; i++)
                if (val >> i & 1) segt[i].reverse(l, r);
        }
    }
}
```

### Segment Tree Beats：区间最值修改、加法与查询

<a id="segment-tree-beats"></a>

维护 `a[1..n]`，`a[0]` 不使用，n≥1。支持 `chmin(l,r,x)` 将每个数变成 min(a[i],x)，`chmax(l,r,x)` 将每个数变成 max(a[i],x)，`add(l,r,x)` 区间加，以及 `query(l,r)` 返回 `.sum / .mn / .mx`。均为非空闭区间；`init(a)` 重建并清空标记。本模板独立复制，不依赖前面的线段树。

建树 $\mathcal{O}(n)$、空间 $\mathcal{O}(n)$；区间加和查询 $\mathcal{O}(\log n)$。包含区间加、chmin、chmax 的 q 次混合操作采用保守总界 $\mathcal{O}((n+q)\log ^{2}(n+1))$，是均摊复杂度，**单次最值修改可能 $\mathcal{O}(n)$**。只保留单向 chmin 且不含加法/chmax 时，可用更强的 $\mathcal{O}((n+q)\log (n+1))$ 总界。

关键：维护最大值 mx、严格次大值 sx、最大值个数 cx，以及最小值 mn、严格次小值 sn、最小值个数 cn。只有 `sx < x < mx` 时才能整段 chmin；`x == sx` 必须下递归，否则无法直接更新最大值个数。chmax 对称。父节点的最值本身承担限幅标记，下传顺序必须是加法、上界、下界。

使用 ±LLINF 表示不存在的严格次值。实际值、修改阈值和下传前的临时值须严格位于 (-LLINF,LLINF)，和、差值乘次数、累计加法等所有中间运算须在 ll 范围内。节点较大，按 `4*n*sizeof(Beats::Node)` 估算主体内存；启用 int 宏后计数和长度也会变成 64 位。

```c++
struct Beats {
    struct Node {
        ll sum = 0, mx = -LLINF, sx = -LLINF;
        ll mn = LLINF, sn = LLINF, tag = 0;
        int cx = 0, cn = 0, len = 0;
    };
    int n;
    vector<Node> t;
    Beats(const vector<ll> &a) {
        init(a);
    }
    Node join(const Node &a, const Node &b) {
        if (!a.len) return b;
        if (!b.len) return a;
        Node c;
        c.len = a.len + b.len;
        c.sum = a.sum + b.sum;
        c.mx = max(a.mx, b.mx);
        c.mn = min(a.mn, b.mn);
        c.cx = (a.mx == c.mx ? a.cx : 0) + (b.mx == c.mx ? b.cx : 0);
        c.cn = (a.mn == c.mn ? a.cn : 0) + (b.mn == c.mn ? b.cn : 0);
        c.sx = max(a.mx == c.mx ? a.sx : a.mx, b.mx == c.mx ? b.sx : b.mx);
        c.sn = min(a.mn == c.mn ? a.sn : a.mn, b.mn == c.mn ? b.sn : b.mn);
        return c;
    }
    void pull(int p) {
        t[p] = join(t[p * 2], t[p * 2 + 1]);
    }
    void build(int p, int l, int r, const vector<ll> &a) {
        if (l == r) {
            t[p].sum = t[p].mn = t[p].mx = a[l];
            t[p].len = t[p].cx = t[p].cn = 1;
            return;
        }
        int m = (l + r) / 2;
        build(p * 2, l, m, a);
        build(p * 2 + 1, m + 1, r, a);
        pull(p);
    }
    void init(const vector<ll> &a) {
        n = (int)a.size() - 1; /* assert(n >= 1) */;
        t.assign(4 * n, Node{});
        build(1, 1, n, a);
    }
    void putAdd(int p, ll x) {
        auto &u = t[p];
        u.sum += x * u.len;
        u.mx += x;
        u.mn += x;
        u.tag += x;
        if (u.sx != -LLINF) u.sx += x;
        if (u.sn != LLINF) u.sn += x;
    }
    void putMin(int p, ll x) { // 前提 sx < x < mx，只改最大值
        auto &u = t[p];
        u.sum += (x - u.mx) * u.cx;
        if (u.mn == u.mx) u.mn = x;
        else if (u.sn == u.mx) u.sn = x;
        u.mx = x;
    }
    void putMax(int p, ll x) { // 前提 mn < x < sn，只改最小值
        auto &u = t[p];
        u.sum += (x - u.mn) * u.cn;
        if (u.mx == u.mn) u.mx = x;
        else if (u.sx == u.mn) u.sx = x;
        u.mn = x;
    }
    void push(int p) {
        for (int c : {p * 2, p * 2 + 1}) {
            if (t[p].tag) putAdd(c, t[p].tag);
            if (t[c].mx > t[p].mx) putMin(c, t[p].mx);
            if (t[c].mn < t[p].mn) putMax(c, t[p].mn);
        }
        t[p].tag = 0;
    }
    void upd(int p, int l, int r, int ql, int qr, ll x, int op) {
        if (qr < l || r < ql) return;
        if (op == 1 && t[p].mx <= x) return;
        if (op == 2 && t[p].mn >= x) return;
        if (ql <= l && r <= qr) {
            if (op == 0) {
                putAdd(p, x);
                return;
            }
            if (op == 1 && t[p].sx < x) {
                putMin(p, x);
                return;
            }
            if (op == 2 && x < t[p].sn) {
                putMax(p, x);
                return;
            }
        }
        push(p);
        int m = (l + r) / 2;
        upd(p * 2, l, m, ql, qr, x, op);
        upd(p * 2 + 1, m + 1, r, ql, qr, x, op);
        pull(p);
    }
    void add(int l, int r, ll x) {
        /* assert(1 <= l && l <= r && r <= n) */;
        upd(1, 1, n, l, r, x, 0);
    }
    void chmin(int l, int r, ll x) {
        /* assert(1 <= l && l <= r && r <= n) */;
        upd(1, 1, n, l, r, x, 1);
    }
    void chmax(int l, int r, ll x) {
        /* assert(1 <= l && l <= r && r <= n) */;
        upd(1, 1, n, l, r, x, 2);
    }
    Node ask(int p, int l, int r, int ql, int qr) {
        if (qr < l || r < ql) return Node{};
        if (ql <= l && r <= qr) return t[p];
        push(p);
        int m = (l + r) / 2;
        return join(ask(p * 2, l, m, ql, qr), ask(p * 2 + 1, m + 1, r, ql, qr));
    }
    Node query(int l, int r) {
        /* assert(1 <= l && l <= r && r <= n) */;
        return ask(1, 1, n, l, r);
    }
};
```

### 区间历史最值线段树

<a id="history-segtree"></a>

维护 `a[1..n]`，历史包含初始状态以及每次区间修改完成后的状态。`rangeAdd/rangeSet` 支持区间加法/赋值，`query(l,r)` 返回 `.mn/.mx` 当前最小/最大值和 `.hmn/.hmx` 历史最小/最大值。建树 $\mathcal{O}(n)$，修改与查询 $\mathcal{O}(\log n)$，空间 $\mathcal{O}(n)$；`init` 会清空旧历史。所有区间均为非空闭区间。

难点不是当前值，而是懒标记下传前不能丢失修改过程中的峰值/谷值。`Tag` 把一串操作压成“最终变换＋赋值前的最大/最小增量＋赋值后的最大/最小常量”；`append(b)` 表示先执行自身，再执行 `b`。实际值、历史值、累计加法以及标记合并中的和均须在 ll 范围内。

```c++
struct HistSeg {
    struct Tag {
        ll add = 0, hiAdd = 0, loAdd = 0;
        ll setv = 0, hiSet = 0, loSet = 0;
        bool hasSet = false;
        static Tag makeAdd(ll x) {
            Tag t;
            t.add = x;
            t.hiAdd = max(0LL, x);
            t.loAdd = min(0LL, x);
            return t;
        }
        static Tag makeSet(ll x) {
            Tag t;
            t.hasSet = true;
            t.setv = t.hiSet = t.loSet = x;
            return t;
        }
        bool empty() const {
            return !hasSet && !add && !hiAdd && !loAdd;
        }
        void append(const Tag &b) { // 先执行自身，再执行 b
            if (hasSet) {
                hiSet = max(hiSet, setv + b.hiAdd);
                loSet = min(loSet, setv + b.loAdd);
                if (b.hasSet) {
                    hiSet = max(hiSet, b.hiSet);
                    loSet = min(loSet, b.loSet);
                    setv = b.setv;
                } else setv += b.add;
            } else {
                hiAdd = max(hiAdd, add + b.hiAdd);
                loAdd = min(loAdd, add + b.loAdd);
                if (b.hasSet) {
                    hasSet = true;
                    setv = b.setv;
                    hiSet = b.hiSet;
                    loSet = b.loSet;
                    add = 0;
                } else add += b.add;
            }
        }
    };
    struct Node {
        ll mn = 0, mx = 0, hmn = 0, hmx = 0;
        Tag tag;
    };
    int n = 0;
    vector<Node> t;
    HistSeg() = default;
    HistSeg(const vector<ll> &a) {
        init(a);
    }
    void init(const vector<ll> &a) {
        n = (int)a.size() - 1; /* assert(n > 0) */;
        t.assign(4 * n + 4, {});
        build(1, 1, n, a);
    }
    void build(int p, int l, int r, const vector<ll> &a) {
        if (l == r) {
            t[p].mn = t[p].mx = t[p].hmn = t[p].hmx = a[l];
            return;
        }
        int m = (l + r) / 2;
        build(p * 2, l, m, a);
        build(p * 2 + 1, m + 1, r, a);
        pull(p);
    }
    static Node merge(const Node &a, const Node &b) {
        Node c;
        c.mn = min(a.mn, b.mn);
        c.mx = max(a.mx, b.mx);
        c.hmn = min(a.hmn, b.hmn);
        c.hmx = max(a.hmx, b.hmx);
        return c;
    }
    void pull(int p) {
        auto v = merge(t[p * 2], t[p * 2 + 1]);
        t[p].mn = v.mn;
        t[p].mx = v.mx;
        t[p].hmn = v.hmn;
        t[p].hmx = v.hmx;
    }
    void apply(int p, const Tag &v) {
        auto &u = t[p];
        ll mn = u.mn, mx = u.mx;
        u.hmx = max(u.hmx, mx + v.hiAdd);
        u.hmn = min(u.hmn, mn + v.loAdd);
        if (v.hasSet) {
            u.hmx = max(u.hmx, v.hiSet);
            u.hmn = min(u.hmn, v.loSet);
            u.mn = u.mx = v.setv;
        } else u.mn += v.add, u.mx += v.add;
        u.tag.append(v);
    }
    void push(int p) {
        if (t[p].tag.empty()) return;
        apply(p * 2, t[p].tag);
        apply(p * 2 + 1, t[p].tag);
        t[p].tag = Tag{};
    }
    void rangeAdd(int l, int r, ll x) {
        /* assert(1 <= l && l <= r && r <= n) */;
        update(1, 1, n, l, r, Tag::makeAdd(x));
    }
    void rangeSet(int l, int r, ll x) {
        /* assert(1 <= l && l <= r && r <= n) */;
        update(1, 1, n, l, r, Tag::makeSet(x));
    }
    void update(int p, int l, int r, int ql, int qr, const Tag &v) {
        if (ql <= l && r <= qr) {
            apply(p, v);
            return;
        }
        push(p);
        int m = (l + r) / 2;
        if (ql <= m) update(p * 2, l, m, ql, qr, v);
        if (qr > m) update(p * 2 + 1, m + 1, r, ql, qr, v);
        pull(p);
    }
    Node query(int l, int r) {
        /* assert(1 <= l && l <= r && r <= n) */;
        return query(1, 1, n, l, r);
    }
    Node query(int p, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr) return t[p];
        push(p);
        int m = (l + r) / 2;
        if (qr <= m) return query(p * 2, l, m, ql, qr);
        if (ql > m) return query(p * 2 + 1, m + 1, r, ql, qr);
        return merge(query(p * 2, l, m, ql, qr), query(p * 2 + 1, m + 1, r, ql, qr));
    }
};
```

调用：`HistSeg seg(vector<ll>{0,5,-2,4}); seg.rangeAdd(1,2,3); seg.rangeSet(2,3,1);`。此时当前数组为 [8,1,1]；`seg.query(1,3)` 的 `mn=1,mx=8,hmn=-2,hmx=8`。即使一个整段标记先 `+5` 再 `-5`，历史最大值也不会因最终增量为 0 而丢失。

### 动态开点线段树与破坏性合并

<a id="merge-segtree"></a>

维护值域 1..V 的非负频次，V>0；所有根属于同一个 MergeSegTree 节点池，0 表示空树。`add(root,x,delta)` 单点增减，删除量不能超过当前频次；`query(root,l,r)` 查询闭区间频次和；`kth(root,k)` 返回第 k 小值域下标，调用前保证 1<=k<=当前树的总频次。频次及其和须在 ll 范围内。

单点更新、查询、第 k 小为 $\mathcal{O}(\log (V+1))$。不预建完整值域树，单点更新和分裂共 q 次时最多创建 $\mathcal{O}(q \log (V+1))$ 个节点；合并回收的节点可复用。用下标管理 vector，递归中不持有节点引用跨越扩容。`clear()` 清空节点池并使所有旧根失效。

`merge(a,b)` 将 b 的频次累加进 a，随后置 b=0。两棵树必须互不共享节点；不支持持久化，不允许将一个根复制给另一个变量后分别修改/合并。一次合并的耗时与两树重叠的非空节点数相关，重叠处的 b 节点被回收复用；一系列合法合并的这些访问次数可按创建的节点总数计摊还。

```c++
// mergeRec：动态线段树合并的递归内核；splitRec：按值域范围分裂的递归内核；splitKRec：按前 k 个元素分裂的递归内核
// split：分离指定值域范围；splitK：分离前 k 个元素
struct MergeSegTree {
    struct Node {
        int ch[2] = {};
        ll sum = 0;
    };
    int V;
    vector<Node> t = vector<Node>(1);
    vector<int> free;
    MergeSegTree(int vn) : V(vn) {
        /* assert(V > 0) */;
    } // vn：值域大小
    void clear() {
        t.assign(1, Node{});
        free.clear();
    }
    int alloc() {
        if (free.empty()) {
            t.emplace_back();
            return (int)(t.size()) - 1;
        }
        int p = free.back();
        free.pop_back();
        t[p] = Node{};
        return p;
    }
    void pull(int p) {
        t[p].sum = t[t[p].ch[0]].sum + t[t[p].ch[1]].sum;
    }
    int update(int p, int l, int r, int x, ll delta) {
        if (!p) p = alloc();
        if (l == r) {
            t[p].sum += delta; /* assert(t[p].sum >= 0) */;
            return p;
        }
        int m = l + (r - l) / 2;
        int d = x > m;
        int son = update(t[p].ch[d], d ? m + 1 : l, d ? r : m, x, delta);
        t[p].ch[d] = son;
        pull(p);
        return p;
    }
    void add(int &root, int x, ll delta) {
        /* assert(1 <= x && x <= V) */;
        root = update(root, 1, V, x, delta);
    }
    ll query(int p, int l, int r, int ql, int qr) const {
        if (!p || qr < l || r < ql) return 0;
        if (ql <= l && r <= qr) return t[p].sum;
        int m = l + (r - l) / 2;
        return query(t[p].ch[0], l, m, ql, qr) + query(t[p].ch[1], m + 1, r, ql, qr);
    }
    ll query(int root, int l, int r) const {
        /* assert(1 <= l && l <= r && r <= V) */;
        return query(root, 1, V, l, r);
    }
    int kth(int p, ll k) const {
        /* assert(1 <= k && k <= t[p].sum) */;
        int l = 1, r = V;
        while (l < r) {
            int m = l + (r - l) / 2;
            ll left = t[t[p].ch[0]].sum;
            if (k <= left) {
                p = t[p].ch[0];
                r = m;
            } else {
                k -= left;
                p = t[p].ch[1];
                l = m + 1;
            }
        }
        return l;
    }
    int mergeRec(int a, int b, int l, int r) {
        if (!a || !b) return a ? a : b;
        /* assert(a != b) */;
        if (l == r) t[a].sum += t[b].sum;
        else {
            int m = l + (r - l) / 2;
            t[a].ch[0] = mergeRec(t[a].ch[0], t[b].ch[0], l, m);
            t[a].ch[1] = mergeRec(t[a].ch[1], t[b].ch[1], m + 1, r);
            pull(a);
        }
        free.push_back(b);
        return a;
    }
    void merge(int &a, int &b) {
        /* assert(&a != &b) */;
        a = mergeRec(a, b, 1, V);
        b = 0;
    }
    // 返回 {保留部分, 取走部分}；只分配边界上的新节点。
    pair<int, int> splitRec(int p, int l, int r, int ql, int qr) {
        if (!p || qr < l || r < ql) return {p, 0};
        if (ql <= l && r <= qr) return {0, p};
        int m = l + (r - l) / 2;
        auto [a, x] = splitRec(t[p].ch[0], l, m, ql, qr);
        auto [b, y] = splitRec(t[p].ch[1], m + 1, r, ql, qr);
        t[p].ch[0] = a;
        t[p].ch[1] = b;
        pull(p);
        int keep = p;
        if (!a && !b) {
            free.push_back(p);
            keep = 0;
        }
        int take = 0;
        if (x || y) {
            take = alloc();
            t[take].ch[0] = x;
            t[take].ch[1] = y;
            pull(take);
        }
        return {keep, take};
    }
    int split(int &root, int l, int r) {
        /* assert(1 <= l && l <= r && r <= V) */;
        auto [keep, take] = splitRec(root, 1, V, l, r);
        root = keep;
        return take;
    }
    pair<int, int> splitKRec(int p, int l, int r, ll k) {
        if (!k) return {p, 0};
        if (k == t[p].sum) return {0, p};
        if (l == r) {
            int take = alloc();
            t[take].sum = k;
            t[p].sum -= k;
            return {p, take}; // 相同值的多个副本也能分到两棵树
        }
        int m = l + (r - l) / 2, lc = t[p].ch[0], rc = t[p].ch[1];
        ll count = t[lc].sum;
        int a, b, x, y;
        if (k <= count) {
            tie(a, x) = splitKRec(lc, l, m, k);
            b = rc;
            y = 0;
        } else {
            a = 0;
            x = lc;
            tie(b, y) = splitKRec(rc, m + 1, r, k - count);
        }
        t[p].ch[0] = a;
        t[p].ch[1] = b;
        pull(p);
        int take = alloc();
        t[take].ch[0] = x;
        t[take].ch[1] = y;
        pull(take);
        return {p, take};
    }
    int splitK(int &root, ll k) {
        /* assert(0 <= k && k <= t[root].sum) */;
        auto [keep, take] = splitKRec(root, 1, V, k);
        root = keep;
        return take;
    }
};
```

调用：`MergeSegTree seg(1000000000); int a=0,b=0; seg.add(a,7,2); seg.add(b,3,1); seg.merge(a,b);` 后 b=0，`seg.query(a,1,7)==3`、`seg.kth(a,2)==7`。树上合并时，孩子的根被消耗后不可继续用于独立查询。

#### 线段树分裂：按值域与前 k 个元素

<a id="split-segtree"></a>

直接复用上方 MergeSegTree，不再维护第二套节点池。`b=seg.split(a,l,r)` 从 a 取走值域闭区间 [l,r]；`b=seg.splitK(a,k)` 取走按值升序的前 k 个元素。原根 a 更新为剩余部分，返回的新根 b 与 a 不共享节点；允许 k=0 或取走全部，重复值的频次可拆到两侧。随后可以分别修改，或 `seg.merge(a,b)` 合回。

两类分裂均为 $\mathcal{O}(\log (V+1))$ 时间，至多新增 $\mathcal{O}(\log (V+1))$ 个节点。操作是破坏性的，不能保留旧根快照作为历史版本；返回根应放入空变量，避免覆盖另一棵仍需使用的树。

调用：先插入 `{2,2,5,9}`，`b=seg.splitK(a,1)` 后 b={2}、a={2,5,9}；再 `c=seg.split(a,3,8)` 后 c={5}、a={2,9}。已有“合并”接口及调用方式不变。

### 李超线段树：离散横坐标上的直线最小值

<a id="li-chao"></a>

预先收集所有可能查询的横坐标，构造时排序去重，至少一个坐标。动态加入整条直线 y=kx+b；查询横坐标必须已收集。仅覆盖这些离散位置，不支持线段限定区间插入。初始化 $\mathcal{O}(n \log n)$，加线/查询 $\mathcal{O}(\log (n+1))$，空间 $\mathcal{O}(n)$。同斜率、重复直线和负坐标均可处理；`clear()` 清除直线、保留横坐标。

k、b、x 为 ll，比较与返回值采用 __int128，避免乘加溢出；查询前检查 `!empty()`，空结构不可查询。若要打印超出 ll 范围的结果，使用[128 位 IO](<12 - 语言、标准库与输入输出.md#cpp-io>)中的函数，不要直接强转 ll。最大值问题可自行改比较方向；不要对 LLONG_MIN 直接取负。

```c++
struct LiChao {
    struct Line {
        ll k, b;
        i128 at(ll x) const {
            return (i128)k * x + b;
        }
    };
    vector<ll> xs;
    vector<Line> t;
    vector<char> used;
    LiChao(vector<ll> x) : xs(move(x)) {
        sort(xs.begin(), xs.end());
        xs.erase(unique(xs.begin(), xs.end()), xs.end());
        /* assert(!xs.empty()) */;
        t.resize(4 * xs.size() + 4);
        used.resize(t.size());
    }
    bool empty() const {
        return !used[1];
    }
    void clear() {
        fill(used.begin(), used.end(), false);
    }
    void add_line(ll k, ll b) {
        add(1, 0, (int)(xs.size()) - 1, Line{k, b});
    }
    void add(int p, int l, int r, Line v) {
        if (!used[p]) {
            t[p] = v;
            used[p] = true;
            return;
        }
        int m = (l + r) / 2;
        bool left = v.at(xs[l]) < t[p].at(xs[l]);
        bool mid = v.at(xs[m]) < t[p].at(xs[m]);
        if (mid) swap(v, t[p]);
        if (l == r) return;
        if (left != mid) add(p * 2, l, m, v);
        else add(p * 2 + 1, m + 1, r, v);
    }
    i128 query(ll x) const {
        /* assert(!empty()) */;
        auto it = lower_bound(xs.begin(), xs.end(), x);
        /* assert(it != xs.end() && *it == x) */;
        int pos = (int)(it - xs.begin()), p = 1, l = 0, r = (int)(xs.size()) - 1;
        i128 ans = t[1].at(x);
        while (true) {
            if (used[p]) ans = min(ans, t[p].at(x));
            if (l == r) break;
            int m = (l + r) / 2;
            if (pos <= m) {
                p *= 2;
                r = m;
            } else {
                p = p * 2 + 1;
                l = m + 1;
            }
        }
        return ans;
    }
};
```

调用：`LiChao seg({-2,0,3}); seg.add_line(2,1); seg.add_line(-1,4);` 后 `seg.query(3)==1`。

### 二进制 Trie（01 字典树）

<a id="binary-trie"></a>

异或线性空间与最大异或和查询另见[数学：线性基](<04 - 数学.md#xor-basis>)；字符检索见[串：字符 Trie](<10 - 串.md#character-trie>)。

连续节点池，支持重复值、删除一个副本、计数和最大异或。模板位数 1≤B≤64；`Trie` 默认 B=31，值和查询参数均须为 $[0,2^{B})$ 内的非负整数（B=64 时为整个 uint64_t）。节点编号和总频次小于 INT_MAX。`query(x)` 返回最大异或值，`argmax(x)` 返回取得该值的一个原元素；查询前须保证树非空。

插入/删除/查询 $\mathcal{O}(B)$（vector 扩容按均摊计），删除后回收空路径，节点池空间 $\mathcal{O}(1+UB)$，U 为历史最大同时存在的不同值数。`clear()` 清空并保留容量。可选构造参数用于预留插入值数量的 B 倍节点容量，默认按需增长；它只是预留，不限制后续插入数量。

```c++
// cap：预计插入元素数，用于预留节点空间
template <int B = 31> struct BinaryTrie {
    /* static_assert(1 <= B && B <= 64, "invalid template argument") */;
    struct Node {
        int ch[2] = {}, cnt = 0;
    };
    vector<Node> t = vector<Node>(1); // 0 为根，也作为孩子不存在的标记
    vector<int> free;
    BinaryTrie(int cap = 0) {
        /* assert(cap >= 0) */;
        t.reserve(1 + size_t(cap) * B);
    }
    static void check(ull x) {
        /* assert((x >> (B - 1) >> 1) == 0) */; // B=64 时也不移位 64 位
    }
    int size() const {
        return t[0].cnt;
    }
    bool empty() const {
        return size() == 0;
    }
    void clear() {
        t.assign(1, Node{});
        free.clear();
    }
    int alloc() {
        if (free.empty()) {
            t.emplace_back();
            return (int)(t.size()) - 1;
        }
        int p = free.back();
        free.pop_back();
        t[p] = Node{};
        return p;
    }
    void insert(ull x) {
        // check(x); /* assert(size() < INT_MAX) */;
        int u = 0;
        ++t[u].cnt;
        for (int i = B - 1; i >= 0; --i) {
            int b = (int)(x >> i & 1);
            if (!t[u].ch[b]) {
                int v = alloc();
                t[u].ch[b] = v;
            }
            u = t[u].ch[b];
            ++t[u].cnt;
        }
    }
    int count(ull x) const {
        check(x);
        int u = 0;
        for (int i = B - 1; i >= 0; --i) {
            u = t[u].ch[x >> i & 1];
            if (!u) return 0;
        }
        return t[u].cnt;
    }
    bool erase(ull x) {
        if (!count(x)) return false;
        array<int, B + 1> path{};
        int u = 0;
        --t[0].cnt;
        for (int dep = 1; dep <= B; ++dep) {
            u = t[u].ch[x >> (B - dep) & 1];
            path[dep] = u;
            --t[u].cnt;
        }
        for (int dep = B; dep >= 1; --dep) {
            int p = path[dep];
            if (t[p].cnt) break;
            t[path[dep - 1]].ch[x >> (B - dep) & 1] = 0;
            free.push_back(p);
        }
        return true;
    }
    ull query(ull x) const {
        check(x); /* assert(!empty()) */;
        int u = 0;
        ull ans = 0;
        for (int i = B - 1; i >= 0; --i) {
            int b = (int)(x >> i & 1), v = t[u].ch[b ^ 1];
            if (v && t[v].cnt) {
                ans |= ull(1) << i;
                u = v;
            } else u = t[u].ch[b];
        }
        return ans;
    }
    ull argmax(ull x) const {
        auto ans = query(x);
        return ans ^ x;
    }
};
using Trie = BinaryTrie<31>;
```

调用：`Trie t; t.insert(5); t.insert(5); t.insert(2); t.erase(5);` 后 `t.count(5)==1`、`t.query(3)==6`、`t.argmax(3)==5`。uint64_t 的完整值域使用 `BinaryTrie<64>`；query/argmax 直接返回 ull，调用前检查 `!t.empty()`；结果为 0 也有效。

### 可持久化 01 Trie：区间最大异或

<a id="persistent-trie"></a>

按顺序 append 构造前缀版本，输入值位数为 B（1≤B≤64），默认 31；版本 i 含前 i 个元素。`max_xor(l,r,x)` 返回第 l..r 个元素与 x 的最大异或值，位置 1-based 闭区间；`argmax` 返回一个取得最值的原元素。值使用 uint64_t，B<64 时须小于 $2^{B}$。允许重复值；查询前须保证 1≤l≤r≤size，不查询空区间。

每次 append 复制 B+1 个节点，更新/查询 $\mathcal{O}(B)$，n 个元素空间 $\mathcal{O}(nB)$，所有节点下标、频次须能存入 int。仅支持前缀追加，不能任意删除或用不相容版本作差；clear 清空所有版本。可选 cap 只预留容量，不设最大元素数。

```c++
// cap：预计插入元素数，用于预留节点空间
template <int B = 31> struct PersistentTrie {
    /* static_assert(1 <= B && B <= 64, "invalid template argument") */;
    struct Node {
        int ch[2] = {}, cnt = 0;
    };
    vector<Node> t = vector<Node>(1);
    vector<int> root{0};
    PersistentTrie(int cap = 0) {
        /* assert(cap >= 0) */;
        t.reserve(1 + size_t(cap) * (B + 1));
        root.reserve(size_t(cap) + 1);
    }
    static void check(ull x) {
        /* assert((x >> (B - 1) >> 1) == 0) */;
    }
    int size() const {
        return (int)(root.size()) - 1;
    }
    void clear() {
        t.assign(1, Node{});
        root.assign(1, 0);
    }
    int copy_node(int p) {
        Node old = t[p];
        t.push_back(old);
        return (int)(t.size()) - 1;
    }
    void append(ull x) {
        check(x);
        int old = root.back(), p = copy_node(old);
        root.push_back(p);
        ++t[p].cnt;
        for (int i = B - 1; i >= 0; --i) {
            int b = (int)(x >> i & 1), son = copy_node(t[old].ch[b]);
            old = t[old].ch[b];
            t[p].ch[b] = son;
            p = son;
            ++t[p].cnt;
        }
    }
    ull max_xor(int l, int r, ull x) const {
        check(x); /* assert(1 <= l && l <= size() + 1 && 0 <= r && r <= size()) */;
        /* assert(l <= r) */;
        int a = root[l - 1], b = root[r];
        ull ans = 0;
        for (int i = B - 1; i >= 0; --i) {
            int d = (int)(x >> i & 1) ^ 1;
            if (t[t[b].ch[d]].cnt > t[t[a].ch[d]].cnt) ans |= ull(1) << i;
            else d ^= 1;
            a = t[a].ch[d];
            b = t[b].ch[d];
        }
        return ans;
    }
    ull argmax(int l, int r, ull x) const {
        auto v = max_xor(l, r, x);
        return v ^ x;
    }
};
```

调用：`PersistentTrie<> t; t.append(5); t.append(2); t.append(5);` 后 `t.max_xor(2,3,3)==6`，`t.argmax(2,2,3)==2`。位置版本是原序列前缀长度，不是值域编号。

#### 长度受限的最大子数组异或：完整示例

依赖 PersistentTrie。输入 n、L、R 和 n 个 uint64_t 值，输出长度在 [L,R] 的非空子数组最大异或；1≤L≤R，无合法子数组输出 EMPTY。前缀 P[0] 必须插入：Trie 的第 j+1 个元素是 P[j]。右端点 i 的候选前缀下标为 [max(0,i-R),i-L]。

```c++
signed main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, L, R;
    cin >> n >> L >> R; /* assert(n >= 0 && 1 <= L && L <= R) */;
    vector<ull> p(n + 1);
    PersistentTrie<64> t;
    t.append(0);
    for (int i = 1; i <= n; ++i) {
        ull x;
        cin >> x;
        p[i] = p[i - 1] ^ x;
        t.append(p[i]);
    }
    ull ans = 0;
    for (int i = L; i <= n; ++i) {
        auto v = t.max_xor(max<int>(0, i - R) + 1, i - L + 1, p[i]);
        ans = max(ans, v);
    }
    if (n >= L) cout << ans << '\n';
    else cout << "EMPTY\n";
}
```

### 平衡树

选型：按值有序集合以 FHQ Treap 为主力，AVL 提供确定性树高保证；动态序列使用隐式 FHQ，静态数组转树见本章独立的[笛卡尔树](#cartesian-tree)。Splay、替罪羊树、SBT、WBLT 作为备选实现保留，当前集合接口高度重合。

除隐式 FHQ Treap 外，统一维护允许重复值的有序集合，键类型为 `ll`；元素总数和节点下标使用 `int`，要求总元素数小于 `INT_MAX`。下面的公共接口表及 u、U 复杂度约定适用于有序集合版本；隐式 FHQ Treap 的序列接口另见其小节。

| 接口 | 含义 |
| --- | --- |
| `insert(x)` / `erase(x)` | 插入一个 / 删除一个；删除返回是否成功 |
| `count(x)` / `contains(x)` | 出现次数 / 是否存在 |
| `less(x)` / `rank(x)` | 严格小于 x 的元素数 / 该数量加一，x 不必存在 |
| `kth(k)` | 第 k 小，k 从 1 开始，重复元素分别占位 |
| `prev(x)` / `next(x)` | 严格小于 / 严格大于 x 的最近元素 |
| `lower_bound(x)` / `upper_bound(x)` | 第一个大于等于 / 严格大于 x 的元素 |
| `min()` / `max()` | 最小 / 最大元素 |
| `rangeCnt(l,r)` | 闭区间 [l,r] 内的元素数；l>r 时返回 0 |
| `size()` / `empty()` / `clear()` | 总元素数 / 判空 / 清空 |
| `val()` | 按升序返回所有元素，包含重复值，耗时和输出空间均为 $\mathcal{O}(n)$ |

返回键值的查询直接返回 ll，保留完整的 LLONG_MIN..LLONG_MAX 值域。调用前检查：kth 需要 1<=k<=size()；min/max 需要 !empty()；prev 需要 less(x)>0；lower_bound 需要 less(x)<size()；next/upper_bound 需要 less(x)+count(x)<size()。函数内 assert 已保留为注释，前置条件由调用方保证，不能用返回值的真假判断答案是否存在。`prev/next` 对应常见题目的前驱/后继；`rank` 与 `kth` 都从 1 开始。

复杂度约定：n 为总元素数（含重复值），u 为当前不同键的数量，U 为历史最大同时存在的不同键数量；各结构的具体复杂度见代码前说明。`size/empty` 为 $\mathcal{O}(1)$，`val` 的时间和输出空间为 $\mathcal{O}(n)$。树操作的复杂度按节点分配为常数计算，`vector` 扩容的搬移成本按均摊计入。

节点用 `vector` 下标管理，删除的节点可复用，对象析构时自动释放内存，复制对象也互相独立；`clear()` 清空节点池但保留容量以便多测复用，耗时按 $\mathcal{O}(U)$ 上界计。不要从外部修改节点字段。

FHQ Treap、AVL、Splay、替罪羊树、SBT、WBLT 这六份按键维护有序集合；序列翻转、区间加和剪切拼接使用下面按位置维护的隐式 FHQ Treap。

#### FHQ Treap

<a id="ordered-treap"></a>

用途：维护可重复有序集合，支持增删查、排名、第 k 小、前驱后继及区间计数；需要按键分裂、合并子树时也很方便。重复值合并在同一节点的 `cnt` 中，按随机优先级保持平衡。`split/merge` 是修改树结构的内部辅助函数。

调用：`Treap tr; tr.insert(3); tr.insert(3); tr.insert(1); tr.erase(3);` 后 `tr.size()==2`、`tr.kth(2)==3`；查询前先检查非空、范围和答案是否存在，`tr.clear()` 清空。

复杂度：单次增删、有序查询及分裂合并的期望时间为 $\mathcal{O}(\log (u+1))$，最坏为 $\mathcal{O}(u)$；节点池空间 $\mathcal{O}(U)$。随机优先级提供期望保证，不保证每次操作都达到对数时间。

```c++
// rangeCnt：统计闭区间内元素个数
struct Treap {
    struct Node {
        ll val = 0;
        ull pri = 0;
        int ch[2] = {}, cnt = 0, sz = 0;
    };
    vector<Node> t = vector<Node>(1); // 下标 0 表示空节点
    vector<int> rec; // 已回收、可复用的节点编号
    int root = 0;

    int alloc() {
        if (rec.empty()) {
            t.emplace_back();
            return (int)(t.size()) - 1;
        }
        int p = rec.back();
        rec.pop_back();
        t[p] = Node{};
        return p;
    }
    void recycle(int p) {
        rec.push_back(p);
    }
    void clear() {
        root = 0;
        t.assign(1, Node{});
        rec.clear();
    }
    int size() const {
        return t[root].sz;
    }
    bool empty() const {
        return root == 0;
    }
    mt19937_64 rng;
    Treap(ull seed = chrono::steady_clock::now().time_since_epoch().count()) : rng(seed) {}
    int make(ll x) {
        int p = alloc();
        t[p].val = x;
        t[p].pri = rng();
        t[p].cnt = t[p].sz = 1;
        return p;
    }
    void pull(int p) {
        t[p].sz = t[t[p].ch[0]].sz + t[t[p].ch[1]].sz + t[p].cnt;
    }
    // eq=false 分为 <x / >=x；eq=true 分为 <=x / >x，避免 x±1 溢出。
    pair<int, int> split(int p, ll x, bool eq) {
        if (!p) return {0, 0};
        if (t[p].val < x || (eq && t[p].val == x)) {
            auto [a, b] = split(t[p].ch[1], x, eq);
            t[p].ch[1] = a;
            pull(p);
            return {p, b};
        }
        auto [a, b] = split(t[p].ch[0], x, eq);
        t[p].ch[0] = b;
        pull(p);
        return {a, p};
    }
    // 内部合并：同一节点池，且 a 中所有键严格小于 b 中所有键。
    int merge(int a, int b) {
        if (!a || !b) return a ? a : b;
        if (t[a].pri > t[b].pri) {
            int c = merge(t[a].ch[1], b);
            t[a].ch[1] = c;
            pull(a);
            return a;
        }
        int c = merge(a, t[b].ch[0]);
        t[b].ch[0] = c;
        pull(b);
        return b;
    }
    void insert(ll x) {
        auto [a, b] = split(root, x, false);
        auto [m, c] = split(b, x, true);
        if (!m) m = make(x);
        else {
            ++t[m].cnt;
            pull(m);
        }
        root = merge(merge(a, m), c);
    }
    bool erase(ll x) {
        auto [a, b] = split(root, x, false);
        auto [m, c] = split(b, x, true);
        bool found = m != 0;
        if (m && --t[m].cnt == 0) {
            recycle(m);
            m = 0;
        } else if (m) pull(m);
        root = merge(merge(a, m), c);
        return found;
    }
    int count(ll x) const {
        int p = root;
        while (p && t[p].val != x) p = t[p].ch[x > t[p].val];
        return t[p].cnt;
    }
    int less(ll x) const { // 严格小于 x 的元素个数
        int p = root, ans = 0;
        while (p)
            if (x <= t[p].val) p = t[p].ch[0];
            else {
                ans += t[t[p].ch[0]].sz + t[p].cnt;
                p = t[p].ch[1];
            }
        return ans;
    }
    ll kth(int k) const {
        /* assert(1 <= k && k <= size()) */;
        int p = root;
        while (p) {
            int l = t[t[p].ch[0]].sz;
            if (k <= l) p = t[p].ch[0];
            else if (k <= l + t[p].cnt) return t[p].val;
            else {
                k -= l + t[p].cnt;
                p = t[p].ch[1];
            }
        }
        /* assert(false) */;
        return 0; // 合法树和合法 k 不会走到这里
    }
    vector<ll> val() const { // 非递归中序遍历，包含重复值
        vector<ll> ans;
        vector<int> st;
        int p = root;
        while (p || !st.empty()) {
            while (p) {
                st.push_back(p);
                p = t[p].ch[0];
            }
            p = st.back();
            st.pop_back();
            ans.insert(ans.end(), t[p].cnt, t[p].val);
            p = t[p].ch[1];
        }
        return ans;
    }
    bool contains(ll x) const {
        return count(x) != 0;
    }
    int rank(ll x) const {
        return less(x) + 1;
    }
    ll prev(ll x) const {
        return kth(less(x));
    }
    ll next(ll x) const {
        return upper_bound(x);
    }
    ll lower_bound(ll x) const {
        return kth(less(x) + 1);
    }
    ll upper_bound(ll x) const {
        return kth(less(x) + count(x) + 1);
    }
    ll min() const {
        return kth(1);
    }
    ll max() const {
        return kth(size());
    }
    int rangeCnt(ll l, ll r) const {
        return l > r ? 0 : less(r) + count(r) - less(l);
    }
};
```

#### 隐式 FHQ Treap

<a id="implicit-treap"></a>

用途：维护动态序列，支持整段插入、删除、区间翻转、区间加、区间求和、按位置查询与赋值、剪切拼接。节点按序列位置组织，位置由左子树大小推算；相同值也分别占用节点，不按值排序。

复杂度：设当前序列长 n，本次处理段长 m，历史最大序列长 N。单点操作、区间加/翻转/求和及剪切拼接期望 $\mathcal{O}(\log (n+1))$，最坏 $\mathcal{O}(n)$；插入一段期望 $\mathcal{O}(m+\log (n+1))$，删除一段因回收节点也为期望 $\mathcal{O}(m+\log (n+1))$。空树上建树 $\mathcal{O}(m)$，`val()` $\mathcal{O}(n)$，`size/empty` $\mathcal{O}(1)$，`clear()` 按 $\mathcal{O}(N)$ 上界计；重新 `build` 还需计入清空旧树的开销。节点池空间 $\mathcal{O}(N)$，批量建树及删除使用 $\mathcal{O}(m)$ 辅助空间；分裂合并的递归栈期望 $\mathcal{O}(\log (n+1))$、最坏 $\mathcal{O}(n)$。节点分配的扩容开销按均摊计入。

约定：元素位置从 1 开始，区间为闭区间 `[l,r]`。插入的 `pos` 表示前面保留的元素数量，0 表示头插，n 表示尾插；剪切拼接的 `pos` 按删除该段后的序列计算。例如 `[1,2,3,4,5]` 执行 `cut_paste(2,3,3)` 得到 `[1,4,5,2,3]`。非法位置或区间不修改序列，更新返回 `false`，get/sum 直接返回 ll，查询前必须保证 1<=k<=size() 或 1<=l<=r<=size()；空数组可建树，合法位置插入空段返回 `true`。元素总数小于 `INT_MAX`；值、区间和、累计加法标记及计算中间量须在 `ll` 范围内。

| 接口 | 含义 |
| --- | --- |
| `build(a)` | 清空旧序列并线性建树，a 为普通 0-based 数组 |
| `insert(pos,x)` / `insert(pos,a)` | 在前 pos 个元素后插入一个值 / 一段序列 |
| `erase(l,r)` | 删除并回收第 l 到 r 个元素 |
| `get(k)` / `set(k,x)` | 查询 / 赋值第 k 个元素 |
| `reverse(l,r)` / `add(l,r,d)` | 区间翻转 / 区间所有值加 d |
| `sum(l,r)` | 查询区间和，直接返回 ll，要求 1<=l<=r<=size() |
| `cut_paste(l,r,pos)` | 将该段移动到剩余序列的前 pos 个元素后 |
| `size()` / `empty()` / `clear()` / `val()` | 长度 / 判空 / 清空 / 按当前顺序导出 |

查询与导出也可能下传懒标记；`split/merge` 是同一节点池内的辅助函数，不直接混用不同对象的节点下标。

调用：`ImplicitTreap tr; tr.build({1,2,3}); tr.reverse(1,3); tr.add(2,3,5);` 后序列为 [3,7,6]，`tr.sum(1,3)==16`；`tr.clear()` 清空。

```c++
// putRev：对隐式 Treap 节点施加翻转标记；buildPart：线性构造一段隐式 Treap；recycle：回收子树全部节点
struct ImplicitTreap {
    struct Node {
        ll val = 0, sum = 0, add = 0;
        ull pri = 0;
        int ch[2] = {}, sz = 0;
        bool rev = false;
    };
    vector<Node> t = vector<Node>(1);
    vector<int> rec; // 已回收、可复用的节点编号
    int root = 0;
    mt19937_64 rng;

    ImplicitTreap(ull seed = chrono::steady_clock::now().time_since_epoch().count()) : rng(seed) {}
    int size() const {
        return t[root].sz;
    }
    bool empty() const {
        return root == 0;
    }
    void clear() {
        root = 0;
        t.assign(1, Node{});
        rec.clear();
    }
    int make(ll x) {
        int p;
        if (rec.empty()) {
            p = (int)(t.size());
            t.emplace_back();
        } else {
            p = rec.back();
            rec.pop_back();
            t[p] = Node{};
        }
        t[p].val = t[p].sum = x;
        t[p].sz = 1;
        t[p].pri = rng();
        return p;
    }
    void pull(int p) {
        int l = t[p].ch[0], r = t[p].ch[1];
        t[p].sz = t[l].sz + t[r].sz + 1;
        t[p].sum = t[l].sum + t[r].sum + t[p].val;
    }
    void apply_add(int p, ll d) {
        if (!p) return;
        t[p].val += d;
        t[p].sum += d * t[p].sz;
        t[p].add += d;
    }
    void putRev(int p) {
        if (!p) return;
        swap(t[p].ch[0], t[p].ch[1]);
        t[p].rev = !t[p].rev;
    }
    void push(int p) { // 改变孩子连接前下传，避免旧标记作用到新接入的节点
        if (t[p].rev) {
            putRev(t[p].ch[0]);
            putRev(t[p].ch[1]);
            t[p].rev = false;
        }
        if (t[p].add) {
            apply_add(t[p].ch[0], t[p].add);
            apply_add(t[p].ch[1], t[p].add);
            t[p].add = 0;
        }
    }
    pair<int, int> split(int p, int k) { // 分成前 k 个与剩余元素
        if (!p) return {0, 0};
        if (k <= 0) return {0, p};
        if (k >= t[p].sz) return {p, 0};
        push(p);
        int l = t[t[p].ch[0]].sz;
        if (k <= l) {
            auto [a, b] = split(t[p].ch[0], k);
            t[p].ch[0] = b;
            pull(p);
            return {a, p};
        }
        auto [a, b] = split(t[p].ch[1], k - l - 1);
        t[p].ch[1] = a;
        pull(p);
        return {p, b};
    }
    int merge(int a, int b) { // 将 b 接在 a 后，不要求值有序
        if (!a || !b) return a ? a : b;
        if (t[a].pri >= t[b].pri) {
            push(a);
            t[a].ch[1] = merge(t[a].ch[1], b);
            pull(a);
            return a;
        }
        push(b);
        t[b].ch[0] = merge(a, t[b].ch[0]);
        pull(b);
        return b;
    }
    int buildPart(const vector<ll> &a) { // 单调栈按优先级 O(m) 建笛卡尔树
        vector<int> st;
        st.reserve(a.size());
        for (ll x : a) {
            int p = make(x), last = 0;
            while (!st.empty() && t[st.back()].pri < t[p].pri) {
                last = st.back();
                st.pop_back();
                pull(last);
            }
            t[p].ch[0] = last;
            if (!st.empty()) t[st.back()].ch[1] = p;
            st.push_back(p);
        }
        for (int i = (int)(st.size()) - 1; i >= 0; --i) pull(st[i]);
        return st.empty() ? 0 : st.front();
    }
    void build(const vector<ll> &a) {
        clear();
        root = buildPart(a);
    }
    void recycle(int p) { // 非递归回收，未下传的标记不影响节点可达性
        vector<int> st;
        if (p) st.push_back(p);
        while (!st.empty()) {
            int q = st.back();
            st.pop_back();
            for (int c : t[q].ch)
                if (c) st.push_back(c);
            rec.push_back(q);
        }
    }
    bool valid(int l, int r) const {
        return 1 <= l && l <= r && r <= size();
    }
    bool insert(int pos, ll x) {
        if (pos < 0 || pos > size()) return false;
        int m = make(x);
        auto [a, b] = split(root, pos);
        root = merge(merge(a, m), b);
        return true;
    }
    bool insert(int pos, const vector<ll> &v) {
        if (pos < 0 || pos > size()) return false;
        int m = buildPart(v);
        auto [a, b] = split(root, pos);
        root = merge(merge(a, m), b);
        return true;
    }
    bool erase(int l, int r) {
        if (!valid(l, r)) return false;
        auto [a, b] = split(root, l - 1);
        auto [m, c] = split(b, r - l + 1);
        root = merge(a, c);
        recycle(m);
        return true;
    }
    ll get(int k) {
        /* assert(1 <= k && k <= size()) */;
        int p = root;
        while (p) {
            push(p);
            int l = t[t[p].ch[0]].sz;
            if (k <= l) p = t[p].ch[0];
            else if (k == l + 1) return t[p].val;
            else {
                k -= l + 1;
                p = t[p].ch[1];
            }
        }
        /* assert(false) */;
        return 0; // 合法树和合法 k 不会走到这里
    }
    bool set(int k, ll x) {
        if (k < 1 || k > size()) return false;
        auto [a, b] = split(root, k - 1);
        auto [m, c] = split(b, 1);
        t[m].val = t[m].sum = x;
        t[m].add = 0;
        t[m].rev = false;
        root = merge(merge(a, m), c);
        return true;
    }
    bool reverse(int l, int r) {
        if (!valid(l, r)) return false;
        auto [a, b] = split(root, l - 1);
        auto [m, c] = split(b, r - l + 1);
        putRev(m);
        root = merge(merge(a, m), c);
        return true;
    }
    bool add(int l, int r, ll d) {
        if (!valid(l, r)) return false;
        auto [a, b] = split(root, l - 1);
        auto [m, c] = split(b, r - l + 1);
        apply_add(m, d);
        root = merge(merge(a, m), c);
        return true;
    }
    ll sum(int l, int r) {
        /* assert(valid(l, r)) */;
        auto [a, b] = split(root, l - 1);
        auto [m, c] = split(b, r - l + 1);
        ll ans = t[m].sum;
        root = merge(merge(a, m), c);
        return ans;
    }
    bool cut_paste(int l, int r, int pos) {
        if (!valid(l, r) || pos < 0 || pos > size() - (r - l + 1)) return false;
        auto [a, b] = split(root, l - 1);
        auto [m, c] = split(b, r - l + 1);
        int rest = merge(a, c);
        auto [x, y] = split(rest, pos);
        root = merge(merge(x, m), y);
        return true;
    }
    vector<ll> val() {
        vector<ll> ans;
        vector<int> st;
        int p = root;
        while (p || !st.empty()) {
            while (p) {
                push(p);
                st.push_back(p);
                p = t[p].ch[0];
            }
            p = st.back();
            st.pop_back();
            ans.push_back(t[p].val);
            p = t[p].ch[1];
        }
        return ans;
    }
};
```

#### AVL平衡树

用途：解决普通平衡树六操作题型，即插入、删除、排名、第 k 小、前驱、后继；同时支持重复计数、边界查询和区间计数。“普通平衡树”并不是唯一的树种，这里采用 AVL，适合需要确定性树高保证的场景。左右子树高度差不超过 1，插入和删除后都旋转修复。

复杂度：单次增删和有序查询的树操作最坏 $\mathcal{O}(\log (u+1))$；节点池空间 $\mathcal{O}(U)$，递归栈 $\mathcal{O}(\log (u+1))$。

```c++
// rangeCnt：统计闭区间内元素个数
struct AVL {
    struct Node {
        ll val = 0;
        int ch[2] = {}, cnt = 0, sz = 0, h = 0;
    };
    vector<Node> t = vector<Node>(1); // 下标 0 表示空节点
    vector<int> rec; // 已回收、可复用的节点编号
    int root = 0;

    int alloc() {
        if (rec.empty()) {
            t.emplace_back();
            return (int)(t.size()) - 1;
        }
        int p = rec.back();
        rec.pop_back();
        t[p] = Node{};
        return p;
    }
    void recycle(int p) {
        rec.push_back(p);
    }
    void clear() {
        root = 0;
        t.assign(1, Node{});
        rec.clear();
    }
    int size() const {
        return t[root].sz;
    }
    bool empty() const {
        return root == 0;
    }
    int make(ll x) {
        int p = alloc();
        t[p].val = x;
        t[p].cnt = t[p].sz = t[p].h = 1;
        return p;
    }
    void pull(int p) {
        int l = t[p].ch[0], r = t[p].ch[1];
        t[p].sz = t[l].sz + t[r].sz + t[p].cnt;
        t[p].h = std::max(t[l].h, t[r].h) + 1;
    }
    int rotate(int p, int d) { // 将 d 侧孩子旋转到根
        int q = t[p].ch[d];
        t[p].ch[d] = t[q].ch[d ^ 1];
        t[q].ch[d ^ 1] = p;
        pull(p);
        pull(q);
        return q;
    }
    int balance(int p) {
        pull(p);
        for (int d = 0; d < 2; ++d) {
            int q = t[p].ch[d], other = t[p].ch[d ^ 1];
            if (t[q].h - t[other].h > 1) {
                if (t[t[q].ch[d]].h < t[t[q].ch[d ^ 1]].h) t[p].ch[d] = rotate(q, d ^ 1);
                return rotate(p, d);
            }
        }
        return p;
    }
    int add(int p, ll x) {
        if (!p) return make(x);
        if (x == t[p].val) ++t[p].cnt;
        else {
            int d = x > t[p].val;
            int q = add(t[p].ch[d], x); // 不持有 vector 元素的引用跨越分配
            t[p].ch[d] = q;
        }
        return balance(p);
    }
    int take_min(int p, int &q) { // 整体摘下最小键节点，保留其 cnt
        if (!t[p].ch[0]) {
            q = p;
            return t[p].ch[1];
        }
        t[p].ch[0] = take_min(t[p].ch[0], q);
        return balance(p);
    }
    int remove(int p, ll x, bool &found) {
        if (!p) return 0;
        if (x != t[p].val) {
            int d = x > t[p].val;
            t[p].ch[d] = remove(t[p].ch[d], x, found);
        } else {
            found = true;
            if (t[p].cnt > 1) --t[p].cnt;
            else if (!t[p].ch[0] || !t[p].ch[1]) {
                int q = t[p].ch[0] ? t[p].ch[0] : t[p].ch[1];
                recycle(p);
                return q;
            } else {
                int q = 0;
                t[p].ch[1] = take_min(t[p].ch[1], q);
                t[p].val = t[q].val;
                t[p].cnt = t[q].cnt;
                recycle(q);
            }
        }
        return balance(p);
    }
    void insert(ll x) {
        root = add(root, x);
    }
    bool erase(ll x) {
        bool found = false;
        root = remove(root, x, found);
        return found;
    }
    int count(ll x) const {
        int p = root;
        while (p && t[p].val != x) p = t[p].ch[x > t[p].val];
        return t[p].cnt;
    }
    int less(ll x) const { // 严格小于 x 的元素个数
        int p = root, ans = 0;
        while (p)
            if (x <= t[p].val) p = t[p].ch[0];
            else {
                ans += t[t[p].ch[0]].sz + t[p].cnt;
                p = t[p].ch[1];
            }
        return ans;
    }
    ll kth(int k) const {
        /* assert(1 <= k && k <= size()) */;
        int p = root;
        while (p) {
            int l = t[t[p].ch[0]].sz;
            if (k <= l) p = t[p].ch[0];
            else if (k <= l + t[p].cnt) return t[p].val;
            else {
                k -= l + t[p].cnt;
                p = t[p].ch[1];
            }
        }
        /* assert(false) */;
        return 0; // 合法树和合法 k 不会走到这里
    }
    vector<ll> val() const { // 非递归中序遍历，包含重复值
        vector<ll> ans;
        vector<int> st;
        int p = root;
        while (p || !st.empty()) {
            while (p) {
                st.push_back(p);
                p = t[p].ch[0];
            }
            p = st.back();
            st.pop_back();
            ans.insert(ans.end(), t[p].cnt, t[p].val);
            p = t[p].ch[1];
        }
        return ans;
    }
    bool contains(ll x) const {
        return count(x) != 0;
    }
    int rank(ll x) const {
        return less(x) + 1;
    }
    ll prev(ll x) const {
        return kth(less(x));
    }
    ll next(ll x) const {
        return upper_bound(x);
    }
    ll lower_bound(ll x) const {
        return kth(less(x) + 1);
    }
    ll upper_bound(ll x) const {
        return kth(less(x) + count(x) + 1);
    }
    ll min() const {
        return kth(1);
    }
    ll max() const {
        return kth(size());
    }
    int rangeCnt(ll l, ll r) const {
        return l > r ? 0 : less(r) + count(r) - less(l);
    }
};
```

#### Splay 树

定位：备选/学习实现；常规有序集合题不必与 FHQ、AVL 同时准备。

用途：维护可重复有序集合，支持增删查、排名、第 k 小、前驱后继及区间计数，适合热点键被反复访问的场景。每次访问后将相应节点伸展到根，查询也会改变树形；查找失败时伸展最后访问的节点，无须随机数或无穷哨兵。

复杂度：单次增删和有序查询均摊 $\mathcal{O}(\log (u+1))$，最坏 $\mathcal{O}(u)$；节点池空间 $\mathcal{O}(U)$。树高本身可能达到 $\mathcal{O}(u)$，保证针对一串操作的均摊开销，而非每次操作。

```c++
// rangeCnt：统计闭区间内元素个数
struct Splay {
    struct Node {
        ll val = 0;
        int ch[2] = {}, fa = 0, cnt = 0, sz = 0;
    };
    vector<Node> t = vector<Node>(1); // 下标 0 表示空节点
    vector<int> rec; // 已回收、可复用的节点编号
    int root = 0;

    int alloc() {
        if (rec.empty()) {
            t.emplace_back();
            return (int)(t.size()) - 1;
        }
        int p = rec.back();
        rec.pop_back();
        t[p] = Node{};
        return p;
    }
    void recycle(int p) {
        rec.push_back(p);
    }
    void clear() {
        root = 0;
        t.assign(1, Node{});
        rec.clear();
    }
    int size() const {
        return t[root].sz;
    }
    bool empty() const {
        return root == 0;
    }
    int make(ll x, int fa) {
        int p = alloc();
        t[p].val = x;
        t[p].fa = fa;
        t[p].cnt = t[p].sz = 1;
        return p;
    }
    void pull(int p) {
        t[p].sz = t[t[p].ch[0]].sz + t[t[p].ch[1]].sz + t[p].cnt;
    }
    void rotate(int x) {
        int y = t[x].fa, z = t[y].fa, d = t[y].ch[1] == x;
        int b = t[x].ch[d ^ 1];
        if (z) t[z].ch[t[z].ch[1] == y] = x;
        else root = x;
        t[x].fa = z;
        t[y].ch[d] = b;
        if (b) t[b].fa = y;
        t[x].ch[d ^ 1] = y;
        t[y].fa = x;
        pull(y);
        pull(x);
    }
    void splay(int x) {
        while (t[x].fa) {
            int y = t[x].fa, z = t[y].fa;
            if (z) rotate((t[y].ch[1] == x) == (t[z].ch[1] == y) ? y : x);
            rotate(x);
        }
        root = x;
    }
    // 查找失败也伸展最后访问的节点，保持查询的均摊复杂度。
    int find_node(ll x) {
        int p = root, last = 0;
        while (p && t[p].val != x) {
            last = p;
            p = t[p].ch[x > t[p].val];
        }
        if (p) splay(p);
        else if (last) splay(last);
        return p;
    }
    void insert(ll x) {
        if (!root) {
            root = make(x, 0);
            return;
        }
        int p = root, fa = 0;
        while (p && t[p].val != x) {
            fa = p;
            p = t[p].ch[x > t[p].val];
        }
        if (p) ++t[p].cnt;
        else {
            int d = x > t[fa].val;
            p = make(x, fa);
            t[fa].ch[d] = p;
        }
        for (int q = p; q; q = t[q].fa) pull(q);
        splay(p);
    }
    bool erase(ll x) {
        int p = find_node(x);
        if (!p) return false;
        if (t[p].cnt > 1) {
            --t[p].cnt;
            pull(p);
            return true;
        }
        int l = t[p].ch[0], r = t[p].ch[1];
        if (l) t[l].fa = 0;
        if (r) t[r].fa = 0;
        recycle(p);
        root = l;
        if (!l) root = r;
        else {
            int q = l;
            while (t[q].ch[1]) q = t[q].ch[1];
            splay(q);
            t[root].ch[1] = r;
            if (r) t[r].fa = root;
            pull(root);
        }
        return true;
    }
    int count(ll x) {
        int p = find_node(x);
        return t[p].cnt;
    }
    int less(ll x) {
        int p = root, last = 0, ans = 0;
        while (p) {
            last = p;
            if (x <= t[p].val) p = t[p].ch[0];
            else {
                ans += t[t[p].ch[0]].sz + t[p].cnt;
                p = t[p].ch[1];
            }
        }
        if (last) splay(last);
        return ans;
    }
    ll kth(int k) {
        /* assert(1 <= k && k <= size()) */;
        int p = root;
        while (p) {
            int l = t[t[p].ch[0]].sz;
            if (k <= l) p = t[p].ch[0];
            else if (k <= l + t[p].cnt) {
                splay(p);
                return t[p].val;
            } else {
                k -= l + t[p].cnt;
                p = t[p].ch[1];
            }
        }
        /* assert(false) */;
        return 0; // 合法树和合法 k 不会走到这里
    }
    vector<ll> val() const { // 非递归中序遍历，包含重复值
        vector<ll> ans;
        vector<int> st;
        int p = root;
        while (p || !st.empty()) {
            while (p) {
                st.push_back(p);
                p = t[p].ch[0];
            }
            p = st.back();
            st.pop_back();
            ans.insert(ans.end(), t[p].cnt, t[p].val);
            p = t[p].ch[1];
        }
        return ans;
    }
    bool contains(ll x) {
        return count(x) != 0;
    }
    int rank(ll x) {
        return less(x) + 1;
    }
    ll prev(ll x) {
        return kth(less(x));
    }
    ll next(ll x) {
        return upper_bound(x);
    }
    ll lower_bound(ll x) {
        return kth(less(x) + 1);
    }
    ll upper_bound(ll x) {
        return kth(less(x) + count(x) + 1);
    }
    ll min() {
        return kth(1);
    }
    ll max() {
        return kth(size());
    }
    int rangeCnt(ll l, ll r) {
        return l > r ? 0 : less(r) + count(r) - less(l);
    }
};
```

#### 替罪羊树

定位：备选/学习实现；常规有序集合题不必与 FHQ、AVL 同时准备。

用途：通过局部重建维护可重复有序集合，支持增删查、排名、第 k 小、前驱后继及区间计数，不依赖旋转或随机优先级。使用 alpha=3/4 和惰性删除；子树偏斜或已删除节点过多时重建并回收这些节点，随后继续检查祖先。

复杂度：有序查询最坏 $\mathcal{O}(\log (u+1))$；插入、删除均摊 $\mathcal{O}(\log (u+1))$，触发重建的单次操作最坏 $\mathcal{O}(u)$。节点池空间 $\mathcal{O}(U)$，一次重建额外需要 $\mathcal{O}(u)$ 空间。`sz` 统计有效元素数，`nodes` 统计结构节点数，`live` 统计有效不同键数；重复次数不参与形状平衡。

```c++
// rangeCnt：统计闭区间内元素个数
struct Scapegoat {
    struct Node {
        ll val = 0;
        int ch[2] = {}, cnt = 0, sz = 0, nodes = 0, live = 0;
    };
    vector<Node> t = vector<Node>(1); // 下标 0 表示空节点
    vector<int> rec; // 已回收、可复用的节点编号
    int root = 0;

    int alloc() {
        if (rec.empty()) {
            t.emplace_back();
            return (int)(t.size()) - 1;
        }
        int p = rec.back();
        rec.pop_back();
        t[p] = Node{};
        return p;
    }
    void recycle(int p) {
        rec.push_back(p);
    }
    void clear() {
        root = 0;
        t.assign(1, Node{});
        rec.clear();
    }
    int size() const {
        return t[root].sz;
    }
    bool empty() const {
        return size() == 0;
    }
    int make(ll x) {
        int p = alloc();
        t[p].val = x;
        t[p].cnt = t[p].sz = t[p].nodes = t[p].live = 1;
        return p;
    }
    void pull(int p) {
        int l = t[p].ch[0], r = t[p].ch[1];
        t[p].sz = t[l].sz + t[r].sz + t[p].cnt;
        t[p].nodes = t[l].nodes + t[r].nodes + 1;
        t[p].live = t[l].live + t[r].live + (t[p].cnt > 0);
    }
    bool bad(int p) const { // alpha=3/4；重复次数不参与结构平衡
        return 4LL * std::max(t[t[p].ch[0]].nodes, t[t[p].ch[1]].nodes) > 3LL * t[p].nodes ||
               4LL * t[p].live < 3LL * t[p].nodes;
    }
    void collect(int p, vector<int> &a) {
        if (!p) return;
        collect(t[p].ch[0], a);
        if (t[p].cnt) a.push_back(p);
        else recycle(p);
        collect(t[p].ch[1], a);
    }
    int build(const vector<int> &a, int l, int r) {
        if (l >= r) return 0;
        int m = l + (r - l) / 2, p = a[m];
        t[p].ch[0] = build(a, l, m);
        t[p].ch[1] = build(a, m + 1, r);
        pull(p);
        return p;
    }
    int rebuild(int p) {
        vector<int> a;
        a.reserve(t[p].live);
        collect(p, a);
        return build(a, 0, (int)(a.size()));
    }
    bool update(ll x, bool ins) { // ins：true 插入，false 删除
        vector<int> path;
        int p = root;
        while (p) {
            path.push_back(p);
            if (t[p].val == x) break;
            p = t[p].ch[x > t[p].val];
        }
        if (ins)
            if (p) ++t[p].cnt; // 也可重新启用已惰性删除的节点
            else {
                p = make(x);
                if (path.empty()) root = p;
                else {
                    int fa = path.back();
                    t[fa].ch[x > t[fa].val] = p;
                }
                path.push_back(p);
            }
        else {
            if (!p || !t[p].cnt) return false;
            --t[p].cnt;
        }
        // 清理墓碑会改变祖先的结构大小，须从下到上继续检查祖先。
        for (int i = (int)(path.size()) - 1; i >= 0; --i) {
            pull(path[i]);
            if (!bad(path[i])) continue;
            int q = rebuild(path[i]);
            if (i == 0) root = q;
            else {
                int fa = path[i - 1], d = t[fa].ch[1] == path[i];
                t[fa].ch[d] = q;
            }
        }
        return true;
    }
    void insert(ll x) {
        update(x, true);
    }
    bool erase(ll x) {
        return update(x, false);
    }
    int count(ll x) const {
        int p = root;
        while (p && t[p].val != x) p = t[p].ch[x > t[p].val];
        return t[p].cnt;
    }
    int less(ll x) const { // 严格小于 x 的元素个数
        int p = root, ans = 0;
        while (p)
            if (x <= t[p].val) p = t[p].ch[0];
            else {
                ans += t[t[p].ch[0]].sz + t[p].cnt;
                p = t[p].ch[1];
            }
        return ans;
    }
    ll kth(int k) const {
        /* assert(1 <= k && k <= size()) */;
        int p = root;
        while (p) {
            int l = t[t[p].ch[0]].sz;
            if (k <= l) p = t[p].ch[0];
            else if (k <= l + t[p].cnt) return t[p].val;
            else {
                k -= l + t[p].cnt;
                p = t[p].ch[1];
            }
        }
        /* assert(false) */;
        return 0; // 合法树和合法 k 不会走到这里
    }
    vector<ll> val() const { // 非递归中序遍历，包含重复值
        vector<ll> ans;
        vector<int> st;
        int p = root;
        while (p || !st.empty()) {
            while (p) {
                st.push_back(p);
                p = t[p].ch[0];
            }
            p = st.back();
            st.pop_back();
            ans.insert(ans.end(), t[p].cnt, t[p].val);
            p = t[p].ch[1];
        }
        return ans;
    }
    bool contains(ll x) const {
        return count(x) != 0;
    }
    int rank(ll x) const {
        return less(x) + 1;
    }
    ll prev(ll x) const {
        return kth(less(x));
    }
    ll next(ll x) const {
        return upper_bound(x);
    }
    ll lower_bound(ll x) const {
        return kth(less(x) + 1);
    }
    ll upper_bound(ll x) const {
        return kth(less(x) + count(x) + 1);
    }
    ll min() const {
        return kth(1);
    }
    ll max() const {
        return kth(size());
    }
    int rangeCnt(ll l, ll r) const {
        return l > r ? 0 : less(r) + count(r) - less(l);
    }
};
```

#### SBT(Size Balanced Tree)

定位：备选/学习实现；常规有序集合题不必与 FHQ、AVL 同时准备。

用途：以子树节点数维护可重复有序集合，支持增删查、排名、第 k 小、前驱后继及区间计数。通过旋转，使每个孙子子树的节点数不超过另一侧孩子子树；`sz` 用于排名，`nodes` 用于平衡。

复杂度：有序查询最坏 $\mathcal{O}(\log (u+1))$，插入均摊 $\mathcal{O}(\log (u+1))$；节点池空间 $\mathcal{O}(U)$。本版删除和摘取后继后会递归执行 `maintain`，删除耗时写为 $\mathcal{O}(\log (u+1)+R)$，R 是本次递归平衡维护的调用次数，不直接承诺删除的单次最坏 $\mathcal{O}(\log u)$。

```c++
// rangeCnt：统计闭区间内元素个数
struct SBT {
    struct Node {
        ll val = 0;
        int ch[2] = {}, cnt = 0, sz = 0, nodes = 0;
    };
    vector<Node> t = vector<Node>(1); // 下标 0 表示空节点
    vector<int> rec; // 已回收、可复用的节点编号
    int root = 0;

    int alloc() {
        if (rec.empty()) {
            t.emplace_back();
            return (int)(t.size()) - 1;
        }
        int p = rec.back();
        rec.pop_back();
        t[p] = Node{};
        return p;
    }
    void recycle(int p) {
        rec.push_back(p);
    }
    void clear() {
        root = 0;
        t.assign(1, Node{});
        rec.clear();
    }
    int size() const {
        return t[root].sz;
    }
    bool empty() const {
        return root == 0;
    }
    int make(ll x) {
        int p = alloc();
        t[p].val = x;
        t[p].cnt = t[p].sz = t[p].nodes = 1;
        return p;
    }
    void pull(int p) {
        int l = t[p].ch[0], r = t[p].ch[1];
        t[p].sz = t[l].sz + t[r].sz + t[p].cnt;
        t[p].nodes = t[l].nodes + t[r].nodes + 1;
    }
    int rotate(int p, int d) {
        int q = t[p].ch[d];
        t[p].ch[d] = t[q].ch[d ^ 1];
        t[q].ch[d ^ 1] = p;
        pull(p);
        pull(q);
        return q;
    }
    int maintain(int p, int d) {
        if (!p) return 0;
        int q = t[p].ch[d], other = t[p].ch[d ^ 1];
        // 每个孙子子树的节点数，不超过另一侧孩子子树的节点数。
        if (t[t[q].ch[d]].nodes > t[other].nodes) p = rotate(p, d);
        else if (t[t[q].ch[d ^ 1]].nodes > t[other].nodes) {
            t[p].ch[d] = rotate(q, d ^ 1);
            p = rotate(p, d);
        } else return p;
        t[p].ch[0] = maintain(t[p].ch[0], 0);
        t[p].ch[1] = maintain(t[p].ch[1], 1);
        p = maintain(p, 0);
        p = maintain(p, 1);
        return p;
    }
    int balance(int p) {
        if (!p) return 0;
        pull(p);
        p = maintain(p, 0);
        return maintain(p, 1);
    }
    int add(int p, ll x) {
        if (!p) return make(x);
        if (x == t[p].val) ++t[p].cnt;
        else {
            int d = x > t[p].val;
            int q = add(t[p].ch[d], x);
            t[p].ch[d] = q;
        }
        return balance(p);
    }
    int take_min(int p, int &q) {
        if (!t[p].ch[0]) {
            q = p;
            return t[p].ch[1];
        }
        t[p].ch[0] = take_min(t[p].ch[0], q);
        return balance(p);
    }
    int remove(int p, ll x, bool &found) {
        if (!p) return 0;
        if (x != t[p].val) {
            int d = x > t[p].val;
            t[p].ch[d] = remove(t[p].ch[d], x, found);
        } else {
            found = true;
            if (t[p].cnt > 1) --t[p].cnt;
            else if (!t[p].ch[0] || !t[p].ch[1]) {
                int q = t[p].ch[0] ? t[p].ch[0] : t[p].ch[1];
                recycle(p);
                return q;
            } else {
                int q = 0;
                t[p].ch[1] = take_min(t[p].ch[1], q);
                t[p].val = t[q].val;
                t[p].cnt = t[q].cnt;
                recycle(q);
            }
        }
        return balance(p); // 删除后也维护，避免删掉一侧后失衡
    }
    void insert(ll x) {
        root = add(root, x);
    }
    bool erase(ll x) {
        bool found = false;
        root = remove(root, x, found);
        return found;
    }
    int count(ll x) const {
        int p = root;
        while (p && t[p].val != x) p = t[p].ch[x > t[p].val];
        return t[p].cnt;
    }
    int less(ll x) const { // 严格小于 x 的元素个数
        int p = root, ans = 0;
        while (p)
            if (x <= t[p].val) p = t[p].ch[0];
            else {
                ans += t[t[p].ch[0]].sz + t[p].cnt;
                p = t[p].ch[1];
            }
        return ans;
    }
    ll kth(int k) const {
        /* assert(1 <= k && k <= size()) */;
        int p = root;
        while (p) {
            int l = t[t[p].ch[0]].sz;
            if (k <= l) p = t[p].ch[0];
            else if (k <= l + t[p].cnt) return t[p].val;
            else {
                k -= l + t[p].cnt;
                p = t[p].ch[1];
            }
        }
        /* assert(false) */;
        return 0; // 合法树和合法 k 不会走到这里
    }
    vector<ll> val() const { // 非递归中序遍历，包含重复值
        vector<ll> ans;
        vector<int> st;
        int p = root;
        while (p || !st.empty()) {
            while (p) {
                st.push_back(p);
                p = t[p].ch[0];
            }
            p = st.back();
            st.pop_back();
            ans.insert(ans.end(), t[p].cnt, t[p].val);
            p = t[p].ch[1];
        }
        return ans;
    }
    bool contains(ll x) const {
        return count(x) != 0;
    }
    int rank(ll x) const {
        return less(x) + 1;
    }
    ll prev(ll x) const {
        return kth(less(x));
    }
    ll next(ll x) const {
        return upper_bound(x);
    }
    ll lower_bound(ll x) const {
        return kth(less(x) + 1);
    }
    ll upper_bound(ll x) const {
        return kth(less(x) + count(x) + 1);
    }
    ll min() const {
        return kth(1);
    }
    ll max() const {
        return kth(size());
    }
    int rangeCnt(ll l, ll r) const {
        return l > r ? 0 : less(r) + count(r) - less(l);
    }
};
```

#### WBLT(Weight Balanced Leafy Tree)

定位：备选/学习实现；常规有序集合题不必与 FHQ、AVL 同时准备。

用途：用叶子保存有序集合中的键，内部节点只负责路由和统计，支持重复值、增删查、排名、第 k 小、前驱后继及区间计数。每个内部节点恰好有两个孩子，适合将实际数据与树的路由信息分开维护。

复杂度：单次增删和有序查询的树操作最坏 $\mathcal{O}(\log (u+1))$；u>0 时有效节点共 2u-1 个，节点池空间 $\mathcal{O}(U)$，递归栈 $\mathcal{O}(\log (u+1))$。以不同键的叶子数作权重，采用 delta=3、ratio=2 的旋转条件；`sz` 包含重复次数。删除最后一个副本时，同时回收叶子和多余的路由节点。

```c++
// rangeCnt：统计闭区间内元素个数
struct WBLT {
    struct Node {
        ll val = 0; // 叶子：键值；内部节点：子树最大键，用于路由
        int ch[2] = {}, cnt = 0, sz = 0, lf = 0; // lf：子树不同键的叶子数
    };
    vector<Node> t = vector<Node>(1); // 下标 0 表示空节点
    vector<int> rec; // 已回收、可复用的节点编号
    int root = 0;

    int alloc() {
        if (rec.empty()) {
            t.emplace_back();
            return (int)(t.size()) - 1;
        }
        int p = rec.back();
        rec.pop_back();
        t[p] = Node{};
        return p;
    }
    void recycle(int p) {
        rec.push_back(p);
    }
    void clear() {
        root = 0;
        t.assign(1, Node{});
        rec.clear();
    }
    int size() const {
        return t[root].sz;
    }
    bool empty() const {
        return root == 0;
    }
    bool leaf(int p) const {
        return t[p].ch[0] == 0;
    }
    int make(ll x) {
        int p = alloc();
        t[p].val = x;
        t[p].cnt = t[p].sz = t[p].lf = 1;
        return p;
    }
    void pull(int p) {
        if (leaf(p)) {
            t[p].sz = t[p].cnt;
            t[p].lf = 1;
            return;
        }
        int l = t[p].ch[0], r = t[p].ch[1];
        t[p].sz = t[l].sz + t[r].sz;
        t[p].lf = t[l].lf + t[r].lf;
        t[p].val = t[r].val;
    }
    int rotate(int p, int d) {
        int q = t[p].ch[d];
        t[p].ch[d] = t[q].ch[d ^ 1];
        t[q].ch[d ^ 1] = p;
        pull(p);
        pull(q);
        return q;
    }
    int balance(int p) { // delta=3, ratio=2；权重为不同键的叶子数
        pull(p);
        for (int d = 0; d < 2; ++d) {
            int q = t[p].ch[d], other = t[p].ch[d ^ 1];
            if (t[q].lf > 3LL * t[other].lf) {
                if (t[t[q].ch[d ^ 1]].lf >= 2LL * t[t[q].ch[d]].lf) t[p].ch[d] = rotate(q, d ^ 1);
                return rotate(p, d);
            }
        }
        return p;
    }
    int add(int p, ll x) {
        if (!p) return make(x);
        if (leaf(p)) {
            if (t[p].val == x) {
                ++t[p].cnt;
                pull(p);
                return p;
            }
            int q = make(x), fa = alloc();
            int d = x > t[p].val;
            t[fa].ch[d] = q;
            t[fa].ch[d ^ 1] = p;
            pull(fa);
            return fa;
        }
        int d = x > t[t[p].ch[0]].val;
        int q = add(t[p].ch[d], x);
        t[p].ch[d] = q;
        return balance(p);
    }
    int remove(int p, ll x, bool &found) {
        if (!p) return 0;
        if (leaf(p)) {
            if (t[p].val != x) return p;
            found = true;
            if (--t[p].cnt) {
                pull(p);
                return p;
            }
            recycle(p);
            return 0;
        }
        int d = x > t[t[p].ch[0]].val;
        t[p].ch[d] = remove(t[p].ch[d], x, found);
        if (!t[p].ch[d]) { // 叶子删空时，将其兄弟提升，并回收分支节点
            int q = t[p].ch[d ^ 1];
            recycle(p);
            return q;
        }
        return balance(p);
    }
    void insert(ll x) {
        root = add(root, x);
    }
    bool erase(ll x) {
        bool found = false;
        root = remove(root, x, found);
        return found;
    }
    int count(ll x) const {
        int p = root;
        if (!p) return 0;
        while (!leaf(p)) p = t[p].ch[x > t[t[p].ch[0]].val];
        return t[p].val == x ? t[p].cnt : 0;
    }
    int less(ll x) const {
        int p = root, ans = 0;
        if (!p) return 0;
        while (!leaf(p)) {
            int l = t[p].ch[0];
            if (x <= t[l].val) p = l;
            else {
                ans += t[l].sz;
                p = t[p].ch[1];
            }
        }
        return ans + (t[p].val < x ? t[p].cnt : 0);
    }
    ll kth(int k) const {
        /* assert(1 <= k && k <= size()) */;
        int p = root;
        while (!leaf(p)) {
            int l = t[p].ch[0];
            if (k <= t[l].sz) p = l;
            else {
                k -= t[l].sz;
                p = t[p].ch[1];
            }
        }
        return t[p].val;
    }
    vector<ll> val() const {
        vector<ll> ans;
        vector<int> st;
        if (root) st.push_back(root);
        while (!st.empty()) {
            int p = st.back();
            st.pop_back();
            if (leaf(p)) ans.insert(ans.end(), t[p].cnt, t[p].val);
            else {
                st.push_back(t[p].ch[1]);
                st.push_back(t[p].ch[0]);
            }
        }
        return ans;
    }
    bool contains(ll x) const {
        return count(x) != 0;
    }
    int rank(ll x) const {
        return less(x) + 1;
    }
    ll prev(ll x) const {
        return kth(less(x));
    }
    ll next(ll x) const {
        return upper_bound(x);
    }
    ll lower_bound(ll x) const {
        return kth(less(x) + 1);
    }
    ll upper_bound(ll x) const {
        return kth(less(x) + count(x) + 1);
    }
    ll min() const {
        return kth(1);
    }
    ll max() const {
        return kth(size());
    }
    int rangeCnt(ll l, ll r) const {
        return l > r ? 0 : less(r) + count(r) - less(l);
    }
};
```

### LCT：动态森林与路径操作

<a id="link-cut-tree"></a>

适用于拓扑会变化的**无向森林**：连边、断边、判断连通、单点赋值、路径加、路径和及路径第 k 个点。固定树通常优先用[树链剖分](<02 - 树上问题.md#hld>)。节点 1..n，0 是空节点；输入 `vector<ll> a(n+1)`，a[0] 忽略，初始化时所有点互不相连，`init(a)` 清空原边和标记。n≥1。

初始化 $\mathcal{O}(n)$，空间 $\mathcal{O}(n)$，各公开操作均摊 $\mathcal{O}(\log n)$，单次操作最坏 $\mathcal{O}(n)$。splay 前用显式栈从上到下下传标记，长链不会递归爆栈；不能把 splay 的均摊界写成每次最坏对数界。

`link(x,y)` 成功加边返回 true；已有连通关系（含自环）返回 false。`cut(x,y)` 仅删除直接相邻的边，不存在该边返回 false。`sum/add/kth` 的两端必须已连通；第 k 个点按 x→y 方向、从 1 开始计数，k 不超过路径点数。路径和、值、累计加法与乘积等中间量均须在 ll 范围内。

`fa` 同时表示辅助 splay 的父亲和虚边父亲，判断辅助树根必须看父节点的两个儿子是否含自己，不能只看 fa 是否为 0。`access(x)` 将当前根到 x 的路径变成实链并把 x 旋到辅助树根；原右儿子转为虚儿子时仍保留 fa。`makeroot(x)` 通过翻转改变原树的根；`split(x,y)` 后 y 的辅助树恰好表示 x→y 的路径。

```c++
struct LCT {
    struct Node {
        int ch[2] = {}, fa = 0, sz = 1;
        ll v = 0, sum = 0, tag = 0;
        bool rev = false;
    };
    vector<Node> t;
    vector<int> st; // splay 下传用栈，保存当前辅助树祖先
    LCT(const vector<ll> &a) {
        init(a);
    }
    void init(const vector<ll> &a) {
        /* assert(a.size() >= 2) */;
        t.assign(a.size(), Node{});
        t[0].sz = 0;
        st.clear();
        st.reserve(a.size());
        for (int i = 1; i < (int)a.size(); i++) t[i].v = t[i].sum = a[i];
    }
    bool isroot(int x) {
        int f = t[x].fa;
        return t[f].ch[0] != x && t[f].ch[1] != x;
    }
    void pull(int x) {
        int l = t[x].ch[0], r = t[x].ch[1];
        t[x].sz = t[l].sz + t[r].sz + 1;
        t[x].sum = t[l].sum + t[r].sum + t[x].v;
    }
    void putRev(int x) {
        if (!x) return;
        swap(t[x].ch[0], t[x].ch[1]);
        t[x].rev ^= 1;
    }
    void putAdd(int x, ll v) {
        if (!x) return;
        t[x].v += v;
        t[x].sum += v * t[x].sz;
        t[x].tag += v;
    }
    void push(int x) {
        if (t[x].rev) {
            putRev(t[x].ch[0]);
            putRev(t[x].ch[1]);
            t[x].rev = false;
        }
        if (t[x].tag) {
            putAdd(t[x].ch[0], t[x].tag);
            putAdd(t[x].ch[1], t[x].tag);
            t[x].tag = 0;
        }
    }
    void rotate(int x) {
        int y = t[x].fa, z = t[y].fa, d = (t[y].ch[1] == x), b = t[x].ch[d ^ 1];
        if (!isroot(y)) t[z].ch[t[z].ch[1] == y] = x;
        t[x].fa = z;
        t[x].ch[d ^ 1] = y;
        t[y].fa = x;
        t[y].ch[d] = b;
        if (b) t[b].fa = y;
        pull(y);
        pull(x);
    }
    void splay(int x) {
        st.clear();
        int y = x;
        st.push_back(y);
        while (!isroot(y)) {
            y = t[y].fa;
            st.push_back(y);
        }
        while (!st.empty()) {
            push(st.back());
            st.pop_back();
        }
        while (!isroot(x)) {
            int f = t[x].fa, g = t[f].fa;
            if (!isroot(f)) rotate((t[f].ch[1] == x) == (t[g].ch[1] == f) ? f : x);
            rotate(x);
        }
    }
    void access(int x) {
        for (int y = 0, u = x; u; y = u, u = t[u].fa) {
            splay(u);
            t[u].ch[1] = y;
            if (y) t[y].fa = u;
            pull(u); // 被换下的原右儿子保留 fa，表示虚边
        }
        splay(x);
    }
    void makeroot(int x) {
        access(x);
        putRev(x);
    }
    int findroot(int x) {
        access(x);
        push(x);
        while (t[x].ch[0]) {
            x = t[x].ch[0];
            push(x);
        }
        splay(x);
        return x;
    }
    bool same(int x, int y) {
        return findroot(x) == findroot(y);
    }
    void split(int x, int y) {
        makeroot(x);
        access(y);
    } // 连通时，暴露 x→y 路径
    bool link(int x, int y) {
        makeroot(x);
        if (findroot(y) == x) return false;
        t[x].fa = y;
        return true;
    }
    bool cut(int x, int y) {
        split(x, y); // 不连通也可安全执行这两步，但不能作路径查询
        if (t[y].ch[0] != x || t[x].ch[1]) return false;
        t[y].ch[0] = t[x].fa = 0;
        pull(y);
        return true;
    }
    void set(int x, ll v) {
        access(x);
        t[x].v = v;
        pull(x);
    }
    void add(int x, int y, ll v) {
        split(x, y);
        putAdd(y, v);
    }
    ll sum(int x, int y) {
        split(x, y);
        return t[y].sum;
    }
    int kth(int x, int y, int k) { // 返回原节点编号，沿 x→y 从 1 计数
        split(x, y); /* assert(1 <= k && k <= t[y].sz) */;
        int u = y;
        while (true) {
            push(u);
            int s = t[t[u].ch[0]].sz;
            if (k == s + 1) {
                splay(u);
                return u;
            }
            if (k <= s) u = t[u].ch[0];
            else {
                k -= s + 1;
                u = t[u].ch[1];
            }
        }
    }
};
```

```c++

tr.link(2, 3);
tr.add(1, 3, 2); // 点权变为 4,5,7,7
```

扩展提醒：本版只聚合辅助树中的**路径信息**，不能直接拿 `.sum` 查询原树子树和。维护子树需要额外记录虚儿子贡献并修改 access/link 等操作。边权题可给每条边建立独立节点 e，连接 u-e-v，原点权置 0；断边需 cut(u,e)、cut(e,v)。这时路径和与单条边赋值可直接使用，但本版路径加会同时修改原点；若只加边权，必须把 sz 改为路径内边节点数，并让原点不参与加法。异或版可替换聚合公式，但不能保留当前“加法”标记。

### 猫树（Disjoint Sparse Table）

<a id="disjoint-sparse-table"></a>

静态数组、0-based 闭区间。`query(l,r)` 计算 `a[l] op ... op a[r]`，运算 `op` 只要求满足结合律，不要求幂等或交换，因此可处理区间和、区间乘积、矩阵乘法、字符串拼接等；数组修改后必须重新 `init`。预处理 $\mathcal{O}(n \log n)$，单次查询 $\mathcal{O}(1)$，空间 $\mathcal{O}(n \log n)$，n>0，结果及中间合并值须合法。

每层把数组分成左右两半，左半保存到分界点的后缀，右半保存从分界点开始的前缀。`l<r` 时取 `l^r` 的最高位即可定位第一次分开的层，并按“左后缀、右前缀”的顺序合并；这也是非交换运算不能调换两项的原因。

```c++
template <class T, class Op> struct CatTree {
    int n = 0;
    vector<T> a;
    vector<vector<T>> st;
    Op op;
    CatTree(const vector<T> &v, Op f = Op{}) : op(f) {
        init(v);
    }
    void init(const vector<T> &v) {
        a = v;
        n = (int)a.size(); /* assert(n > 0) */;
        int lg = 0;
        while ((1LL << lg) < n) ++lg;
        st.assign(lg, a);
        for (int k = 0; k < lg; ++k) {
            int h = 1LL << k, b = h * 2;
            for (int l = 0; l < n; l += b) {
                int m = min(l + h, n), r = min(l + b, n);
                if (l < m) {
                    st[k][m - 1] = a[m - 1];
                    for (int i = m - 2; i >= l; --i) st[k][i] = op(a[i], st[k][i + 1]);
                }
                if (m < r) {
                    st[k][m] = a[m];
                    for (int i = m + 1; i < r; ++i) st[k][i] = op(st[k][i - 1], a[i]);
                }
            }
        }
    }
    T query(int l, int r) const {
        /* assert(0 <= l && l <= r && r < n) */;
        if (l == r) return a[l];
        int k = 63 - __builtin_clzll((ull)(l ^ r));
        return op(st[k][l], st[k][r]);
    }
};
```

调用：`struct Add { ll operator()(ll x,ll y) const { return x+y; } }; CatTree<ll,Add> t({1,2,3,4});`，则 `t.query(1,3)==9`。若 `op(x,y)=x+y` 用于字符串，查询会保持原数组次序。

### 析合树：排列连续段

<a id="divide-combine-tree"></a>

输入带占位的 1..n 排列 `p`。若 `p[l..r]` 的值恰好组成一段连续整数，则 [l,r] 称为连续段。析合树由所有本原（强）连续段构成：节点 `1..cnt`，`root` 为根，`L/R` 是位置区间，`fa/son` 给出树形关系，`leaf[i]` 是位置 i 对应的叶子；`son[u]` 按位置从左到右排列。`type[u]` 为 0/1/2，分别表示叶子、合点、析点。

合点的任意连续若干个孩子都能组成连续段；析点除单个孩子和全部孩子外，没有连续若干个孩子组成连续段。因此这些节点既可用于连续段计数，也可承载按连续段分治的树上 DP。`segment(l,r)` 返回包含位置区间 [l,r] 的最小连续段；`isSegment(l,r)` 判断 [l,r] 本身是否连续。

构造使用单调栈、区间加/最小值线段树与静态区间最值，时间 $\mathcal{O}(n \log n)$，节点数不超过 2n-1；LCA 预处理使 `segment` 为 $\mathcal{O}(\log n)$，`isSegment` 为 $\mathcal{O}(\log n)$。总空间 $\mathcal{O}(n \log n)$，主要来自 LCA 倍增表，其余为 $\mathcal{O}(n)$。要求 n>0，元素不重不漏且均在 1..n；下标和节点显式存为 `int32_t`，故 n 与最多 2n-1 个节点都必须能用 `int32_t` 表示。

```c++
struct DivideCombineTree {
    using I = int32_t;
    struct StaticRange {
        int n = 1;
        vector<I> mn, mx;
        void init(const vector<I> &a) {
            while (n < (int)a.size() - 1) n *= 2;
            mn.assign(2 * n, INT32_MAX);
            mx.assign(2 * n, INT32_MIN);
            for (int i = 1; i < (int)a.size(); ++i) mn[n + i - 1] = mx[n + i - 1] = a[i];
            for (int i = n - 1; i; --i) {
                mn[i] = min(mn[i * 2], mn[i * 2 + 1]);
                mx[i] = max(mx[i * 2], mx[i * 2 + 1]);
            }
        }
        pair<I, I> query(int l, int r) const {
            I lo = INT32_MAX, hi = INT32_MIN;
            for (l += n - 1, r += n; l < r; l /= 2, r /= 2) {
                if (l & 1) lo = min(lo, mn[l]), hi = max(hi, mx[l++]);
                if (r & 1) --r, lo = min(lo, mn[r]), hi = max(hi, mx[r]);
            }
            return {lo, hi};
        }
    } rmq;
    struct AddMin {
        int n;
        vector<I> mn, lz;
        AddMin(int n = 0) : n(n), mn(4 * n + 4), lz(4 * n + 4) {}
        void put(int p, I x) {
            mn[p] += x;
            lz[p] += x;
        }
        void push(int p) {
            if (!lz[p]) return;
            put(p * 2, lz[p]);
            put(p * 2 + 1, lz[p]);
            lz[p] = 0;
        }
        void add(int l, int r, I x) {
            if (l <= r) add(1, 1, n, l, r, x);
        }
        void add(int p, int l, int r, int ql, int qr, I x) {
            if (ql <= l && r <= qr) {
                put(p, x);
                return;
            }
            push(p);
            int m = (l + r) / 2;
            if (ql <= m) add(p * 2, l, m, ql, qr, x);
            if (qr > m) add(p * 2 + 1, m + 1, r, ql, qr, x);
            mn[p] = min(mn[p * 2], mn[p * 2 + 1]);
        }
        int firstZero(int p, int l, int r) {
            if (l == r) return l;
            push(p);
            int m = (l + r) / 2;
            return mn[p * 2] == 0 ? firstZero(p * 2, l, m) : firstZero(p * 2 + 1, m + 1, r);
        }
        int firstZero() {
            /* assert(mn[1] == 0) */;
            return firstZero(1, 1, n);
        }
    };

    int n = 0, cnt = 0;
    I root = 0;
    vector<I> L, R, fa, last, leaf, dep;
    vector<unsigned char> type;
    vector<vector<I>> son, up;
    DivideCombineTree() = default;
    DivideCombineTree(const vector<int> &p) {
        init(p);
    }
    int makeNode(I l, I r, unsigned char tp) {
        ++cnt;
        L[cnt] = l;
        R[cnt] = r;
        type[cnt] = tp;
        return cnt;
    }
    bool isSegment(int l, int r) const {
        /* assert(1 <= l && l <= r && r <= n) */;
        auto [lo, hi] = rmq.query(l, r);
        return hi - lo == r - l;
    }
    void init(const vector<int> &p) {
        n = (int)p.size() - 1; /* assert(n > 0) */;
        vector<I> a(n + 1);
        for (int i = 1; i <= n; ++i) a[i] = (I)p[i];
        rmq = StaticRange{};
        rmq.init(a);
        cnt = 0;
        int cap = 2 * n + 3;
        L.assign(cap, 0);
        R.assign(cap, 0);
        fa.assign(cap, 0);
        last.assign(cap, 0);
        type.assign(cap, 0);
        son.assign(cap, {});
        leaf.assign(n + 1, 0);
        AddMin q(n);
        vector<I> smin(n + 1), smax(n + 1), stk; // 尚未归并的树根栈
        int tmin = 0, tmax = 0;
        for (int i = 1; i <= n; ++i) {
            while (tmin && a[i] <= a[smin[tmin]]) {
                q.add(smin[tmin - 1] + 1, smin[tmin], a[smin[tmin]]);
                --tmin;
            }
            while (tmax && a[i] >= a[smax[tmax]]) {
                q.add(smax[tmax - 1] + 1, smax[tmax], -a[smax[tmax]]);
                --tmax;
            }
            q.add(smin[tmin] + 1, i, -a[i]);
            smin[++tmin] = i;
            q.add(smax[tmax] + 1, i, a[i]);
            smax[++tmax] = i;

            I now = makeNode(i, i, 0);
            leaf[i] = now;
            int le = q.firstZero();
            while (!stk.empty() && L[stk.back()] >= le) {
                I top = stk.back();
                if (type[top] == 1 && isSegment(last[top], i)) {
                    stk.pop_back();
                    R[top] = i;
                    last[top] = L[now];
                    son[top].push_back(now);
                    fa[now] = top;
                    now = top;
                } else if (isSegment(L[top], i)) {
                    stk.pop_back();
                    I u = makeNode(L[top], i, 1);
                    last[u] = L[now];
                    son[u] = {top, now};
                    fa[top] = fa[now] = u;
                    now = u;
                } else {
                    vector<I> sons{now};
                    I l = 0;
                    do {
                        /* assert(!stk.empty()) */;
                        I v = stk.back();
                        stk.pop_back();
                        sons.push_back(v);
                        if (isSegment(L[v], i)) {
                            l = L[v];
                            break;
                        }
                    } while (true);
                    reverse(sons.begin(), sons.end());
                    I u = makeNode(l, i, 2);
                    son[u] = move(sons);
                    for (I v : son[u]) fa[v] = u;
                    now = u;
                }
            }
            stk.push_back(now);
            q.add(1, i, -1);
        }
        /* assert(stk.size() == 1) */;
        root = stk[0];
        L.resize(cnt + 1);
        R.resize(cnt + 1);
        fa.resize(cnt + 1);
        last.resize(cnt + 1);
        type.resize(cnt + 1);
        son.resize(cnt + 1);
        initLca();
    }
    void initLca() {
        fa[root] = root;
        dep.assign(cnt + 1, 0);
        int lg = 1;
        while ((1LL << lg) <= cnt) ++lg;
        up.assign(lg, vector<I>(cnt + 1, root));
        vector<I> ord{root}; // 父节点先于子节点的遍历序
        for (int i = 0; i < (int)ord.size(); ++i) {
            I u = ord[i];
            up[0][u] = fa[u];
            for (I v : son[u]) dep[v] = dep[u] + 1, ord.push_back(v);
        }
        for (int k = 1; k < lg; ++k)
            for (int u = 1; u <= cnt; ++u) up[k][u] = up[k - 1][up[k - 1][u]];
    }
    I climb(I u, int d) const {
        for (int k = 0; d; ++k, d >>= 1)
            if (d & 1) u = up[k][u];
        return u;
    }
    I lca(I u, I v) const {
        if (dep[u] < dep[v]) swap(u, v);
        u = climb(u, dep[u] - dep[v]);
        if (u == v) return u;
        for (int k = (int)up.size() - 1; k >= 0; --k)
            if (up[k][u] != up[k][v]) u = up[k][u], v = up[k][v];
        return fa[u];
    }
    pair<int, int> segment(int l, int r) const {
        /* assert(1 <= l && l <= r && r <= n) */;
        if (l == r) return {l, r};
        I x = leaf[l], y = leaf[r], z = lca(x, y);
        if (type[z] == 1) {
            x = climb(x, dep[x] - dep[z] - 1);
            y = climb(y, dep[y] - dep[z] - 1);
            return {L[x], R[y]};
        }
        return {L[z], R[z]};
    }
};
```

调用：`DivideCombineTree t(vector<int>{0,5,3,4,1,2});`。`t.isSegment(2,3)` 为 true，`t.segment(1,2)==pair<int,int>{1,3}`。遍历节点时，合点可枚举连续孩子区间；析点通常只处理整个节点，具体 DP 含义由题目定义。

### 笛卡尔树

<a id="cartesian-tree"></a>

用途：把静态数组转成满足“下标中序有序、值满足堆序”的树，用于区间最值查询和按最值划分的树上 DP。默认 `less<T>` 建最小堆，改用 `greater<T>` 建最大堆；值相同时优先选择更靠左的位置。节点编号和查询下标均从 0 开始，空节点为 -1。

复杂度：n 为数组长度，单调栈建树 `build(v)` 的时间、空间均为 $\mathcal{O}(n)$；LCA 预处理的时间、空间为 $\mathcal{O}(n \log (n+1))$，随后单次 LCA 或区间最值查询为 $\mathcal{O}(\log (n+1))$。若未调用 `initLca()`，第一次合法查询会自动预处理，需计入这部分开销。

`l/r/fa/root` 可直接用于树上 DP。`arg_query(l,r)` 返回闭区间最值的下标，`query(l,r)` 返回最值，分别直接返回 int 和 T，查询前要求 0<=l<=r<n；lca 的两个下标也必须在 [0,n) 内。这里只建树时不分配 LCA 表。

```c++
// initLca：预处理 LCA
template <class T = ll, class Compare = less<T>> struct CartesianTree {
    int n = 0, root = -1;
    vector<T> a;
    vector<int> l, r, fa, dep;
    vector<vector<int>> up;
    Compare cmp;

    CartesianTree() = default;
    CartesianTree(const vector<T> &v, Compare comp = Compare{}) : cmp(comp) {
        build(v);
    }
    // 输入和节点编号均为 0-based；相等时保留靠左的位置为祖先。
    void build(const vector<T> &v) {
        a = v;
        n = (int)(a.size());
        root = -1;
        l.assign(n, -1);
        r.assign(n, -1);
        fa.assign(n, -1);
        dep.clear();
        up.clear();
        vector<int> st;
        for (int i = 0; i < n; ++i) {
            int last = -1;
            while (!st.empty() && cmp(a[i], a[st.back()])) {
                last = st.back();
                st.pop_back();
            }
            if (!st.empty()) {
                r[st.back()] = i;
                fa[i] = st.back();
            }
            if (last != -1) {
                l[i] = last;
                fa[last] = i;
            }
            st.push_back(i);
        }
        if (!st.empty()) root = st.front();
    }
    // 区间最值可转成两端点的 LCA；需要查询时才做 O(n log n) 预处理。
    void initLca() {
        if (!n) return;
        int lg = 1;
        while ((1LL << lg) <= n) ++lg;
        up.assign(lg, vector<int>(n));
        dep.assign(n, 0);
        vector<int> ord{root}; // 父节点先于子节点的遍历序
        up[0][root] = root;
        for (int i = 0; i < (int)(ord.size()); ++i) {
            int p = ord[i];
            for (int q : {l[p], r[p]})
                if (q != -1) {
                    dep[q] = dep[p] + 1;
                    up[0][q] = p;
                    ord.push_back(q);
                }
        }
        for (int k = 1; k < lg; ++k)
            for (int i = 0; i < n; ++i) up[k][i] = up[k - 1][up[k - 1][i]];
    }
    int lca(int u, int v) {
        /* assert(0 <= u && u < n && 0 <= v && v < n) */;
        if (up.empty()) initLca();
        if (dep[u] < dep[v]) swap(u, v);
        int diff = dep[u] - dep[v];
        for (int k = 0; k < (int)(up.size()); ++k)
            if ((diff >> k) & 1) u = up[k][u];
        if (u == v) return u;
        for (int k = (int)(up.size()) - 1; k >= 0; --k)
            if (up[k][u] != up[k][v]) {
                u = up[k][u];
                v = up[k][v];
            }
        return up[0][u];
    }
    int arg_query(int l, int r) { // 闭区间 [l,r] 最值的下标
        /* assert(l <= r) */;
        return lca(l, r);
    }
    T query(int l, int r) {
        return a[arg_query(l, r)];
    }
};
```

用法：`CartesianTree<long long> tr(a);` 查询区间最小值；`CartesianTree<long long, greater<long long>> tr(a);` 查询区间最大值。数组修改后需重新 `build`，旧的 LCA 预处理会自动清除。

### 左偏树：可并最小堆

<a id="leftist-heap"></a>

同一对象管理多棵堆，外部 int 根变量初始为 0。`insert(root,x)` 插入，`top(root)` 查看最小值，`pop(root)` 删除并返回最小值；top/pop 直接返回 ll，调用前检查 root!=0。`merge(a,b)` 合并到 a 并清空 b。各根不能共享节点或是同一个变量，复制根变量不是复制堆；不支持直接改键或按任意节点编号删除。

小于等于的相同键可重复，键覆盖全部 ll。空节点 dis=0，节点 dis=右孩子 dis+1，并保证左 dis≥右 dis。合并/删除堆顶 $\mathcal{O}(\log (n+1))$，插入 $\mathcal{O}(\log (n+1))$，top/size $\mathcal{O}(1)$，节点池空间为历史最大同时存活节点数 $\mathcal{O}(N)$，删除节点回收复用。clear 清空所有堆并使旧根失效。若题目通过“某元素编号”定位所在堆，需要额外维护归属信息，不能直接把元素编号当根。

```c++
struct LeftistHeap {
    struct Node {
        ll key = 0;
        int ch[2] = {}, dis = 0, sz = 0;
    };
    vector<Node> t = vector<Node>(1);
    vector<int> free;
    void clear() {
        t.assign(1, Node{});
        free.clear();
    }
    int make(ll x) {
        int p;
        if (free.empty()) {
            p = (int)(t.size());
            t.emplace_back();
        } else {
            p = free.back();
            free.pop_back();
            t[p] = Node{};
        }
        t[p].key = x;
        t[p].dis = t[p].sz = 1;
        return p;
    }
    int size(int p) const {
        return t[p].sz;
    }
    int meld(int a, int b) {
        if (!a || !b) return a ? a : b;
        /* assert(a != b) */;
        if (t[a].key > t[b].key) swap(a, b);
        t[a].ch[1] = meld(t[a].ch[1], b);
        if (t[t[a].ch[0]].dis < t[t[a].ch[1]].dis) swap(t[a].ch[0], t[a].ch[1]);
        t[a].dis = t[t[a].ch[1]].dis + 1;
        t[a].sz = 1 + t[t[a].ch[0]].sz + t[t[a].ch[1]].sz;
        return a;
    }
    void merge(int &a, int &b) {
        /* assert(&a != &b) */;
        a = meld(a, b);
        b = 0;
    }
    void insert(int &root, ll x) {
        int p = make(x);
        root = meld(root, p);
    }
    ll top(int root) const {
        /* assert(root) */;
        return t[root].key;
    }
    ll pop(int &root) {
        /* assert(root) */;
        int p = root;
        ll x = t[p].key;
        root = meld(t[p].ch[0], t[p].ch[1]);
        free.push_back(p);
        return x;
    }
};
```

调用：`LeftistHeap h; int a=0,b=0; h.insert(a,5); h.insert(b,-2); h.merge(a,b);` 后 b=0、`h.pop(a)==-2`、`h.top(a)==5`。多测时重新构造对象，或 clear 后将所有根变量置 0。

### 坐标压缩与离散化

#### 简单版本

把原值数组 alls 排序去重，get(x) 返回 0-based 下界位置；仅当 x 已登记时才是其编号。排序 $\mathcal{O}(n \log n)$、查询 $\mathcal{O}(\log n)$、原地去重额外空间 $\mathcal{O}(1)$（排序递归栈 $\mathcal{O}(\log n)$）。放在 solve 内。

```c++
sort(alls.begin(), alls.end());
alls.erase(unique(alls.begin(), alls.end()), alls.end());
auto get = [&](ll x) {
    return lower_bound(alls.begin(), alls.end(), x) - alls.begin();
};
```

#### 封装

默认 0-based，可设置 `shift=1` 切换；`Get(c[x])==x`，极值也正常参与离散化，不额外插入哨兵。收集阶段使用 add，查询前必须 init；再次 add 后须重新 init，旧编号可能变化。预处理 $\mathcal{O}(n \log n)$，查编号/存在性 $\mathcal{O}(\log n)$，反查 $\mathcal{O}(1)$，空间 $\mathcal{O}(n)$。

```c++
template <class T> struct Compress_ {
    int n = 0, shift = 0; // 默认 0-based；需要 1-based 时设置 shift=1
    vector<T> alls;

    Compress_() = default;
    Compress_(const vector<T> &in) : alls(in) {
        init();
    }
    void add(T x) {
        alls.push_back(x);
    }
    template <class... Args> void add(T x, Args... args) {
        add(x);
        int unused[] = {0, (add(args), 0)...};
        (void)unused;
    }
    void init() {
        sort(alls.begin(), alls.end());
        alls.erase(unique(alls.begin(), alls.end()), alls.end());
        n = (int)(alls.size());
    }
    int size() const {
        return n;
    }
    int operator[](T x) const {
        auto it = lower_bound(alls.begin(), alls.end(), x);
        /* assert(it != alls.end() && *it == x) */; // 只映射已加入的元素
        return (int)(it - alls.begin()) + shift;
    }
    T Get(int index) const {
        int i = index - shift;
        /* assert(0 <= i && i < n) */;
        return alls[i];
    }
    bool count(T x) const {
        return binary_search(alls.begin(), alls.end(), x);
    }
    friend ostream &operator<<(ostream &out, const Compress_ &c) {
        out << "{";
        for (const T &x : c.alls) out << x << " ";
        return out << "}";
    }
};
using Compress = Compress_<int>;
```

### 小波矩阵树：高效静态区间第 K 大查询

静态区间第 k 小/大：传入普通 0-based 数组，查询闭区间 [l,r]，k 从 1 开始且不能超过区间长度。支持负数、重复值，空数组只能构造不能查询。预处理 $\mathcal{O}(n \log n)$，压位存储空间 $\mathcal{O}(n+n \log n/64)$，单次查询 $\mathcal{O}(\log n)$；不支持修改。内部位偏移采用 int，要求 n*ceil(log₂(不同值数)) 不超过 INT_MAX。位向量统计前缀 1 的个数，通过当前位为 0 的数量决定第 k 小走哪侧；重新构造重建。

调用：`Wavelet w(vector<int>{5,-2,5});` 后 `w.small(0,2,2)==5`、`w.large(0,2,3)==-2`。

```c++
struct Wavelet {
    vector<int> val, sum;
    vector<ull> bit;
    int t, n;

    int getSum(int i) {
        return sum[i >> 6] + pcntll(bit[i >> 6] & ((1ULL << (i & 63)) - 1));
    }

    Wavelet(vector<int> v) : val(v), n(v.size()) {
        if (n == 0) {
            t = 0;
            return;
        }
        sort(val.begin(), val.end());
        val.erase(unique(val.begin(), val.end()), val.end());

        int n_ = val.size();
        t = __lg(2LL * n_ - 1);
        bit.resize((1ULL * t * n + 64) >> 6);
        sum.resize(bit.size());
        vector<int> cnt(n_ + 1);

        for (int &x : v) {
            x = lower_bound(val.begin(), val.end(), x) - val.begin();
            cnt[x + 1]++;
        }
        for (int i = 1; i < n_; ++i) cnt[i] += cnt[i - 1];
        for (int j = 0; j < t; ++j) {
            for (int i : v) {
                int tmp = i >> (t - 1 - j);
                int pos = (tmp >> 1) << (t - j);
                auto setBit = [&](int i, ull v) {
                    bit[i >> 6] |= (v << (i & 63));
                };
                setBit(j * n + cnt[pos], tmp & 1);
                cnt[pos]++;
            }
            for (int i : v) cnt[(i >> (t - j)) << (t - j)]--;
        }
        for (int i = 1; i < sum.size(); ++i) sum[i] = sum[i - 1] + pcntll(bit[i - 1]);
    }

    int small(int l, int r, int k) {
        /* assert(0 <= l && l <= r && r < n && 1 <= k && k <= r - l + 1) */;
        r++, k--;
        for (int j = 0, x = 0, y = n, res = 0;; ++j) {
            if (j == t) return val[res];
            int A = getSum(n * j + x), B = getSum(n * j + l);
            int C = getSum(n * j + r), D = getSum(n * j + y);
            int ab_zeros = r - l - C + B;
            if (ab_zeros > k) {
                res = res << 1;
                y -= D - A;
                l -= B - A;
                r -= C - A;
            } else {
                res = (res << 1) | 1;
                k -= ab_zeros;
                x += y - x - D + A;
                l += y - l - D + B;
                r += y - r - D + C;
            }
        }
    }
    int large(int l, int r, int k) {
        return small(l, r, r - l + 2 - k);
    }
};
```

### 普通莫队

<a id="mo"></a>

完整示例：离线查询区间不同值数量，初值先离散化，支持负数及重复值。输入 n、n 个初值、q，再输入 q 个闭区间；按原查询顺序输出答案。数组和查询均为 1-based，n>0、q≥0。改做其他统计时替换 add/del，并让 val 始终等于当前窗口答案；若统计依赖顺序，左右端点须分别维护。

一次增删 $\mathcal{O}(1)$，块长 B 时总时间 $\mathcal{O}(n \log n+q \log (q+1)+n^{2}/B+qB)$，空间 $\mathcal{O}(n+q)$。q≈n、$B\approx\sqrt n$ 时为 $\mathcal{O}(n\sqrt{n})$；q=0 直接退出。

离线处理查询，块长可取 `max<int>(1, (int)(n / sqrt(max<int>(1,q))))`，也可在 q 与 n 同阶时取 sqrt(n)。

```c++
signed main() {
    int n;
    cin >> n;
    vector<int> w(n + 1);
    for (int i = 1; i <= n; i++) cin >> w[i];

    vector<int> xs(w.begin() + 1, w.end());
    sort(xs.begin(), xs.end());
    xs.erase(unique(xs.begin(), xs.end()), xs.end());
    for (int i = 1; i <= n; i++) w[i] = lower_bound(xs.begin(), xs.end(), w[i]) - xs.begin();
    vector<int> cnt(xs.size());
    int q;
    cin >> q;
    if (q == 0) return 0;
    vector<array<int, 3>> query(q + 1);
    for (int i = 1; i <= q; i++) {
        int l, r;
        cin >> l >> r;
        query[i] = {l, r, i};
    }

    int Knum = n / min<int>(n, sqrt(q)); // 计算块长
    vector<int> K(n + 1);
    for (int i = 1; i <= n; i++) { // 固定块长
        K[i] = (i - 1) / Knum + 1;
    }
    sort(query.begin() + 1, query.end(), [&](auto x, auto y) {
        if (K[x[0]] != K[y[0]]) return x[0] < y[0];
        if (K[x[0]] & 1) return x[1] < y[1];
        return x[1] > y[1];
    });

    int l = 1, r = 0, val = 0;
    vector<int> ans(q + 1);
    auto add = [&](int x) {
        if (cnt[x]++ == 0) ++val;
    };
    auto del = [&](int x) {
        if (--cnt[x] == 0) --val;
    };
    for (int i = 1; i <= q; i++) {
        auto [ql, qr, id] = query[i];
        while (l > ql) add(w[--l]);
        while (r < qr) add(w[++r]);
        while (l < ql) del(w[l++]);
        while (r > qr) del(w[r--]);
        ans[id] = val;
    }
    for (int i = 1; i <= q; i++) cout << ans[i] << endl;
}
```

需要注意的是，在普通莫队中，`K` 数组的作用是根据左边界的值进行排序，当询问次数很少时（$q \ll n$），可以直接合并到 `query` 数组中。

```c++
vector<array<int, 4>> query(q + 1);
for (int i = 1; i <= q; i++) {
    int l, r;
    cin >> l >> r;
    query[i] = {l, r, i, (l - 1) / Knum + 1}; // 合并
}
sort(query.begin() + 1, query.end(), [&](auto x, auto y) {
    if (x[3] != y[3]) return x[3] < y[3];
    if (x[3] & 1) return x[1] < y[1];
    return x[1] > y[1];
});
```

### 带修改的莫队（带时间维度的莫队）

算法骨架：需实现 add/del/time，time 必须支持修改和撤销。1-based 闭区间；以下 $\mathcal{O}(n^{5/3})$ 估计假设查询数、修改数与 n 同阶且单次维护 $\mathcal{O}(1)$。完整“颜色种数”例子见文末。

在上述规模假设下，块长通常取 $\lfloor n^{2/3}\rfloor$，且至少为 1；代码使用 `max<int>(1, (int)(pow(n, 2.0 / 3)))`，按实际 n 计算。查询数和修改数相差很大时，需重新估计移动次数并调整块长。

```c++
signed main() {
    int n, q;
    cin >> n >> q;
    vector<int> w(n + 1);
    for (int i = 1; i <= n; i++) cin >> w[i];

    vector<array<int, 4>> query = {{}}; // {左区间, 右区间, 累计修改次数, 下标}
    vector<array<int, 2>> modify = {{}}; // {修改的值, 修改的元素下标}
    for (int i = 1; i <= q; i++) {
        char op;
        cin >> op;
        if (op == 'Q') {
            int l, r;
            cin >> l >> r;
            query.push_back({l, r, (int)modify.size() - 1, (int)query.size()});
        } else {
            int idx, w;
            cin >> idx >> w;
            modify.push_back({w, idx});
        }
    }

    int Knum = max<int>(1, (int)(pow(n, 2.0 / 3))); // 按规模选择块长
    vector<int> K(n + 1);
    for (int i = 1; i <= n; i++) { // 固定块长
        K[i] = (i - 1) / Knum + 1;
    }
    sort(query.begin() + 1, query.end(), [&](auto x, auto y) {
        if (K[x[0]] != K[y[0]]) return x[0] < y[0];
        if (K[x[1]] != K[y[1]]) return (K[x[0]] & 1) ? x[1] < y[1] : x[1] > y[1];
        return (K[x[1]] & 1) ? x[2] < y[2] : x[2] > y[2]; // 时间维蛇形排序
    });

    int l = 1, r = 0, val = 0;
    int t = 0; // 累计修改次数
    vector<int> ans(query.size());
    for (int i = 1; i < query.size(); i++) {
        auto [ql, qr, qt, id] = query[i];
        auto add = [&](int x) -> void {
        };
        auto del = [&](int x) -> void {
        };
        auto time = [&](int x, int l, int r) -> void {
        };
        while (l > ql) add(w[--l]);
        while (r < qr) add(w[++r]);
        while (l < ql) del(w[l++]);
        while (r > qr) del(w[r--]);
        while (t < qt) time(++t, ql, qr);
        while (t > qt) time(t--, ql, qr);
        ans[id] = val;
    }
    for (int i = 1; i < ans.size(); i++) cout << ans[i] << endl;
}
```

### 主席树（可持久化线段树）

<a id="persistent-order-statistics"></a>

前缀频次版本：先将值离散到 1..V，再创建 `PresidentTree pst(n,keys)`，keys 是升序去重后的原值数组，0-based。依次调用 `modify(root[i],root[i-1],1,V,a[i])` 插入离散编号；`kth(root[l-1],root[r],1,V,k)` 返回区间 [l,r] 的第 k 小值的离散编号。`count_le` 查询不大于给定编号的元素数，`sumK` 查询最小 k 个元素的原值之和，后者允许 k=0。

保留旧构造 `PresidentTree(n)`：省略 keys 时把离散编号本身当作值，kth/count_le 不受影响，但 sumK 求出的是编号之和。需要原值和时必须传 keys。值域的 V 必须与 keys.size() 一致；每个版本的总和及查询中间和须在 ll 范围内。重复值分别计数，允许原值为负。

每次插入/查询 $\mathcal{O}(\log (V+1))$，n 次插入总时间、空间 $\mathcal{O}(n \log (V+1))$，外加 root/keys 的存储。k 必须合法，查询的两个版本之差必须表示非负频次；不是任意两版本都可相减。节点池放在堆上，init 可重置多测，清空后旧节点编号失效。keys 在一次初始化后不可改动。

新增 sum 会增加节点内存：本机普通 64 位 GNU 环境下节点为 24 字节，原纯频次节点为 12 字节。大数据先估算节点数 `1+n*(ceil(log2(V))+1)`，必要时 `pst.tr.reserve(上界)` 避免扩容峰值；仅需频次且内存紧张时可移除 sum 字段、add 中的 sum 更新及 sumK 接口。数值检查与主机内存测试均不能代替题目实际内存限制。

```c++
// sumK：计算最小 k 个元素之和；tot：已分配的最大节点编号
struct PresidentTree {
    struct node {
        int l = 0, r = 0, cnt = 0;
        ll sum = 0;
    };
    vector<node> tr;
    vector<int> root;
    vector<ll> keys;
    int tot = 0;

    PresidentTree(int ver = 0, vector<ll> val = {}) {
        init(ver, move(val));
    }
    void init(int ver, vector<ll> val = {}) {
        /* assert(ver >= 0) */;
        /* assert(is_sorted(val.begin(), val.end())) */;
        /* assert(adjacent_find(val.begin(), val.end()) == val.end()) */;
        keys = move(val);
        tr.assign(1, node{}); // 0 为全零空节点
        root.assign(ver + 1, 0);
        tot = 0;
    }
    int add(int v, int l, int r, int x) {
        /* assert(l <= x && x <= r) */;
        node old = tr[v];
        int u = ++tot;
        tr.push_back(old);
        ++tr[u].cnt;
        tr[u].sum += value(x);
        if (l == r) return u;
        int mid = l + (r - l) / 2;
        // 不让节点成员引用跨越 vector 扩容。
        if (x <= mid) {
            int son = add(old.l, l, mid, x);
            tr[u].l = son;
        } else {
            int son = add(old.r, mid + 1, r, x);
            tr[u].r = son;
        }
        return u;
    }
    void modify(int &u, int v, int l, int r, int x) {
        u = add(v, l, r, x);
    }
    ll value(int x) const {
        if (keys.empty()) return x;
        /* assert(1 <= x && x <= (int)(keys.size())) */;
        return keys[x - 1];
    }
    int kth(int u, int v, int l, int r, int k) const {
        /* assert(1 <= k && k <= tr[v].cnt - tr[u].cnt) */;
        if (l == r) return l;
        int count = tr[tr[v].l].cnt - tr[tr[u].l].cnt;
        int mid = l + (r - l) / 2;
        if (k <= count) return kth(tr[u].l, tr[v].l, l, mid, k);
        return kth(tr[u].r, tr[v].r, mid + 1, r, k - count);
    }
    int count_le(int u, int v, int l, int r, int x) const {
        if (x < l) return 0;
        if (r <= x) return tr[v].cnt - tr[u].cnt;
        int mid = l + (r - l) / 2;
        if (x <= mid) return count_le(tr[u].l, tr[v].l, l, mid, x);
        return tr[tr[v].l].cnt - tr[tr[u].l].cnt + count_le(tr[u].r, tr[v].r, mid + 1, r, x);
    }
    ll sumK(int u, int v, int l, int r, int k) const {
        /* assert(0 <= k && k <= tr[v].cnt - tr[u].cnt) */;
        if (!k) return 0;
        if (l == r) return value(l) * k;
        int count = tr[tr[v].l].cnt - tr[tr[u].l].cnt;
        int mid = l + (r - l) / 2;
        if (k <= count) return sumK(tr[u].l, tr[v].l, l, mid, k);
        return tr[tr[v].l].sum - tr[tr[u].l].sum + sumK(tr[u].r, tr[v].r, mid + 1, r, k - count);
    }
};
```

### 树套树：树状数组套权值线段树

<a id="bit-segtree"></a>

用于**单点赋值＋区间顺序统计**：排名、第 k 小、严格前驱/后继、小于或不大于 x 的数量。外层 BIT 按位置拆前缀，内层权值线段树维护值的频次；区间 [l,r] 用前缀 r 减前缀 l-1，kth 将两组根同步向左/右下降。静态查询优先使用前面的主席树；这里各棵内树互不共享节点，也不保存历史版本。

输入 `a[1..n]`，a[0] 忽略，n≥1；值为 ll，可负数、可重复。构造 `BITSeg(a, vals)` 时 vals 收集**未来所有赋值**，初始值由 init 自动加入；也可直接传全部可能值。查询的 x 不必登记。需要先读完修改以离散化，但可按原操作顺序输出；若修改值依赖此前答案且值域未知，应改用已知整数值域的动态内树或“线段树套平衡树”，不能直接套此离散版本。

设 m 为不同候选值数、s=n+vals.size() 为离散化前的总元素数；离散化 $\mathcal{O}(s \log (s+1))$，逐点建树 $\mathcal{O}(n \log (n+1)\log (m+1))$；单点赋值、计数、排名、第 k 小、前驱/后继均为 $\mathcal{O}(\log (n+1)\log (m+1))$。节点按需开辟，频次降为 0 时回收，空间保守上界 $\mathcal{O}(n \log (n+1)\log (m+1)+m)$，初始化另需 $\mathcal{O}(s)$ 临时空间。实际内存通常较大，应结合 `t.capacity()*sizeof(BITSeg::Node)`、回收栈及其他数组估算；启用 int 宏会扩大节点。

`rank(l,r,x)` 是严格小于 x 的数量加 1，即使 x 不存在也可查询；`kth` 从 1 开始，返回原值。`pre/suf` 分别返回严格小于/大于 x 的值，**调用前须保证存在**，可用 `less(l,r,x)>0`、`leq(l,r,x)<r-l+1` 判断。所有区间均为 1-based 非空闭区间。`set(p,x)` 中 x 必须已登记；`init(a,vals)` 清空旧结构。

```c++
struct BITSeg {
    struct Node {
        int ch[2] = {}, cnt = 0;
    };
    int n, m;
    vector<Node> t;
    vector<int> rt, a, rec; // BIT 根、当前位置的压缩值、可复用节点
    vector<ll> val; // 排序去重后的原值，下标 0..m-1
    BITSeg(const vector<ll> &v, vector<ll> vals) {
        init(v, vals);
    }
    int id(ll x) const {
        return lower_bound(val.begin(), val.end(), x) - val.begin() + 1;
    }
    int alloc() {
        if (rec.empty()) {
            t.emplace_back();
            return (int)t.size() - 1;
        }
        int p = rec.back();
        rec.pop_back();
        t[p] = Node{};
        return p;
    }
    int upd(int p, int l, int r, int x, int d) {
        if (!p) p = alloc();
        t[p].cnt += d;
        if (l < r) {
            int mid = (l + r) / 2, b = x > mid;
            // 不把 t[p].ch[b] 按引用传递；递归可能使 vector 扩容。
            int c = upd(t[p].ch[b], b ? mid + 1 : l, b ? r : mid, x, d);
            t[p].ch[b] = c;
        }
        if (!t[p].cnt) {
            rec.push_back(p);
            return 0;
        }
        return p;
    }
    void change(int p, int x, int d) {
        for (; p <= n; p += p & -p) rt[p] = upd(rt[p], 1, m, x, d);
    }
    void init(const vector<ll> &v, vector<ll> vals) {
        n = (int)v.size() - 1; /* assert(n >= 1) */;
        vals.insert(vals.end(), v.begin() + 1, v.end());
        sort(vals.begin(), vals.end());
        vals.erase(unique(vals.begin(), vals.end()), vals.end());
        val.swap(vals);
        m = val.size();
        t.assign(1, Node{});
        rec.clear();
        rt.assign(n + 1, 0);
        a.assign(n + 1, 0);
        for (int i = 1; i <= n; i++) {
            a[i] = id(v[i]);
            change(i, a[i], 1);
        }
    }
    void set(int p, ll x) {
        /* assert(1 <= p && p <= n) */;
        int k = id(x); /* assert(k <= m && val[k - 1] == x) */;
        if (a[p] == k) return;
        change(p, a[p], -1);
        change(p, k, 1);
        a[p] = k;
    }
    int count(int p, int l, int r, int k) const { // 内树中压缩值 <= k 的个数
        if (!p || k < l) return 0;
        if (r <= k) return t[p].cnt;
        int mid = (l + r) / 2;
        if (k <= mid) return count(t[p].ch[0], l, mid, k);
        return t[t[p].ch[0]].cnt + count(t[p].ch[1], mid + 1, r, k);
    }
    int prefix(int p, int k) const {
        int ans = 0;
        for (; p; p -= p & -p) ans += count(rt[p], 1, m, k);
        return ans;
    }
    int less(int l, int r, ll x) const {
        int k = lower_bound(val.begin(), val.end(), x) - val.begin();
        return prefix(r, k) - prefix(l - 1, k);
    }
    int leq(int l, int r, ll x) const {
        int k = upper_bound(val.begin(), val.end(), x) - val.begin();
        return prefix(r, k) - prefix(l - 1, k);
    }
    int rank(int l, int r, ll x) const {
        return less(l, r, x) + 1;
    }
    ll kth(int l, int r, int k) const {
        /* assert(1 <= l && l <= r && r <= n && 1 <= k && k <= r - l + 1) */;
        vector<int> L, R;
        for (int p = l - 1; p; p -= p & -p) L.push_back(rt[p]);
        for (int p = r; p; p -= p & -p) R.push_back(rt[p]);
        int lo = 1, hi = m;
        while (lo < hi) {
            int mid = (lo + hi) / 2, cnt = 0;
            for (int p : R) cnt += t[t[p].ch[0]].cnt;
            for (int p : L) cnt -= t[t[p].ch[0]].cnt;
            int b = k > cnt;
            if (b) {
                k -= cnt;
                lo = mid + 1;
            } else hi = mid;
            for (int &p : L) p = t[p].ch[b];
            for (int &p : R) p = t[p].ch[b];
        }
        return val[lo - 1];
    }
    ll pre(int l, int r, ll x) const {
        return kth(l, r, less(l, r, x));
    }
    ll suf(int l, int r, ll x) const {
        return kth(l, r, leq(l, r, x) + 1);
    }
};
```

树套树不是单一结构，按查询维度选择内外层：本版适合已知全部修改值时的带修区间顺序统计；修改值必须在线产生时使用下一节的[线段树套平衡树](#segtree-balanced-tree)。已知更新坐标的二维计数/矩形和可直接复用[离线离散化二维 BIT](#sparse-bit-2d)，避免再建一套同用途模板。

### 树套树：线段树套 Treap

<a id="segtree-balanced-tree"></a>

用于**在线单点赋值＋区间顺序统计**。外层是按位置维护的迭代式线段树，每个节点的根指向共享节点池中的随机 Treap，键为 `(值, 位置)`；位置使相同值的元素仍有不同键。支持区间内小于/不大于 x 的数量、排名、第 k 小、严格前驱/后继，修改值无需预先离散化。若能提前收集全部修改值，优先使用上一节 `BITSeg`：它的第 k 小更快。

输入 `a[1..n]`，a[0] 忽略，n≥1；值为 ll，可重复并支持 `LLONG_MIN..LLONG_MAX`。`init(a)` 清空并重建，所有区间均为 1-based 非空闭区间。各棵内树互不共享节点；删除节点会进入全局回收栈，后续插入复用。`#define int long long` 下可直接使用，但节点下标和子树大小会变宽，内存随之增加。

建树期望 $\mathcal{O}(n \log ^{2}(n+1))$，单点赋值、`less/leq/rank/pre/suf` 期望 $\mathcal{O}(\log ^{2}(n+1))$，最坏可退化为 $\mathcal{O}(n\log (n+1))$；空间 $\mathcal{O}(n \log (n+1))$。`kth` 在完整 64 位有符号整数域上二分 64 次，期望 $\mathcal{O}(64\log ^{2}(n+1))$、最坏 $\mathcal{O}(64n\log (n+1))$；这里的 64 不能漏写成一次 $\mathcal{O}(\log ^{2} n)$。随机优先级不提供单次最坏对数保证。

`rank(l,r,x)=less(l,r,x)+1`，x 不必存在；`kth` 从 1 开始。`pre/suf` 分别返回严格小于/大于 x 的值，调用前须用 `less(l,r,x)>0`、`leq(l,r,x)<r-l+1` 保证答案存在。`set(p,x)` 中 x 可为此前从未出现的值。

```c++
struct SegTreap {
    using Key = pair<ll, ll>; // (值, 位置)，位置用于区分重复值
    struct Node {
        Key key{};
        ull pri = 0;
        int ch[2] = {}, sz = 0;
    };
    int n = 0;
    vector<ll> a;
    vector<int> rt, rec;
    vector<Node> t;
    mt19937_64 rng;

    SegTreap(const vector<ll> &v, ull seed = chrono::steady_clock::now().time_since_epoch().count()) : rng(seed) {
        init(v);
    }
    int size(int p) const {
        return t[p].sz;
    }
    void pull(int p) {
        t[p].sz = size(t[p].ch[0]) + size(t[p].ch[1]) + 1;
    }
    int make(const Key &key) {
        int p;
        if (rec.empty()) {
            t.emplace_back();
            p = (int)t.size() - 1;
        } else {
            p = rec.back();
            rec.pop_back();
            t[p] = Node{};
        }
        t[p].key = key;
        t[p].pri = rng();
        t[p].sz = 1;
        return p;
    }
    int rotate(int p, int d) {
        int q = t[p].ch[d];
        t[p].ch[d] = t[q].ch[d ^ 1];
        t[q].ch[d ^ 1] = p;
        pull(p);
        pull(q);
        return q;
    }
    int insertNode(int p, int q) {
        if (!p) return q;
        int d = t[p].key < t[q].key;
        t[p].ch[d] = insertNode(t[p].ch[d], q);
        if (t[t[p].ch[d]].pri > t[p].pri) p = rotate(p, d);
        else pull(p);
        return p;
    }
    int insert(int p, const Key &key) {
        return insertNode(p, make(key));
    }
    int merge(int x, int y) {
        if (!x || !y) return x | y;
        if (t[x].pri > t[y].pri) {
            t[x].ch[1] = merge(t[x].ch[1], y);
            pull(x);
            return x;
        }
        t[y].ch[0] = merge(x, t[y].ch[0]);
        pull(y);
        return y;
    }
    int erase(int p, const Key &key) {
        /* assert(p) */;
        if (t[p].key == key) {
            int q = merge(t[p].ch[0], t[p].ch[1]);
            rec.push_back(p);
            return q;
        }
        int d = t[p].key < key;
        t[p].ch[d] = erase(t[p].ch[d], key);
        pull(p);
        return p;
    }
    void init(const vector<ll> &v) {
        n = (int)v.size() - 1; /* assert(n >= 1) */;
        a = v;
        rt.assign(2 * n, 0);
        t.assign(1, Node{});
        rec.clear();
        for (int i = 1; i <= n; i++)
            for (int p = i + n - 1; p; p >>= 1) rt[p] = insert(rt[p], {a[i], i});
    }
    void set(int p, ll x) {
        /* assert(1 <= p && p <= n) */;
        if (a[p] == x) return;
        Key old = {a[p], p}, now = {x, p};
        for (int u = p + n - 1; u; u >>= 1) {
            rt[u] = erase(rt[u], old);
            rt[u] = insert(rt[u], now);
        }
        a[p] = x;
    }
    template <class F> void each(int l, int r, F f) const {
        /* assert(1 <= l && l <= r && r <= n) */;
        for (l += n - 1, r += n - 1; l <= r; l >>= 1, r >>= 1) {
            if (l & 1) f(rt[l++]);
            if (!(r & 1)) f(rt[r--]);
        }
    }
    int countRoot(int p, const Key &key) const {
        int ans = 0;
        while (p)
            if (t[p].key < key) {
                ans += size(t[p].ch[0]) + 1;
                p = t[p].ch[1];
            } else p = t[p].ch[0];
        return ans;
    }
    int countKey(int l, int r, const Key &x) const {
        int ans = 0;
        each(l, r, [&](int p) { ans += countRoot(p, x); });
        return ans;
    }
    int less(int l, int r, ll x) const {
        return countKey(l, r, {x, 0});
    }
    int leq(int l, int r, ll x) const {
        return countKey(l, r, {x, (ll)n + 1});
    }
    int rank(int l, int r, ll x) const {
        return less(l, r, x) + 1;
    }

    ll pre(int l, int r, ll x) const {
        bool ok = false;
        ll ans = 0;
        each(l, r, [&](int p) {
            int q = 0;
            Key key = {x, 0};
            while (p)
                if (t[p].key < key) q = p, p = t[p].ch[1];
                else p = t[p].ch[0];
            if (q && (!ok || t[q].key.first > ans)) ans = t[q].key.first, ok = true;
        });
        /* assert(ok) */;
        return ans;
    }
    ll suf(int l, int r, ll x) const {
        bool ok = false;
        ll ans = 0;
        each(l, r, [&](int p) {
            int q = 0;
            Key key = {x, (ll)n + 1};
            while (p)
                if (!(t[p].key < key)) q = p, p = t[p].ch[0];
                else p = t[p].ch[1];
            if (q && (!ok || t[q].key.first < ans)) ans = t[q].key.first, ok = true;
        });
        /* assert(ok) */;
        return ans;
    }
    static ll decode(ull x) { // 有序编码 [0,2^64-1] 转回 ll，避免有符号溢出
        x ^= 1ULL << 63;
        ll y;
        memcpy(&y, &x, sizeof y);
        return y;
    }
    ll kth(int l, int r, int k) const {
        /* assert(1 <= l && l <= r && r <= n && 1 <= k && k <= r - l + 1) */;
        ull lo = 0, hi = ULLONG_MAX;
        while (lo < hi) {
            ull mid = lo + ((hi - lo) >> 1);
            if (leq(l, r, decode(mid)) >= k) hi = mid;
            else lo = mid + 1;
        }
        return decode(lo);
    }
};
```

### 可持久化数组：任意历史版本单点赋值

<a id="persistent-array"></a>

按位置维护，与主席树的值域频次不同。输入 1-based 带占位数组，n>0；初始版本编号 0。`set(ver,pos,value)` 从指定历史版本派生新版本，返回新版本号；`get(ver,pos)` 查询；`clone(ver)` 仅复制版本根，$\mathcal{O}(1)$。旧版本保持不变，允许从旧分支继续修改。

初建 $\mathcal{O}(n)$，单点赋值/查询 $\mathcal{O}(\log n)$，q 次修改的节点空间 $\mathcal{O}(n+q \log (n+1))$，版本根另占 $\mathcal{O}(\text{版本数})$。不含懒标记、不支持区间更新；T 须可默认构造和复制，节点下标及版本数须能存入 int。init 重建并废弃全部旧版本。

```c++
template <class T = ll> struct PersistentArray {
    struct Node {
        int ch[2] = {};
        T value{};
    };
    int n = 0;
    vector<Node> t;
    vector<int> root;
    PersistentArray(const vector<T> &a) {
        init(a);
    }
    int alloc(const Node &node) {
        Node copy = node;
        t.push_back(copy);
        return (int)(t.size()) - 1;
    }
    int build(int l, int r, const vector<T> &a) {
        int p = alloc(Node{});
        if (l == r) {
            t[p].value = a[l];
            return p;
        }
        int m = (l + r) / 2;
        int x = build(l, m, a), y = build(m + 1, r, a);
        t[p].ch[0] = x;
        t[p].ch[1] = y;
        return p;
    }
    void init(const vector<T> &a) {
        n = (int)(a.size()) - 1; /* assert(n > 0) */;
        t.assign(1, Node{});
        root.clear();
        root.push_back(build(1, n, a));
    }
    void check(int ver, int pos) const {
        /* assert(0 <= ver && ver < (int)(root.size()) && 1 <= pos && pos <= n) */;
    }
    int update(int old, int l, int r, int pos, const T &value) {
        int p = alloc(t[old]);
        if (l == r) {
            t[p].value = value;
            return p;
        }
        int m = (l + r) / 2, d = pos > m;
        int son = update(t[p].ch[d], d ? m + 1 : l, d ? r : m, pos, value);
        t[p].ch[d] = son;
        return p;
    }
    int set(int ver, int pos, const T &value) {
        check(ver, pos);
        T copy = value; // 即便实参来自节点池也先复制
        int p = update(root[ver], 1, n, pos, copy);
        root.push_back(p);
        return (int)(root.size()) - 1;
    }
    T get(int ver, int pos) const {
        check(ver, pos);
        int p = root[ver], l = 1, r = n;
        while (l < r) {
            int m = (l + r) / 2;
            if (pos <= m) {
                p = t[p].ch[0];
                r = m;
            } else {
                p = t[p].ch[1];
                l = m + 1;
            }
        }
        return t[p].value;
    }
    int clone(int ver) {
        check(ver, 1);
        int p = root[ver];
        root.push_back(p);
        return (int)(root.size()) - 1;
    }
};
```

调用：`PersistentArray<ll> a(vector<ll>{0,10,20}); int v=a.set(0,1,7); int w=a.set(0,2,9);` 后 `a.get(0,1)==10`、`a.get(v,2)==20`、`a.get(w,1)==10`，两次修改属于不同分支。

### KD Tree

当前模板维护二维点，支持插入，以及统计各坐标都不大于给定坐标的点数（重复点分别计数）。容量最多 N-1 个点，调用 init 可清空。查询通过包围盒剪枝，但本实现按最大方差选轴，不能直接套用循环选轴 KD Tree 的最坏复杂度结论；保守按单次查询最坏 $\mathcal{O}(n)$ 估计，整棵重建期望 $\mathcal{O}(n \log n)$。插入按大小失衡重建，保守期望均摊 $\mathcal{O}(\log ^{2}(n+1))$，触发大重建时单次可达期望 $\mathcal{O}(n \log (n+1))$；空间 $\mathcal{O}(N)$，init 清零时间 $\mathcal{O}(N)$。report(rt,sx,tx,sy,ty,res) 将闭矩形内点的 0-based 插入编号追加到 res，最坏 $\mathcal{O}(n)$，不保证输出顺序。对象含大数组，建议放全局或用堆分配；空节点大小已在 init 中清零。

调用：`vector<int> p{2,3}; kd.insert(p); vector<int> q{4,5};` 后 `kd.query(q)==1`。重复点分别计数；批量 build 前必须初始化 id 和 cur，具体见下方完整示例。

```c++
struct KDT {
    constexpr static int N = 1e5 + 10, K = 2;
    double alpha = 0.725;
    struct node {
        int info[K];
        int mn[K], mx[K];
    } tr[N];
    int ls[N], rs[N], siz[N], id[N], d[N];
    int idx, rt, cur;
    int ans;
    KDT() {
        init();
    }
    void init() {
        rt = cur = idx = ans = 0;
        memset(siz, 0, sizeof siz);
        memset(ls, 0, sizeof ls);
        memset(rs, 0, sizeof rs);
        memset(d, 0, sizeof d);
    }
    void apply(int p, int son) {
        if (son) {
            for (int i = 0; i < K; i++) {
                tr[p].mn[i] = min(tr[p].mn[i], tr[son].mn[i]);
                tr[p].mx[i] = max(tr[p].mx[i], tr[son].mx[i]);
            }
            siz[p] += siz[son];
        }
    }
    void maintain(int p) {
        for (int i = 0; i < K; i++) {
            tr[p].mn[i] = tr[p].info[i];
            tr[p].mx[i] = tr[p].info[i];
        }
        siz[p] = 1;
        apply(p, ls[p]);
        apply(p, rs[p]);
    }
    int build(int l, int r) {
        if (l > r) return 0;
        array<ld, K> avg{};
        for (int i = 0; i < K; i++) {
            for (int j = l; j <= r; j++) avg[i] += tr[id[j]].info[i];
            avg[i] /= (r - l + 1);
        }
        array<ld, K> var{};
        for (int i = 0; i < K; i++)
            for (int j = l; j <= r; j++) var[i] += (tr[id[j]].info[i] - avg[i]) * (tr[id[j]].info[i] - avg[i]);
        int mid = (l + r) / 2;
        int x = max_element(var.begin(), var.end()) - var.begin();
        nth_element(id + l, id + mid, id + r + 1, [&](int a, int b) { return tr[a].info[x] < tr[b].info[x]; });
        d[id[mid]] = x;
        ls[id[mid]] = build(l, mid - 1);
        rs[id[mid]] = build(mid + 1, r);
        maintain(id[mid]);
        return id[mid];
    }
    void print(int p) {
        if (!p) return;
        print(ls[p]);
        id[++idx] = p;
        print(rs[p]);
    }
    void rebuild(int &p) {
        idx = 0;
        print(p);
        p = build(1, idx);
    }
    bool bad(int p) {
        return alpha * siz[p] <= max(siz[ls[p]], siz[rs[p]]);
    }
    void insert(int &p, int cur) {
        if (!p) {
            p = cur;
            maintain(p);
            return;
        }
        if (tr[p].info[d[p]] > tr[cur].info[d[p]]) insert(ls[p], cur);
        else insert(rs[p], cur);
        maintain(p);
        if (bad(p)) rebuild(p);
    }
    void insert(const vector<int> &a) {
        /* assert((int)(a.size()) == K && cur + 1 < N) */;
        cur++;
        for (int i = 0; i < K; i++) tr[cur].info[i] = a[i];
        insert(rt, cur);
    }
    bool out(int p, const vector<int> &a) {
        for (int i = 0; i < K; i++)
            if (a[i] < tr[p].mn[i]) return true;
        return false;
    }
    bool in(int p, const vector<int> &a) {
        for (int i = 0; i < K; i++)
            if (a[i] < tr[p].info[i]) return false;
        return true;
    }
    bool all(int p, const vector<int> &a) {
        for (int i = 0; i < K; i++)
            if (a[i] < tr[p].mx[i]) return false;
        return true;
    }
    void query(int p, const vector<int> &a) {
        if (!p) return;
        if (out(p, a)) return;
        if (all(p, a)) {
            ans += siz[p];
            return;
        }
        if (in(p, a)) ans++;
        query(ls[p], a);
        query(rs[p], a);
    }
    int query(const vector<int> &a) {
        ans = 0;
        query(rt, a);
        return ans;
    }
    void report(int p, int sx, int tx, int sy, int ty, vector<int> &res) {
        if (!p) return;

        // 整棵子树的包围盒与查询矩形不相交，直接跳过
        if (tr[p].mx[0] < sx || tr[p].mn[0] > tx || tr[p].mx[1] < sy || tr[p].mn[1] > ty) return;

        int x = tr[p].info[0], y = tr[p].info[1];
        if (sx <= x && x <= tx && sy <= y && y <= ty) res.push_back(p - 1);

        report(ls[p], sx, tx, sy, ty, res);
        report(rs[p], sx, tx, sy, ty, res);
    }
};
```

### 单调栈：左右边界与贡献

<a id="monotonic-stack"></a>

普通 0-based 数组，返回每个元素的左右最近边界 L、R；不存在时分别为 -1、n。默认 less 比较，`ls=true` 表示左边找严格更小，false 表示小于等于；`rs` 同理。改用 greater 可找更大/大于等于。每次 $\mathcal{O}(n)$ 时间、$\mathcal{O}(n)$ 空间，空数组可用。

子数组最小值贡献默认使用“左严格小于，右小于等于”：重复最小值交给更靠右的位置，元素 i 负责 (i-L[i])*(R[i]-i) 个子数组。左右都严格会重复计数，都非严格会漏计；只查询最近边界时则按题意选择，不必一严一松。

```c++
// monoBounds：求左右第一个更优元素的边界
template <class T, class Better = less<T>> pair<vector<int>, vector<int>>
monoBounds(const vector<T> &a, Better better = {}, bool ls = true, bool rs = false) { // 左/右边界是否严格更优
    int n = (int)(a.size());
    vector<int> L(n, -1), R(n, n), st;
    auto pop = [&](int j, int i, bool strict) {
        return strict ? !better(a[j], a[i]) : better(a[i], a[j]);
    };
    for (int i = 0; i < n; ++i) {
        while (!st.empty() && pop(st.back(), i, ls)) st.pop_back();
        if (!st.empty()) L[i] = st.back();
        st.push_back(i);
    }
    st.clear();
    for (int i = n - 1; i >= 0; --i) {
        while (!st.empty() && pop(st.back(), i, rs)) st.pop_back();
        if (!st.empty()) R[i] = st.back();
        st.push_back(i);
    }
    return {L, R};
}
```

#### 单调栈应用：子数组最小值之和与柱状图

依赖 monoBounds。最小值之和返回 __int128，支持负数；柱状图高度非负、单位柱宽，面积须能存入 ll。空输入返回 0。柱状图使用同一组一严一松边界，在一组相同最小高度中仍有一个位置能覆盖整个候选矩形。

```c++
// monoBounds：求左右第一个更优元素的边界；sumMin：求所有子数组最小值之和；maxRect：求直方图最大矩形面积
i128 sumMin(const vector<ll> &a) {
    auto [L, R] = monoBounds(a);
    i128 ans = 0;
    for (int i = 0; i < (int)(a.size()); ++i) ans += (i128)a[i] * (i - L[i]) * (R[i] - i);
    return ans;
}
ll maxRect(const vector<ll> &h) { // h：柱高
    auto [L, R] = monoBounds(h);
    ll ans = 0;
    for (int i = 0; i < (int)(h.size()); ++i) {
        /* assert(h[i] >= 0) */;
        i128 area = (i128)h[i] * (R[i] - L[i] - 1);
        /* assert(area <= LLONG_MAX) */;
        ans = max(ans, (ll)area);
    }
    return ans;
}
```

### 单调队列

<a id="monotonic-queue"></a>

返回每个长度为 k 的窗口最小值/最大值，输入普通 0-based 数组，1≤k≤n。输出第 i 项对应窗口 [i,i+k-1]；数组值可以为负，支持重复值，时间 $\mathcal{O}(n)$、辅助队列空间 $\mathcal{O}(k)$，输出空间 $\mathcal{O}(n-k+1)$。每次调用独立，无 MAX、IOS 等隐藏依赖。

```c++
// windowExt：求滑动窗口最小值与最大值
template <class T> pair<vector<T>, vector<T>> windowExt(const vector<T> &a, int k) {
    int n = (int)(a.size()); /* assert(1 <= k && k <= n) */;
    deque<int> lo, hi;
    vector<T> mn, mx;
    mn.reserve(n - k + 1);
    mx.reserve(n - k + 1);
    for (int i = 0; i < n; ++i) {
        while (!lo.empty() && lo.front() <= i - k) lo.pop_front();
        while (!hi.empty() && hi.front() <= i - k) hi.pop_front();
        while (!lo.empty() && a[lo.back()] >= a[i]) lo.pop_back();
        while (!hi.empty() && a[hi.back()] <= a[i]) hi.pop_back();
        lo.push_back(i);
        hi.push_back(i);
        if (i >= k - 1) {
            mn.push_back(a[lo.front()]);
            mx.push_back(a[hi.front()]);
        }
    }
    return {mn, mx};
}
```

调用：`auto [mn,mx]=windowExt(vector<int>{1,3,2},2);` 得 mn={1,2}、mx={3,3}。仅需一种最值时可删去另一条队列及对应输出。

### ST 表

<a id="sparse-table"></a>

静态最值主力模板，n>0，1-based 闭区间。先填 in1/in2，再 init；getMax 查询 in1，getMin 查询 in2，需要查询同一数组时两者填相同数据。

调用：`ST st(3); st.in1=st.in2=vector<int>{0,3,-1,2}; st.init();` 后 `st.getMin(1,3)==-1`、`st.getMax(2,3)==2`。重新填入数据后必须再 init；若只需一种最值，可删除另一张表。

用于解决区间可重复贡献问题，合并运算必须满足结合律及幂等性 $x \text{ 运算符 } x=x$ （如区间最大值：$\max(x,x)=x$ 、区间 $\gcd$：$\gcd(x,x)=x$ 等），但是不支持修改操作。不能直接用重叠区间法查询和或异或。$\mathcal O(N\log N)$ 预处理，$\mathcal O(1)$ 查询。

```c++
struct ST {
    const int n, k;
    vector<int> in1, in2;
    vector<vector<int>> Max, Min;
    ST(int n) : n(n), in1(n + 1), in2(n + 1), k(__lg(n)) {
        Max.resize(k + 1, vector<int>(n + 1));
        Min.resize(k + 1, vector<int>(n + 1));
    }
    void init() {
        for (int i = 1; i <= n; i++) {
            Max[0][i] = in1[i];
            Min[0][i] = in2[i];
        }
        for (int i = 0, t = 1; i < k; i++, t <<= 1) {
            const int T = n - (t << 1) + 1;
            for (int j = 1; j <= T; j++) {
                Max[i + 1][j] = max(Max[i][j], Max[i][j + t]);
                Min[i + 1][j] = min(Min[i][j], Min[i][j + t]);
            }
        }
    }
    int getMax(int l, int r) {
        if (l > r) swap(l, r);
        int k = __lg(r - l + 1);
        return max(Max[k][l], Max[k][r - (1 << k) + 1]);
    }
    int getMin(int l, int r) {
        if (l > r) swap(l, r);
        int k = __lg(r - l + 1);
        return min(Min[k][l], Min[k][r - (1 << k) + 1]);
    }
};
```

### 基于状压的线性 RMQ 算法

进阶静态 RMQ：0-based 半开区间 [l,r)，要求 0≤l<r≤n。块宽 B=64，块内用位运算，块间用 ST 表；按固定块宽计，预处理和空间为 $\mathcal{O}(n+(n/B)\log (n/B))$，单次查询 $\mathcal{O}(1)$。GNU/Clang 的内建位运算需可用。

```c++
template <class T, class Cmp = less<T>> struct RMQ {
    const Cmp cmp = Cmp();
    static constexpr unsigned B = 64;
    int n;
    vector<vector<T>> a;
    vector<T> pre, suf, ini;
    vector<ull> stk;
    RMQ() {}
    RMQ(const vector<T> &v) {
        init(v);
    }
    void init(const vector<T> &v) {
        n = v.size();
        pre = suf = ini = v;
        stk.resize(n);
        if (!n) return;
        const int M = (n - 1) / B + 1;
        const int lg = __lg(M);
        a.assign(lg + 1, vector<T>(M));
        for (int i = 0; i < M; i++) {
            a[0][i] = v[i * B];
            for (int j = 1; j < B && i * B + j < n; j++) a[0][i] = min(a[0][i], v[i * B + j], cmp);
        }
        for (int i = 1; i < n; i++)
            if (i % B) pre[i] = min(pre[i], pre[i - 1], cmp);
        for (int i = n - 2; i >= 0; i--)
            if (i % B != B - 1) suf[i] = min(suf[i], suf[i + 1], cmp);
        for (int j = 0; j < lg; j++)
            for (int i = 0; i + (2 << j) <= M; i++) a[j + 1][i] = min(a[j][i], a[j][i + (1 << j)], cmp);
        for (int i = 0; i < M; i++) {
            const int l = i * B;
            const int r = min<int>(n, l + B);
            ull s = 0;
            for (int j = l; j < r; j++) {
                while (s && cmp(v[j], v[__lg(s) + l])) s ^= 1ULL << __lg(s);
                s |= 1ULL << (j - l);
                stk[j] = s;
            }
        }
    }
    T operator()(int l, int r) {
        if (l / B != (r - 1) / B) {
            T ans = min(suf[l], pre[r - 1], cmp);
            l = l / B + 1;
            r = r / B;
            if (l < r) {
                int k = __lg(r - l);
                ans = min({ans, a[k][l], a[k][r - (1 << k)]}, cmp);
            }
            return ans;
        } else {
            int x = B * (l / B);
            return ini[ctzll(stk[r - 1] >> (l - x)) + l];
        }
    }
};
```

### pbds 扩展库实现平衡二叉树

<a id="pbds-ordered-set"></a>

PBDS 头文件与哈希表接口见[语言、标准库与输入输出：pb_ds](<12 - 语言、标准库与输入输出.md#pbds>)。

GNU PBDS 完整操作示例，依赖下方两个头文件，必须放在缺省源的所有宏定义（包括 endl、int）之前。比较器决定键的排序方向；用 (值,编号) 支持重复值，增删和顺序统计 $\mathcal{O}(\log n)$。`order_of_key` 返回严格小于键的数量，`find_by_order` 使用 0-based 序号；示例要求第 k 小及前驱/后继确实存在。join 要求两树键域严格分离，不能当作任意集合并集。

> 附常见成员函数：
>
> ```c++
> empty() / size()
> insert(x) // 插入元素x
> erase(x) // 删除元素/迭代器x
> order_of_key(x) // 严格小于x的元素数（0-based）
> find_by_order(x) // 返回排名为x的元素迭代器
> lower_bound(x) / upper_bound(x) // 返回迭代器
> join(Tree) // 合并严格分离的键域，两个树不能交错或有重复键
> split(x, Tree) // 将大于x的元素放入Tree树
> ```

```c++

#include <ext/pb_ds/tree_policy.hpp>

using namespace __gnu_pbds;
using V = pair<int, int>;
tree<V, null_type, less<V>, rb_tree_tag, tree_order_statistics_node_update> ver;
map<int, int> dic;

signed main() {
    int n;
    cin >> n;
    for (int i = 1, op, x; i <= n; i++) {
        cin >> op >> x;
        if (op == 1) { // 插入一个元素x，允许重复
            ver.insert({x, ++dic[x]});
        } else if (op == 2) { // 删除元素x，若有重复，则任意删除一个
            if (dic[x] > 0) {
                ver.erase({x, dic[x]});
                --dic[x];
            }
        } else if (op == 3) { // 查询元素x的排名（排名定义为比当前数小的数的个数+1）
            cout << ver.order_of_key({x, 1}) + 1 << endl;
        } else if (op == 4) { // 查询排名为x的元素
            cout << ver.find_by_order(--x)->first << endl;
        } else if (op == 5) { // 查询元素x的前驱
            int idx = ver.order_of_key({x, 1}) - 1; // 无论x存不存在，idx都代表x的位置，需要-1
            cout << ver.find_by_order(idx)->first << endl;
        } else if (op == 6) { // 查询元素x的后继
            cout << ver.upper_bound({x, dic[x]})->first << endl;
        }
    }
}
```

### 有序 vector 模拟有序集合

小数据备用示例，不是平衡树。二分定位 $\mathcal{O}(\log n)$，插入/删除因搬移元素为 $\mathcal{O}(n)$，第 k 小直接访问 $\mathcal{O}(1)$，空间 $\mathcal{O}(n)$。循环放在 main 内；第 k 小和前驱/后继查询须保证答案存在。末尾撤销局部缩写宏，避免污染其他模板中的 pre/suf 名称。

### 珂朵莉树 (OD Tree)

分段常值序列：`ODT(n,value)` 初始化 1..n，支持闭区间赋值、加法、第 k 小和幂次和；n=0 时只允许初始化和清空，不能查询。随机数据下常有较好表现，但无条件的对数复杂度保证不成立。设当前段数为 S、操作覆盖 s 段，split 为 $\mathcal{O}(\log S)$，区间加 $\mathcal{O}(\log S+s)$，赋值 $\mathcal{O}(\log S+s)$，第 k 小 $\mathcal{O}(\log S+s \log s)$，幂次和 $\mathcal{O}(\log (S+1)+s \log (x+1)+s)$。对抗数据可能退化，应按题目操作分布选用。值和加法结果须在 ll 范围内，幂指数非负、1≤mod≤LLONG_MAX；模乘使用 i128。节点空间 $\mathcal{O}(S)$，kth 临时空间 $\mathcal{O}(s)$。

调用：`ODT s(4,1); s.assign(2,3,5); s.add(1,2,-1);` 后数组为 [0,4,5,1]，`s.kth(1,4,2)==1`、`s.powersum(1,4,2,100)==42`；init 重置。

**公共依赖**：[公共模运算](<05 - 数论.md#default-mod>)。

```c++
struct ODT {
    struct node {
        int l, r;
        mutable ll v;
        node(int l, int r = -1, ll v = 0) : l(l), r(r), v(v) {}
        bool operator<(const node &o) const {
            return l < o.l;
        }
    };
    set<node> s;
    int n = 0;
    ODT(int len = 0, ll value = 0) {
        init(len, value);
    }
    void init(int len, ll value = 0) {
        /* assert(len >= 0 && len < INT_MAX) */;
        n = len;
        s.clear();
        if (n) s.insert(node(1, n, value));
    }
    auto split(int pos) {
        /* assert(1 <= pos && pos <= n + 1) */;
        if (pos == n + 1) return s.end();
        auto it = s.lower_bound(node(pos));
        if (it != s.end() && it->l == pos) return it;
        it--;
        int l = it->l, r = it->r;
        ll v = it->v;
        s.erase(it);
        s.insert(node(l, pos - 1, v));
        return s.insert(node(pos, r, v)).first;
    }
    void assign(int l, int r, ll x) {
        /* assert(1 <= l && l <= r && r <= n) */;
        auto itr = split(r + 1), itl = split(l);
        s.erase(itl, itr);
        s.insert(node(l, r, x));
    }
    void add(int l, int r, ll x) {
        /* assert(1 <= l && l <= r && r <= n) */;
        auto itr = split(r + 1), itl = split(l);
        for (auto it = itl; it != itr; it++) it->v += x;
    }
    ll kth(int l, int r, int k) {
        /* assert(1 <= l && l <= r && r <= n) */;
        /* assert(1 <= k && k <= r - l + 1) */;
        vector<pair<ll, int>> a;
        auto itr = split(r + 1), itl = split(l);
        for (auto it = itl; it != itr; it++) a.push_back(pair<ll, int>(it->v, it->r - it->l + 1));
        sort(a.begin(), a.end());
        for (auto [val, len] : a) {
            k -= len;
            if (k <= 0) return val;
        }
        return -1; // 合法 k 不会执行到此处
    }
    ll powersum(int l, int r, ll x, ll mod) {
        /* assert(1 <= l && l <= r && r <= n) */;
        auto itr = split(r + 1), itl = split(l);
        ll ans = 0;
        for (auto it = itl; it != itr; it++) ans = (ans + (i128)qpow(it->v, x, mod) * (it->r - it->l + 1)) % mod;
        return ans;
    }
};
```

### 常见结论

题意：（区间移位问题）要求将整个序列左移/右移若干个位置，例如，原序列为 $A=(a_1, a_2, \dots, a_n)$ ，左移 $x$ 位后变为 $A=(a_{x+1}, a_{x+2}, \dots, a_n,a_1,a_2,\dots, a_x)$ 。

区间的端点只是一个数字，即使被改变了，通过一定的转换也能够还原，所以我们可以 $\mathcal O(1)$ 解决这一问题。为了方便计算，我们规定下标从 $0$ 开始，即整个线段的区间为 $[0, n)$ ，随后，使用一个偏移量 `shift` 记录。n>0，左移使用 `shift = (shift + x % n) % n;`，右移使用 `shift = (shift - x % n + n) % n;` 更新偏移量（x 非负）；此后的区间查询/修改前，再将坐标偏移回去即可，下方用 0-based 计算偏移，再转回本章加法/赋值线段树的 1-based modify 接口；单次原操作至多拆成两次区间操作，仍为 $\mathcal{O}(\log n)$。偏移与下标的加法中间量须在 int 范围内。

```c++
cin >> l >> r >> x;
l--; // 坐标修改为 0 开始
r--;
l = (l + shift) % n; // 偏移
r = (r + shift) % n;
if (l > r) { // 区间分离则分别操作
    segt.modify(l + 1, n, x);
    segt.modify(1, r + 1, x);
} else {
    segt.modify(l + 1, r + 1, x);
}
```

### 常见例题

题意：（带修莫队 - 维护队列）要求能够处理以下操作：

- `'Q' l r` ：询问区间 $[l,r]$ 有几个颜色；
- `'R' idx w` ：将下标 $\tt idx$ 的颜色修改为 $\tt w$ 。

输入格式为：第一行 $n$ 和 $q\ (1\le n, q\le 133333)$ 分别代表区间长度和操作数量；第二行 $n$ 个整数 $a_1,a_2\dots,a_n\ (1\le a_i\le 10^6)$ 代表初始颜色；随后 $q$ 行为具体操作。

```c++
const int N = 1e6 + 7;
signed main() {
    int n, q;
    cin >> n >> q;
    vector<int> w(n + 1);
    for (int i = 1; i <= n; i++) cin >> w[i];

    vector<array<int, 4>> query = {{}}; // {左区间, 右区间, 累计修改次数, 下标}
    vector<array<int, 2>> modify = {{}}; // {修改的值, 修改的元素下标}
    for (int i = 1; i <= q; i++) {
        char op;
        cin >> op;
        if (op == 'Q') {
            int l, r;
            cin >> l >> r;
            query.push_back({l, r, (int)modify.size() - 1, (int)query.size()});
        } else {
            int idx, w;
            cin >> idx >> w;
            modify.push_back({w, idx});
        }
    }

    int Knum = max<int>(1, (int)(pow(n, 2.0 / 3))); // 按规模选择块长
    vector<int> K(n + 1);
    for (int i = 1; i <= n; i++) { // 固定块长
        K[i] = (i - 1) / Knum + 1;
    }
    sort(query.begin() + 1, query.end(), [&](auto x, auto y) {
        if (K[x[0]] != K[y[0]]) return x[0] < y[0];
        if (K[x[1]] != K[y[1]]) return (K[x[0]] & 1) ? x[1] < y[1] : x[1] > y[1];
        return (K[x[1]] & 1) ? x[2] < y[2] : x[2] > y[2]; // 时间维蛇形排序
    });

    int l = 1, r = 0, val = 0;
    int t = 0; // 累计修改次数
    vector<int> ans(query.size()), cnt(N);
    for (int i = 1; i < query.size(); i++) {
        auto [ql, qr, qt, id] = query[i];
        auto add = [&](int x) -> void {
            if (cnt[x] == 0) ++val;
            ++cnt[x];
        };
        auto del = [&](int x) -> void {
            --cnt[x];
            if (cnt[x] == 0) --val;
        };
        auto time = [&](int x, int l, int r) -> void {
            if (l <= modify[x][1] && modify[x][1] <= r) { //当修改的位置在询问期间内部时才会改变num的值
                del(w[modify[x][1]]);
                add(modify[x][0]);
            }
            swap(w[modify[x][1]], modify[x][0]); //直接交换修改数组的值与原始值，减少额外的数组开销，且方便复原
        };
        while (l > ql) add(w[--l]);
        while (r < qr) add(w[++r]);
        while (l < ql) del(w[l++]);
        while (r > qr) del(w[r--]);
        while (t < qt) time(++t, ql, qr);
        while (t > qt) time(t--, ql, qr);
        ans[id] = val;
    }
    for (int i = 1; i < ans.size(); i++) cout << ans[i] << endl;
}
```
