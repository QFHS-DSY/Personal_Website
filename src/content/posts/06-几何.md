## 第六章：几何

沿用[第 01 章缺省源](<01 - 基础算法.md>)，GNU C++17，支持 `#define int long long`；断言保留为注释，调用方保证前提。默认 `ld=long double`，普通整数几何优先 `Point<ll>`，大乘积使用 `Point<i128>` 或下方提升位宽的函数。

**复制约定与精度**：基础几何复制「预置函数 → 点线封装 → 叉乘 → 点乘 → 欧几里得距离公式」，再按小节依赖选函数。二维 `projLine/projSeg` 返回「点、距离」，三维 `projLine/projPlane` 沿用「距离、点」。浮点数必须有限；未明确支持退化的直线、平面、单位向量，分别要求端点不同、三点不共线、向量非零。基础点线圆与三角形公式均为 $\mathcal{O}(1)$ 时间和额外空间，例外在对应小节标明。

- 排序、去重、极角比较使用精确关系；**不能把 EPS 放进排序比较器**。`samePt/sign` 才使用 EPS。凸包统一逆时针、不重复末点；卡壳和二分点查询使用 `convexHull(p,1)` 去掉共线中间点。
- EPS 沿用绝对误差 1e-7，它同时用于长度、叉积、面积等不同量纲，适合常规比赛尺度；极小图形、极大平移、近切和近乎平行数据要按题意缩放或调整判定。`long double` 的有效精度依赖编译器，不能把本地测试当作所有平台保证。
- `Point<T>` 的相减、点积、叉积仍按 T 计算；**赋给 i128 并不能挽救此前已发生的溢出**。应在相减/相乘前扩宽。`Point<i128>` 也不覆盖任意 64 位坐标极值的全部乘积；输出 i128 需使用专用输出函数。
- 圆周求交与圆盘面积是不同用途；边界返回值见下表。半平面交需保证交集有界，只输出正面积结果。三维旧 `triInter` 仅检测内部穿越；需要共面或边界相交时使用 `triInterFull`。

```c++
using Point = complex<double>;

double cross(const Point &a, const Point &b) {

double dot(const Point &a, const Point &b) {
```

### 平面几何必要初始化

#### 字符串读入浮点数

把十进制串精确放大 $10^k$ 倍，时间 $\mathcal{O}(\text{串长}+k)$、空间 $\mathcal{O}(\text{串长}+k)$。支持正负号、小数点；不支持科学计数法。要求 k≥0、小数位数≤k，结果可存入 int（启用 int 宏后为 ll）。先用 stoll 解析，避免 int 宏下仍受 stoi 的 32 位限制。

```c++
const int Knum = 4;
int read(int k = Knum) {
    string s;
    cin >> s;
    auto p = s.find('.');
    int d = p == string::npos ? 0 : s.size() - p - 1;
    /* assert(k >= d) */;
    if (p != string::npos) s.erase(p, 1);
    s.append(k - d, '0');
    return stoll(s);
}
```

#### 预置函数

以下 EPS 沿用原值 1e-7；sign/equal 用于几何判定，不能用作排序比较器。fgcd 仅是带阈值的余数迭代，浮点数不存在通用的精确 gcd；已知小数位数时优先放大为整数再求 gcd。不要与第 01 章同名 sign 一起复制。

```c++
const ld PI = acosl(-1.L);
const ld EPS = 1e-7;
#define cc(x) cout << fixed << setprecision(x);

ld fgcd(ld x, ld y) { // 实数域gcd
    return abs(y) < EPS ? abs(x) : fgcd(y, fmod(x, y));
}
template <class T, class S> bool equal(T x, S y) {
    return -EPS < x - y && x - y < EPS;
}
template <class T> int sign(T x) {
    if (-EPS < x && x < EPS) return 0;
    return x < 0 ? -1 : 1;
}
```

#### 点线封装

排序和 operator== 使用精确字典序/坐标比较，保证排序与有序容器的要求；几何近似判等调用 samePt。浮点坐标须为有限值，EPS 不参与排序。Pd、Ld 分别是浮点点和直线类型，ld 只表示标量。

```c++
// samePt：几何点近似判等；整数点精确比较
template <class T> struct Point {
    T x, y;
    Point(T x_ = 0, T y_ = 0) : x(x_), y(y_) {} // 初始化
    template <class U> operator Point<U>() const { // 自动类型匹配
        return Point<U>(U(x), U(y));
    }
    Point &operator+=(Point p) & {
        return x += p.x, y += p.y, *this;
    }
    Point &operator+=(T t) & {
        return x += t, y += t, *this;
    }
    Point &operator-=(Point p) & {
        return x -= p.x, y -= p.y, *this;
    }
    Point &operator-=(T t) & {
        return x -= t, y -= t, *this;
    }
    Point &operator*=(T t) & {
        return x *= t, y *= t, *this;
    }
    Point &operator/=(T t) & {
        return x /= t, y /= t, *this;
    }
    Point operator-() const {
        return Point(-x, -y);
    }
    friend Point operator+(Point a, Point b) {
        return a += b;
    }
    friend Point operator+(Point a, T b) {
        return a += b;
    }
    friend Point operator-(Point a, Point b) {
        return a -= b;
    }
    friend Point operator-(Point a, T b) {
        return a -= b;
    }
    friend Point operator*(Point a, T b) {
        return a *= b;
    }
    friend Point operator*(T a, Point b) {
        return b *= a;
    }
    friend Point operator/(Point a, T b) {
        return a /= b;
    }
    friend bool operator<(Point a, Point b) {
        return a.x != b.x ? a.x < b.x : a.y < b.y;
    }
    friend bool operator>(Point a, Point b) {
        return b < a;
    }
    friend bool operator==(Point a, Point b) {
        return !(a < b) && !(b < a);
    }
    friend bool operator!=(Point a, Point b) {
        return a < b || b < a;
    }
    friend auto &operator>>(istream &is, Point &p) {
        return is >> p.x >> p.y;
    }
    friend auto &operator<<(ostream &os, Point p) {
        return os << "(" << p.x << ", " << p.y << ")";
    }
};
template <class T> struct Line {
    Point<T> a, b;
    Line(Point<T> a_ = Point<T>(), Point<T> b_ = Point<T>()) : a(a_), b(b_) {}
    template <class U> operator Line<U>() const { // 自动类型匹配
        return Line<U>(Point<U>(a), Point<U>(b));
    }
    friend auto &operator<<(ostream &os, Line l) {
        return os << "<" << l.a << ", " << l.b << ">";
    }
};
using Pd = Point<ld>;
using Ld = Line<ld>;
template <class T> bool samePt(Point<T> a, Point<T> b) {
    if (is_integral<T>::value) return a == b;
    return equal(a.x, b.x) && equal(a.y, b.y);
}
```

#### 叉乘

定义公式 $a\times b=|a||b|\sin \theta$。

```c++
template <class T> T cross(Point<T> a, Point<T> b) { // 叉乘
    return a.x * b.y - a.y * b.x;
}
template <class T> T cross(Point<T> p1, Point<T> p2, Point<T> p0) { // 叉乘 (p1 - p0) x (p2 - p0);
    return cross(p1 - p0, p2 - p0);
}
```

#### 点乘

定义公式 $a\cdot b=|a||b|\cos \theta$。

```c++
template <class T> T dot(Point<T> a, Point<T> b) { // 点乘
    return a.x * b.x + a.y * b.y;
}
template <class T> T dot(Point<T> p1, Point<T> p2, Point<T> p0) { // 点乘 (p1 - p0) * (p2 - p0);

}
```

#### 欧几里得距离公式

最常用的距离公式。**需要注意**，开根号会丢失精度，如无强制要求，先不要开根号，留到最后一步一起开。

```c++
template <class T> ld dis(T x1, T y1, T x2, T y2) {
    ld dx = ld(x1) - ld(x2), dy = ld(y1) - ld(y2); // 先转换再相减，避免整数溢出
    return hypotl(dx, dy);
}
template <class T> ld dis(Point<T> a, Point<T> b) {
    return dis(a.x, a.y, b.x, b.y);
}
```

#### 曼哈顿距离公式

```c++
template <class T> T dis1(Point<T> p1, Point<T> p2) { // 曼哈顿距离公式
    return abs(p1.x - p2.x) + abs(p1.y - p2.y);
}
```

#### 将向量转换为单位向量

```c++
// unit：向量单位化，要求非零
Point<ld> unit(Point<ld> vec) { // 转换为单位向量
    return vec / sqrt(vec.x * vec.x + vec.y * vec.y);
```

#### 向量旋转

将当前向量移动至原点后顺时针旋转 $90^{\circ}$ ，即获取垂直于当前向量的、起点为原点的向量。在计算垂线时非常有用。例如，要想获取点 $a$ 绕点 $o$ 顺时针旋转 $90^{\circ}$ 后的点，可以这样书写代码：`auto ans = o + rotate(o, a);` ；如果是逆时针旋转，那么只需更改符号即可：`auto ans = o - rotate(o, a);` 。

```c++
template <class T> Point<T> rotate(Point<T> p1, Point<T> p2) { // 旋转
    Point<T> vec = p1 - p2;
    return {-vec.y, vec.x};
}
```

#### 距离平方与整数判向

<a id="geo-polar"></a>

$\mathcal{O}(1)$。`norm2/dis2` 按 T 运算，适合比较距离而不立即开方；要求中间值可表示。`orient(a,b,c)` 返回 (b-a)×(c-a)，输入为整数且在相减前扩宽为 i128；正数为逆时针，零为共线。它仍要求叉积可存入 i128。

```c++
template <class T> T norm2(Point<T> p) {
    return dot(p, p);
}
template <class T> T dis2(Point<T> a, Point<T> b) {
    return norm2(a - b);
}
template <class T> i128 orient(Point<T> a, Point<T> b, Point<T> c) {
    i128 x = i128(b.x) - a.x, y = i128(b.y) - a.y;
    i128 u = i128(c.x) - a.x, v = i128(c.y) - a.y;
    return x * v - y * u;
}
```

#### 极角排序

绕原点，从 +x 轴开始逆时针排序；同向向量按长度升序，零向量放最前。单次比较 $\mathcal{O}(1)$，排序 $\mathcal{O}(n \log n)$。依赖 Point/cross/dot；整数可先转 Point<i128>，浮点不使用 EPS。坐标差、叉积、长度平方必须可表示。围绕点 o 排序时先取 p-o。

```c++
template <class T> bool polarLess(Point<T> a, Point<T> b) {
    bool za = a.x == 0 && a.y == 0, zb = b.x == 0 && b.y == 0;
    if (za || zb) return za && !zb;
    auto half = [](Point<T> p) {
        return p.y < 0 || (p.y == 0 && p.x < 0);
    };
    if (half(a) != half(b)) return half(a) < half(b);
    T c = cross(a, b);
    return c != 0 ? c > 0 : dot(a, a) < dot(b, b);
}
```

#### 弧度角度相互转换

#### 正弦定理

$\dfrac{a}{\sin A}=\dfrac{b}{\sin B}=\dfrac{c}{\sin C}=2R$ ，其中 $R$ 为三角形外接圆半径；

#### 余弦定理（已知三角形三边，求角）

$\cos C=\dfrac{a^2+b^2-c^2}{2ab},\cos B=\dfrac{a^2+c^2-b^2}{2ac},\cos A=\dfrac{b^2+c^2-a^2}{2bc}$。可以借此推导出三角形面积公式 $S_{\triangle ABC}=\dfrac{ab\cdot\sin C}{2}=\dfrac{bc\cdot\sin A}{2}=\dfrac{ac\cdot\sin B}{2}$。

注意，计算格式是：由 $b,c,a$ 三边求 $\angle A$；由 $a, c, b$ 三边求 $\angle B$；由 $a, b, c$ 三边求 $\angle C$。

```c++
ld angle(ld a, ld b, ld c) { // 余弦定理
    /* assert(a > 0 && b > 0 && c >= 0) */;
    ld val = acosl(max(-1.L, min(1.L, (a * a + b * b - c * c) / (2 * a * b)))); // 计算弧度
    return val;
}
```

#### 求两向量的夹角

能够计算 $[0^{\circ},180^{\circ}]$ 区间的角度。

```c++
ld angle(Point<ld> a, Point<ld> b) {
    ld val = abs(cross(a, b));
    return abs(atan2(val, a.x * b.x + a.y * b.y));
}
```

#### 向量旋转任意角度

逆时针旋转，转换公式：$\left\{\begin{matrix}
x'=x\cos \theta-y\sin \theta \\
y'=x\sin \theta+y\cos \theta
\end{matrix}\right.$

```c++
Point<ld> rotate(Point<ld> p, ld rad) {
    return {p.x * cos(rad) - p.y * sin(rad), p.x * sin(rad) + p.y * cos(rad)};
}
```

#### 点绕点旋转任意角度

逆时针旋转，转换公式：$\left\{\begin{matrix}
x'=(x_0-x_1)\cos\theta-(y_0-y_1)\sin\theta+x_1 \\
y'=(x_0-x_1)\sin\theta+(y_0-y_1)\cos\theta+y_1
\end{matrix}\right.$

```c++
Point<ld> rotate(Point<ld> a, Point<ld> b, ld rad) {
    ld x = (a.x - b.x) * cos(rad) - (a.y - b.y) * sin(rad) + b.x;
    ld y = (a.x - b.x) * sin(rad) + (a.y - b.y) * cos(rad) + b.y;
    return {x, y};
}
```

#### 两点是否在直线同侧/异侧

```c++
// sameSide：判断两点在直线同侧；oppSide：判断两点在直线异侧
template <class T> bool sameSide(Point<T> p1, Point<T> p2, Line<T> vec) {
    int val = sign(cross(p1, vec.a, vec.b)) * sign(cross(p2, vec.a, vec.b));
    return sign(val) == 1;
}
template <class T> bool oppSide(Point<T> p1, Point<T> p2, Line<T> vec) {
    int val = sign(cross(p1, vec.a, vec.b)) * sign(cross(p2, vec.a, vec.b));
    return sign(val) == -1;
}
```

#### 两直线相交交点

在使用前需要先判断直线是否平行。

```c++
// lineInter：求直线交点
Pd lineInter(Ld l1, Ld l2) {
    ld val = cross(l2.b - l2.a, l1.a - l2.a) / cross(l2.b - l2.a, l1.a - l1.b);
    return l1.a + (l1.b - l1.a) * val;
}
```

#### 两直线是否平行/垂直/相同

```c++
template <class T> bool linePar(Line<T> a, Line<T> b) {
    return sign(cross(a.b - a.a, b.b - b.a)) == 0;
}
template <class T> bool linePerp(Line<T> a, Line<T> b) {
    return sign(dot(a.b - a.a, b.b - b.a)) == 0;
}
```

#### 点到直线的最近距离与最近点

```c++
pair<Pd, ld> projLine(Pd p, Ld l) { // 非退化直线；返回垂足、距离
    Pd v = l.b - l.a, q = l.a + v * (dot(p - l.a, v) / dot(v, v));

}
```

如果只需要计算最近距离，下方的写法可以减少书写的代码量，效果一致。

```c++
// distLine：计算点到直线的距离
template <class T> ld distLine(Point<T> p, Line<T> l) {
    ld ans = cross(p, l.a, l.b);
    return abs(ans) / dis(l.a, l.b); // 面积除以底边长
}
```

#### 点是否在线段上

```c++
// samePt：几何点近似判等；整数点精确比较；onSeg：判断点在线段上，包含端点；inSeg：判断点在线段内部，排除端点
template <class T> bool onSeg(Point<T> p, Line<T> l) { // 端点也算作在直线上
    return sign(cross(p, l.a, l.b)) == 0 && min(l.a.x, l.b.x) <= p.x && p.x <= max(l.a.x, l.b.x) &&
           min(l.a.y, l.b.y) <= p.y && p.y <= max(l.a.y, l.b.y);
}
template <class T> bool inSeg(Point<T> p, Line<T> l) { // 端点不算
    return onSeg(p, l) && !samePt(p, l.a) && !samePt(p, l.b);
}
```

#### 两线段是否相交及交点

$\mathcal{O}(1)$ 时间和空间，支持点线段。segInter 返回 `(状态, p, q)`：0 不相交；1 内部相交；2 共线重叠（p、q 为重叠端点）；3 端点接触。状态 1、3 中 p=q。输入先转为浮点，返回点恒为 Pd，整数输入也不截断交点；极大整数的精确判交应调用 segCross，并保证整数中间值不溢出。

```c++
template <class T> tuple<int, Pd, Pd> segInter(Line<T> x, Line<T> y) {
    Ld a = x, b = y;
    if (a.a == a.b) return onSeg(a.a, b) ? tuple<int, Pd, Pd>{3, a.a, a.a} : tuple<int, Pd, Pd>{0, {}, {}};
    if (b.a == b.b) return onSeg(b.a, a) ? tuple<int, Pd, Pd>{3, b.a, b.a} : tuple<int, Pd, Pd>{0, {}, {}};
    if (sign(cross(a.b - a.a, b.b - b.a)) == 0) {
        if (sign(cross(a.b - a.a, b.a - a.a)) != 0) return {0, {}, {}};
        if (a.b < a.a) swap(a.a, a.b);
        if (b.b < b.a) swap(b.a, b.b);
        Pd p = max(a.a, b.a), q = min(a.b, b.b);
        if (q < p) return {0, {}, {}};
        return {samePt(p, q) ? 3 : 2, p, q};
    }
    int s1 = sign(cross(a.b - a.a, b.a - a.a)), s2 = sign(cross(a.b - a.a, b.b - a.a));
    int t1 = sign(cross(b.b - b.a, a.a - b.a)), t2 = sign(cross(b.b - b.a, a.b - b.a));
    if (s1 * s2 > 0 || t1 * t2 > 0) return {0, {}, {}};
    Pd p = lineInter(a, b);
    return {s1 * s2 < 0 && t1 * t2 < 0 ? 1 : 3, p, p};
}
```

只判断是否相交时，用下方快速排斥＋跨立实验；重叠、端点接触均为 true。依赖 Point/Line、cross、sign，不依赖求交点函数。

```c++
template <class T> bool segCross(Line<T> a, Line<T> b) {
    if (max(a.a.x, a.b.x) < min(b.a.x, b.b.x) || max(b.a.x, b.b.x) < min(a.a.x, a.b.x) ||
        max(a.a.y, a.b.y) < min(b.a.y, b.b.y) || max(b.a.y, b.b.y) < min(a.a.y, a.b.y))
        return false;
    return sign(cross(a.b - a.a, b.a - a.a)) * sign(cross(a.b - a.a, b.b - a.a)) <= 0 &&
           sign(cross(b.b - b.a, a.a - b.a)) * sign(cross(b.b - b.a, a.b - b.a)) <= 0;
}
```

#### 直线是否与圆相交及交点

$0$ 代表不相交；$1$ 代表相切；$2$ 代表相交。

```c++
// unit：向量单位化，要求非零；lineCircle：求直线与圆的交点
tuple<int, Pd, Pd> lineCircle(Ld l, Pd o, ld r) {
    Pd P = project(o, l);

    else if (sign(tmp) == 0) return {1, P, {}};
```

#### 线段是否与圆相交及交点

$0$ 代表不相交；$1$ 代表非退化线段与圆相切；$2$ 代表相交于一个点；$3$ 代表相交于两个点。这里求圆周交点；零长度线段恰在圆周上返回 2。依赖 lineCircle、onSeg。

```c++
// onSeg：判断点在线段上，包含端点；lineCircle：求直线与圆的交点；segCircle：求线段与圆的交点
tuple<int, Pd, Pd> segCircle(Ld l, Pd o, ld r) {
    if (l.a == l.b) { // 点在线圆周上时沿用状态 2：仅一个交点
        if (sign(dis(l.a, o) - r) == 0) return {2, l.a, {}};
        return {0, {}, {}};
    }
    auto [type, U, V] = lineCircle(l, o, r);
    bool f1 = onSeg(U, l), f2 = onSeg(V, l);
    if (type == 1 && f1) return {1, U, {}};
    else if (type == 2 && f1 && f2) return {3, U, V};
    else if (type == 2 && f1) return {2, U, {}};
    else if (type == 2 && f2) return {2, V, {}};
    else return {0, {}, {}};
}
```

#### 两圆是否相交及交点

$0$ 代表内含；$1$ 代表相离；$2$ 代表相切；$3$ 代表两个交点；$4$ 代表两圆重合、有无穷多个交点。要求正半径；仅状态 2、3 的返回点有效。浮点分类使用本章的绝对误差 EPS。

```c++
// circleInter：求两圆交点及位置状态
// 正半径；0 内含，1 相离，2 相切，3 两交点，4 重合（无穷交点）。
tuple<int, Pd, Pd> circleInter(Pd p1, ld r1, Pd p2, ld r2) {
    if (r1 <= 0 || r2 <= 0) throw invalid_argument("radii must be positive");
    Pd v = p2 - p1;
    ld d = dis(p1, p2);
    if (sign(d) == 0) return {sign(r1 - r2) == 0 ? 4 : 0, {}, {}};
    if (sign(d - abs(r1 - r2)) < 0) return {0, {}, {}};
    if (sign(d - r1 - r2) > 0) return {1, {}, {}};
    ld a = (r1 * r1 - r2 * r2 + d * d) / (2 * d);
    Pd u = v / d, base = p1 + u * a;
    if (sign(d - r1 - r2) == 0 || sign(d - abs(r1 - r2)) == 0) return {2, base, base};
    ld h = sqrt(max(ld(0), r1 * r1 - a * a));
    Pd off = Pd{-u.y, u.x} * h;
    return {3, base + off, base - off};
}
```

#### 两圆相交面积

上述所言四种相交情况均可计算，之所以不使用三角形面积计算公式是因为在计算过程中会出现“负数”面积（扇形面积与三角形面积的符号关系会随圆的位置关系发生变化），故公式全部重新推导，这里采用的是扇形面积减去扇形内部的那个三角形的面积。

```c++
ld circleArea(Pd a, ld r, Pd b, ld s) { // 两个圆盘，r,s>=0
    ld d = dis(a, b);
    if (d >= r + s) return 0;
    if (d <= abs(r - s)) return PI * min(r * r, s * s);
    ld x = angle(r, d, s), y = angle(s, d, r);
    return r * r * (x - sin(2 * x) / 2) + s * s * (y - sin(2 * y) / 2);
}
```

#### 求解两圆的内公、外公切线数量与切点

同时返回公切线数量以及每个圆的切点。

```c++
// 正半径；返回数量及两圆上的对应切点。-1 表示重合、无穷条。
// 第一个点数组始终属于输入圆 A，第二个属于 B。
tuple<int, vector<Pd>, vector<Pd>> tangent(Pd A, ld Ar, Pd B, ld Br) {
    if (Ar <= 0 || Br <= 0) throw invalid_argument("radii must be positive");
    vector<Pd> a, b;
    Pd v = B - A;
    ld d2 = dot(v, v), d = sqrt(d2);
    if (sign(d) == 0) return {sign(Ar - Br) == 0 ? -1 : 0, {}, {}};
    for (int side : {1, -1}) { // 外公切线、内公切线
        ld dr = Ar - side * Br;
        int pos = sign(d - abs(dr));
        if (pos < 0) continue;
        ld h = pos == 0 ? 0 : sqrt(max(ld(0), d2 - dr * dr));
        for (int sgn : {1, -1}) {
            Pd normal = (v * dr + Pd{-v.y, v.x} * (h * sgn)) / d2;
            a.push_back(A + normal * Ar);
            b.push_back(B + normal * (side * Br));
            if (pos == 0) break;
        }
    }
    return {(int)(a.size()), a, b};
}
```

#### 圆封装与最小覆盖圆

<a id="geo-mec"></a>

覆盖所有点的最小圆盘，随机增量**期望 $\mathcal{O}(n)$、最坏 $\mathcal{O}(n^{3})$**，空间 $\mathcal{O}(n)$（输入副本）。重复点、共线点可用；空集返回圆心 (0,0)、半径 0。依赖 Pd、dot/cross/dis、EPS；不依赖三角形外心模板。随机种子可传入以复现本地对拍。

`Circle` 供下方圆并面积复用；contains 含边界并容忍 EPS 的径向误差。三点定圆使用精确零判断，近共线的巨大外接圆仍受浮点误差影响；不要随意以 EPS 判共线后丢弃支撑点。

```c++
struct Circle {
    Pd o;
    ld r;
    Circle(Pd o_ = {}, ld r_ = 0) : o(o_), r(r_) {}
    bool contains(Pd p) const {
        return dis(o, p) <= r + EPS;
    }
};
Circle minCircle(vector<Pd> p, ull seed = chrono::steady_clock::now().time_since_epoch().count()) {
    mt19937_64 rng(seed);
    shuffle(p.begin(), p.end(), rng);
    Circle c;
    auto two = [](Pd a, Pd b) {
        return Circle((a + b) / 2, dis(a, b) / 2);
    };
    auto three = [&](Pd a, Pd b, Pd d) {
        Pd u = b - a, v = d - a;
        ld z = 2 * cross(u, v);
        if (z == 0) { // 精确共线：最远两点作直径
            Circle q = two(a, b), w = two(a, d), e = two(b, d);
            if (w.r > q.r) q = w;
            if (e.r > q.r) q = e;
            return q;
        }
        Pd o = a + Pd{v.y * dot(u, u) - u.y * dot(v, v), u.x * dot(v, v) - v.x * dot(u, u)} / z;
        return Circle(o, dis(a, o));
    };
    for (int i = 0; i < (int)p.size(); i++)
        if (!c.contains(p[i])) {
            c = Circle(p[i], 0);
            for (int j = 0; j < i; j++)
                if (!c.contains(p[j])) {
                    c = two(p[i], p[j]);
                    for (int k = 0; k < j; k++)
                        if (!c.contains(p[k])) c = three(p[i], p[j], p[k]);
                }
        }
    return c;
}
```

#### 圆与多边形的交面积

<a id="geo-circle-area"></a>

求圆盘和简单多边形的交集面积，允许凹多边形、任意顶点方向、重复相邻点；r≥0。时间 $\mathcal{O}(n)$、额外空间 $\mathcal{O}(1)$。依赖 Pd、dot/cross、sign、EPS；把每条边按圆周交点切分，圆内片段贡献三角形面积，圆外片段贡献扇形面积，最后取绝对值。输入首点不重复；圆内、圆外以及穿过圆心的边均可用。

```c++
ld circlePoly(Pd o, ld r, const vector<Pd> &p) {
    if (r == 0) return 0;
    ld ans = 0;
    for (int i = 0; i < (int)p.size(); i++) {
        Pd a = p[i] - o, b = p[(i + 1) % p.size()] - o, d = b - a;
        ld aa = dot(d, d);
        if (aa == 0) continue;
        vector<ld> ts{0, 1};
        ld t = -dot(a, d) / aa;
        Pd h = a + d * t;
        ld z = r * r - dot(h, h);
        if (z > 0) {
            ld dt = sqrtl(z / aa);
            if (t - dt > 0 && t - dt < 1) ts.push_back(t - dt);
            if (t + dt > 0 && t + dt < 1) ts.push_back(t + dt);
        }
        sort(ts.begin(), ts.end());
        for (int j = 1; j < (int)ts.size(); j++) {
            Pd u = a + d * ts[j - 1], v = a + d * ts[j], m = (u + v) / 2;
            if (dot(m, m) < r * r) ans += cross(u, v) / 2;
            else ans += r * r * atan2l(cross(u, v), dot(u, v)) / 2;
        }
    }
    return abs(ans);
}
```

#### 多圆并的面积与周长

返回圆盘并集的 `(面积,周长)`，时间 $\mathcal{O}(n^{2} \log n)$、额外空间 $\mathcal{O}(n)$，适合中等数量的圆。`circleUnion/circleUnionLen` 是仅取面积/周长的包装。依赖 Circle、Pd、dis、PI、EPS。半径≥0，自动略过零半径、包含圆和重复圆；重复圆保留输入编号较小者。对每个圆合并被其他圆覆盖的圆弧区间，裸露圆弧长度累加得周长，并用格林公式积分得面积；圆心先整体平移以减小消去误差。近重合、近相切及极大尺度仍需要结合题目调 EPS。

```c++
pair<ld, ld> circleUnionInfo(vector<Circle> c) {
    if (c.empty()) return {0, 0};
    Pd shift = c[0].o;
    for (auto &q : c) q.o -= shift;
    ld area = 0, len = 0;
    for (int i = 0; i < (int)c.size(); i++) {
        Pd o = c[i].o;
        ld r = c[i].r;
        if (r == 0) continue;
        bool covered = false;
        vector<pair<ld, ld>> seg;
        for (int j = 0; j < (int)c.size(); j++)
            if (i != j && c[j].r > 0) {
                ld d = dis(o, c[j].o), s = c[j].r;
                if (d <= EPS && abs(r - s) <= EPS)
                    if (j < i) {
                        covered = true;
                        break;
                    } else continue;
                if (d + r <= s + EPS) {
                    covered = true;
                    break;
                }
                if (d >= r + s - EPS || d + s <= r + EPS) continue;
                ld a = atan2l(c[j].o.y - o.y, c[j].o.x - o.x);
                if (a < 0) a += 2 * PI;
                ld b = acosl(max(-1.L, min(1.L, (r * r + d * d - s * s) / (2 * r * d))));
                ld l = a - b, h = a + b;
                if (l < 0) seg.push_back({l + 2 * PI, 2 * PI}), l = 0;
                if (h > 2 * PI) seg.push_back({0, h - 2 * PI}), h = 2 * PI;
                seg.push_back({l, h});
            }
        if (covered) continue;
        sort(seg.begin(), seg.end());
        auto add = [&](ld l, ld h) {
            area += (r * r * (h - l) + r * o.x * (sinl(h) - sinl(l)) + r * o.y * (cosl(l) - cosl(h))) / 2;
            len += r * (h - l);
        };
        ld end = 0;
        for (auto q : seg) {
            if (q.first > end) add(end, q.first);
            end = max(end, q.second);
        }
        add(end, 2 * PI);
    }
    return {max(0.L, area), max(0.L, len)};
}
ld circleUnion(vector<Circle> c) {
    return circleUnionInfo(c).first;
}
ld circleUnionLen(vector<Circle> c) {
    return circleUnionInfo(c).second;
}
```

#### 三角形费马点

<a id="geo-fermat"></a>

三点距离之和最小的点，$\mathcal{O}(1)$。某内角大于等于 120° 时答案为该顶点；否则三条连线两两成 120°。支持重复点和共线点。依赖 Pd、dis/dot/cross、rotate(向量,弧度)、lineInter。阈值附近受浮点误差影响。

```c++
Pd fermatPoint(Pd a, Pd b, Pd c) {
    if (samePt(a, b)) return a;
    if (samePt(b, c)) return b;
    if (samePt(c, a)) return c;
    ld ab = dis(a, b), bc = dis(b, c), ca = dis(c, a);
    if (dot(b - a, c - a) <= -ab * ca / 2) return a;
    if (dot(a - b, c - b) <= -ab * bc / 2) return b;
    if (dot(a - c, b - c) <= -ca * bc / 2) return c;
    if (cross(b - a, c - a) < 0) return lineInter({a, b + rotate(c - b, PI / 3)}, {b, c + rotate(a - c, PI / 3)});
    return lineInter({a, c + rotate(b - c, PI / 3)}, {c, b + rotate(a - b, PI / 3)});
}
```

#### 浮点数计算直线的斜率

#### 分数精确计算直线的斜率

#### 两点式转一般式

整数坐标返回约分后的 Ax+By=C；T 必须是有符号整数，差、乘积、绝对值及符号取反均须能存入 T。两点重合返回 (0,0,0)，它不代表唯一的直线。

```c++
template <class T> tuple<T, T, T> getfun(Line<T> l) { // 整数，Ax+By=C；所有中间值须能存入 T
    T a = l.a.y - l.b.y, b = l.b.x - l.a.x, c = a * l.a.x + b * l.a.y;
    if (a < 0 || (a == 0 && b < 0)) a = -a, b = -b, c = -c;
    auto gcdT = [](T x, T y) {
        if (x < 0) x = -x;
        if (y < 0) y = -y;
        while (y) {
            T z = x % y;
            x = y;
            y = z;
        }
        return x;
    };
    T g = gcdT(gcdT(a, b), c);
    if (g) a /= g, b /= g, c /= g;
    return {a, b, c};
}
```

#### 一般式转两点式

由于整数点可能很大或者不存在，故直接采用浮点数；如果与 $x,y$ 轴有交点则取交点。可以处理平行于 $x,y$ 轴的情况。

```c++
Ld getfun(ll a, ll b, ll c) { // Ax+By=C，要求 a,b 不全为 0
```

#### 抛物线与 x 轴是否相交及交点

解 $ax^2+bx+c=0$，$\mathcal{O}(1)$。状态 0 无实根、1 一个根、2 两个根、-1 恒等式。状态 1 的根在第二项，状态 2 按升序返回。先精确判断 a、b 是否为零；判别式按 EPS 判零。使用 q/a 与 c/q 减少相近数相减造成的消去误差；近平方根和极大系数仍需按题意控制尺度。

```c++
tuple<int, ld, ld> getAns(ld a, ld b, ld c) {
    if (a == 0) return b == 0 ? tuple<int, ld, ld>{c == 0 ? -1 : 0, 0, 0} : tuple<int, ld, ld>{1, -c / b, 0};
    ld d = b * b - 4 * a * c;
    if (sign(d) < 0) return {0, 0, 0};
    if (sign(d) == 0) return {1, -b / (2 * a), 0};
    ld q = -(b + copysignl(sqrtl(d), b)) / 2;
    ld x = q / a, y = c / q;
    return {2, min(x, y), max(x, y)};
}
```

#### 判断四个点能否组成矩形/正方形

可以处理浮点数、共点的情况。返回分为三种情况：$2$ 代表构成正方形；$1$ 代表构成矩形；$0$ 代表其他情况。

```c++
template <class T> int isSquare(vector<Point<T>> x) {
    sort(x.begin(), x.end());
    if (equal(dis(x[0], x[1]), dis(x[2], x[3])) && sign(dis(x[0], x[1])) && equal(dis(x[0], x[2]), dis(x[1], x[3])) &&

        linePar(Line<T>{x[0], x[2]}, Line<T>{x[1], x[3]}) && linePerp(Line<T>{x[0], x[1]}, Line<T>{x[0], x[2]}))
```

#### 点是否在任意多边形内

射线法判定，$t$ 为穿越次数，当其为奇数时即代表点在多边形内部；返回 $2$ 代表点在多边形边界上。

```c++
// onLeft：判断点在有向直线左侧；onSeg：判断点在线段上，包含端点；inPoly：点与一般多边形的位置关系
template <class T> int inPoly(Point<T> a, const vector<Point<T>> &p) {
    int n = p.size();
    for (int i = 0; i < n; i++)
        if (onSeg(a, Line<T>{p[i], p[(i + 1) % n]})) return 2;
    int t = 0;
    for (int i = 0; i < n; i++) {
        auto u = p[i], v = p[(i + 1) % n];
        if (u.x < a.x && v.x >= a.x && onLeft(a, Line<T>{v, u})) t ^= 1;
        if (u.x >= a.x && v.x < a.x && onLeft(a, Line<T>{u, v})) t ^= 1;
    }
    return t == 1;
}
```

#### 线段是否在任意多边形内部

输入不自交的简单多边形，允许凹、多边形边界和零长度查询线段；时间 $\mathcal{O}(n^{2})$，不适合直接套用在线性复杂度要求的题目中。沿用本章 EPS 和数值尺度限制。

```c++
// segInPoly：整条线段是否位于简单多边形内（含边界），不要求顶点方向。
// 先按所有边界交点切分，再检测每段中点；时间 O(n²)，额外空间 O(n)。
template <class T> bool segInPoly(Line<T> l, const vector<Point<T>> &p) {
    if (p.size() < 3) return false;
    vector<Point<ld>> q; // 转浮点后求交，避免整数交点截断
    for (auto v : p) q.push_back({ld(v.x), ld(v.y)});
    Point<ld> a{ld(l.a.x), ld(l.a.y)}, b{ld(l.b.x), ld(l.b.y)}, d = b - a;
    if (!inPoly(a, q) || !inPoly(b, q)) return false;
    if (samePt(a, b)) return true;
    vector<ld> cut{0, 1}; // 线段参数 a+t*d 的所有分界点
    auto add = [&](ld t) {
        if (t >= -EPS && t <= 1 + EPS) cut.push_back(max(0.L, min(1.L, t)));
    };
    for (int i = 0; i < (int)(q.size()); ++i) {
        auto u = q[i], v = q[(i + 1) % q.size()], e = v - u;
        ld den = cross(d, e);
        if (sign(den) == 0) {
            if (sign(cross(u - a, d)) == 0) {
                add(dot(u - a, d) / dot(d, d));
                add(dot(v - a, d) / dot(d, d));
            }
        } else {
            ld t = cross(u - a, e) / den, z = cross(u - a, d) / den;
            if (z >= -EPS && z <= 1 + EPS) add(t);
        }
    }
    sort(cut.begin(), cut.end());
    for (int i = 1; i < (int)(cut.size()); ++i)
        if (cut[i] > cut[i - 1] && !inPoly(a + d * ((cut[i] + cut[i - 1]) / 2), q)) return false;
    return true;
}
```

#### 任意多边形的面积

鞋带公式，$\mathcal{O}(n)$ 时间、$\mathcal{O}(1)$ 额外空间。返回**有向面积**：逆时针为正，顺时针为负；实际面积取 abs。简单多边形不重复末点；空集、点、线段返回 0。以首点为原点作三角剖分，先转 ld 再相减，减少大平移造成的误差；需要格点精确倍面积时用下节 area2。

```c++
template <class T> ld area(const vector<Point<T>> &p) {
    ld s = 0;
    for (int i = 1; i + 1 < (int)p.size(); i++) s += cross(Pd(p[i]) - Pd(p[0]), Pd(p[i + 1]) - Pd(p[0]));
    return s / 2;
}
```

#### 皮克定理

绘制在方格纸上的多边形面积公式可以表示为 $S=n+\dfrac{s}{2}-1$ ，其中 $n$ 表示多边形内部的点数、$s$ 表示多边形边界上的点数。一条线段上的点数为 $\gcd(|x_1-x_2|,|y_1-y_2|)+1$。

#### 任意多边形上/内的网格点个数（仅能处理整数）

输入非退化、无孔的简单格点多边形，首点不重复。返回 i128，坐标输入可用 Point<int>/Point<ll>；面积累计、坐标差及 gcd 都在 i128 中计算。差、叉积和累计仍不能超过 i128；不是任意 64 位极值输入都安全。area2 是有向倍面积；borderGrid 为边界格点数，innerGrid 为内部格点数。面积 $\mathcal{O}(n)$，边界计数 $\mathcal{O}(n \log C)$，额外空间 $\mathcal{O}(1)$，C 为最大坐标差。输出 i128 时使用第 13 章对应输出工具或确认能存入 ll 后转换。

```c++
template <class T> i128 area2(const vector<Point<T>> &p) {
    i128 s = 0;
    for (int i = 1; i + 1 < (int)p.size(); i++) {
        i128 x = i128(p[i].x) - p[0].x, y = i128(p[i].y) - p[0].y;
        i128 u = i128(p[i + 1].x) - p[0].x, v = i128(p[i + 1].y) - p[0].y;
        s += x * v - y * u;
    }
    return s;
}
template <class T> i128 borderGrid(const vector<Point<T>> &p) {
    i128 s = 0;
    for (int i = 0; i < (int)p.size(); i++) {
        auto a = p[i], b = p[(i + 1) % p.size()];
        i128 x = i128(a.x) - b.x, y = i128(a.y) - b.y;
        if (x < 0) x = -x;
        if (y < 0) y = -y;
        while (y) {
            i128 z = x % y;
            x = y;
            y = z;
        }
        s += x;
    }
    return s;
}
template <class T> i128 innerGrid(const vector<Point<T>> &p) {
    i128 s = area2(p);
    if (s < 0) s = -s;
    return (s - borderGrid(p)) / 2 + 1;
}
```

#### 多边形重心与半平面裁剪

<a id="geo-polygon-extra"></a>

`centroid` 求均匀面密度的简单多边形重心，$\mathcal{O}(n)$ 时间、$\mathcal{O}(1)$ 空间；要求面积非零，不要求凸或逆时针。依赖 Pd/cross；平移到首点计算，减小大坐标下的误差。

`cutPoly` 将**凸**多边形裁到有向直线左侧（含边界），时间与输出空间 $\mathcal{O}(n)$，支持裁为空集、点或线段。依赖 Pd/Ld、cross、samePt；边界使用精确符号切分，避免 EPS 内外分类和交点公式不一致。近边界浮点误差仍可能影响退化结果。输入为凸包有序顶点，裁剪线非退化。

```c++
Pd centroid(const vector<Pd> &p) {
    /* assert(p.size() >= 3) */;
    Pd q{0, 0}, o = p[0];
    ld s = 0;
    for (int i = 1; i + 1 < (int)p.size(); i++) {
        Pd a = p[i] - o, b = p[i + 1] - o;
        ld z = cross(a, b);
        s += z;
        q += (a + b) * z;
    }
    /* assert(s != 0) */;
    return o + q / (3 * s);
}
vector<Pd> cutPoly(const vector<Pd> &p, Ld l) {
    vector<Pd> q;
    auto add = [&](Pd a) {
        if (q.empty() || !samePt(q.back(), a)) q.push_back(a);
    };
    for (int i = 0; i < (int)p.size(); i++) {
        Pd a = p[i], b = p[(i + 1) % p.size()];
        ld x = cross(l.b - l.a, a - l.a), y = cross(l.b - l.a, b - l.a);
        if (x >= 0) add(a);
        if ((x < 0 && y > 0) || (x > 0 && y < 0)) add(a + (b - a) * (x / (x - y)));
    }
    if (q.size() > 1 && samePt(q.front(), q.back())) q.pop_back();
    return q;
}
```

#### 两凸多边形的交

反复调用 cutPoly，时间 $\mathcal{O}(m(n+m))$ 的一般上界；对于两凸多边形，第 k 次结果的边来自前 k 条裁剪线和原 n 条边。常见写法中若两者边数同阶即 $\mathcal{O}(n^{2})$，不要当作线性算法。空间 $\mathcal{O}(n+m)$。输入 a、b 都逆时针、至少三个点且面积非零，不重复末点；输出可能为空、点、线段。依赖 cutPoly；大规模可把两者全部有向边送入 halfcut，$\mathcal{O}((n+m) \log (n+m))$，但 halfcut 不保留零面积交集。

```c++
vector<Pd> convexInter(vector<Pd> a, const vector<Pd> &b) {
    for (int i = 0; i < (int)b.size() && !a.empty(); i++) a = cutPoly(a, {b[i], b[(i + 1) % b.size()]});
    return a;
}
```

### 二维凸包

#### 获取二维静态凸包（Andrew算法）

`flag=0` 保留共线边界点，`flag=1` 只保留极点。精确重复点始终去重；全共线时分别返回排序后的全部不同点或两个端点；非退化凸包逆时针返回且不重复起点。时间 $\mathcal O(N\log N)$。整数叉乘须不溢出；浮点版使用精确叉乘比较，近共线或尺度悬殊的数据需要按题意调整误差策略。

```c++
// convexHull：Andrew 静态凸包
template <class T> vector<Point<T>> convexHull(vector<Point<T>> A, int flag = 1) {
    sort(A.begin(), A.end());
    A.erase(unique(A.begin(), A.end()), A.end()); // 精确重复点去重
    int n = A.size();
    if (n <= 2) return A;
    bool collinear = true;
    for (int i = 1; i + 1 < n; i++)
        if (cross(A.back() - A[0], A[i] - A[0]) != 0) collinear = false;
    if (collinear) return flag ? vector<Point<T>>{A.front(), A.back()} : A;
    vector<Point<T>> lower, upper;
    auto append = [&](vector<Point<T>> &h, Point<T> p) {
        while (h.size() >= 2) {
            auto z = cross(h.back() - h[h.size() - 2], p - h.back());
            if (flag ? z > 0 : z >= 0) break;
            h.pop_back();
        }
        h.push_back(p);
    };
    for (auto p : A) append(lower, p);
    for (auto it = A.rbegin(); it != A.rend(); ++it) append(upper, *it);
    lower.pop_back();
    upper.pop_back();
    lower.insert(lower.end(), upper.begin(), upper.end());
    return lower; // 从字典序最小点起，逆时针，不重复起点
}
```

#### 二维动态凸包

依赖 Point/cross/dot。专用边类型改名 HullEdge，可与本章 Line<T> 同时复制。内部坐标固定为 ll，建议 $\lvert\text{坐标}\rvert\le10^9$，所有差、点积、叉积及取负须能存入 ll；只支持整数点。插入均摊 $\mathcal{O}(\log n)$，点查询 $\mathcal{O}(\log n)$，isIntersect 直线查询 $\mathcal{O}(h)$，h 为凸包顶点数。set 始终按点坐标排序，不在查询时改变比较规则。支持：

- 动态插入点 $(x,y)$ 到当前凸包中；
- 判断点 $(x,y)$ 是否在凸包上或是在内部（包括边界）。

```c++
template <class T> bool turnRight(Point<T> a, Point<T> b) {
    return cross(a, b) < 0 || (cross(a, b) == 0 && dot(a, b) < 0);
}
struct HullEdge {
    mutable Point<ll> a, b;
    friend bool operator<(HullEdge x, HullEdge y) {
        return x.a < y.a; // 始终保持同一种严格序，方向字段 b 不参与排序
    }
    friend auto &operator<<(ostream &os, HullEdge l) {
        return os << "<" << l.a << ", " << l.b << ">";
    }
};

struct UpperConvexHull : set<HullEdge> {
    bool contains(const Point<ll> &p) const {
        auto it = lower_bound({p, 0});
        if (it != end() && it->a == p) return true;
        if (it != begin() && it != end() && cross(prev(it)->b, p - prev(it)->a) <= 0) return true;
        return false;
    }
    void add(const Point<ll> &p) {
        if (contains(p)) return;
        auto it = lower_bound({p, 0});
        for (; it != end(); it = erase(it))
            if (turnRight(it->a - p, it->b)) break;
        for (; it != begin() && prev(it) != begin(); erase(prev(it)))
            if (turnRight(prev(prev(it))->b, p - prev(prev(it))->a)) break;
        if (it != begin()) prev(it)->b = p - prev(it)->a;
        if (it == end()) insert({p, {0, -1}});
        else insert({p, it->a - p});
    }
};
struct ConvexHull {
    UpperConvexHull up, low;
    bool empty() const {
        return up.empty();
    }
    bool contains(const Point<ll> &p) const {
        return up.contains(p) && low.contains(-p);
    }
    void add(const Point<ll> &p) {
        up.add(p);
        low.add(-p);
    }
    bool isIntersect(ll A, ll B, ll C) const { // 直线与凸包（含边界）相交；O(h)
        /* assert(A != 0 || B != 0) */;
        if (empty()) return false;
        i128 lo = (i128(1) << 120), hi = -lo;
        auto check = [&](Point<ll> p) {
            i128 v = (i128)A * p.x + (i128)B * p.y - C;
            lo = min(lo, v);
            hi = max(hi, v);
        };
        for (auto l : up) check(l.a);
        for (auto l : low) check(-l.a);
        return lo <= 0 && hi >= 0;
    }
    friend ostream &operator<<(ostream &out, const ConvexHull &ch) {
        for (const auto &line : ch.up) out << "(" << line.a.x << "," << line.a.y << ")";
        out << "/";
        for (const auto &line : ch.low) out << "(" << -line.a.x << "," << -line.a.y << ")";
        return out;
    }
};
```

#### 点与凸包的位置关系

$0$ 代表点在凸包外面；$1$ 代表在凸壳上；$2$ 代表在凸包内部。

```c++
template <class T> int contains(Point<T> p, const vector<Point<T>> &A) {
    int n = A.size();
    bool in = false;
    for (int i = 0; i < n; i++) {
        Point<T> a = A[i] - p, b = A[(i + 1) % n] - p;
        if (a.y > b.y) swap(a, b);
        if (a.y <= 0 && 0 < b.y && cross(a, b) < 0) in = !in;
        if (cross(a, b) == 0 && dot(a, b) <= 0) return 1;
    }
    return in ? 2 : 0;
}
```

#### 凸多边形内点查询（二分）

<a id="geo-convex-query"></a>

单次 $\mathcal{O}(\log n)$ 时间、$\mathcal{O}(1)$ 额外空间，适合大量点查询。依赖 Point/Line、cross、sign、samePt、onSeg。输入用 convexHull(p,1) 获得：逆时针、无重复末点、无共线中间点。支持空集、点、线段。返回 0 外、1 内、2 边界，**与 inPoly 相同，但与旧 contains 的 1/2 相反**。

```c++
template <class T> int inConvex(Point<T> p, const vector<Point<T>> &a) {
    int n = a.size();
    if (!n) return 0;
    if (n == 1) return samePt(p, a[0]) ? 2 : 0;
    if (n == 2) return onSeg(p, Line<T>{a[0], a[1]}) ? 2 : 0;
    int x = sign(cross(a[1] - a[0], p - a[0])), y = sign(cross(a[n - 1] - a[0], p - a[0]));
    if (x < 0 || y > 0) return 0;
    if (!x) return onSeg(p, Line<T>{a[0], a[1]}) ? 2 : 0;
    if (!y) return onSeg(p, Line<T>{a[0], a[n - 1]}) ? 2 : 0;
    int l = 1, r = n - 1;
    while (r - l > 1) {
        int m = (l + r) / 2;
        if (sign(cross(a[m] - a[0], p - a[0])) >= 0) l = m;
        else r = m;
    }
    int z = sign(cross(a[r] - a[l], p - a[l]));
    return z < 0 ? 0 : z == 0 ? 2 : 1;
}
```

#### 凸多边形方向极值查询

<a id="geo-support"></a>

给定非零方向 `v`，返回使 `dot(p[i],v)` 最大的一个顶点下标；构造 $\mathcal{O}(n \log n)$，单次查询 $\mathcal{O}(\log n)$，空间 $\mathcal{O}(n)$。适合反复求支撑点、投影范围和凸多边形与直线的关系。输入为 `convexHull(p,1)` 的结果，允许点和线段；空集不可查询。若最优是一条边，返回其任一端点。

`ConvexSupport` 保存输入顶点及其原下标，内部按边的外法向极角排序。比较器不使用 EPS；整数坐标须保证边差、取负、叉积和点积不溢出，大坐标可用 `Point<i128>`。

```c++
template <class T> struct ConvexSupport {
    vector<Point<T>> p, norm;
    vector<int> id;
    static bool angleLess(Point<T> a, Point<T> b) {
        auto half = [](Point<T> q) {
            return q.y < 0 || (q.y == 0 && q.x < 0);
        };
        if (half(a) != half(b)) return half(a) < half(b);
        return cross(a, b) > 0;
    }
    ConvexSupport(const vector<Point<T>> &a) : p(a) {
        if (p.size() < 3) return;
        vector<pair<Point<T>, int>> e;
        for (int i = 0; i < (int)p.size(); i++) {
            Point<T> d = p[(i + 1) % p.size()] - p[i];
            e.push_back({{d.y, -d.x}, i}); // 逆时针凸包边的外法向
        }
        sort(e.begin(), e.end(), [](auto a, auto b) { return angleLess(a.first, b.first); });
        for (auto q : e) norm.push_back(q.first), id.push_back(q.second);
    }
    int argmax(Point<T> v) const {
        /* assert(!p.empty() && (v.x != 0 || v.y != 0)) */;
        if (p.size() == 1) return 0;
        if (p.size() == 2) return dot(p[0], v) >= dot(p[1], v) ? 0 : 1;
        auto it = lower_bound(norm.begin(), norm.end(), v, [](Point<T> a, Point<T> b) { return angleLess(a, b); });
        return id[it == norm.end() ? 0 : it - norm.begin()];
    }
    T maxDot(Point<T> v) const {
        return dot(p[argmax(v)], v);
    }
};
```

#### 旋转卡壳：凸包直径

<a id="geo-calipers"></a>

返回最远点对的**距离平方**，$\mathcal{O}(n)$ 时间、$\mathcal{O}(1)$ 额外空间；空集或单点返回 0。输入 convexHull(p,1) 的结果。依赖 Point、cross/dot/dis2；T 必须容纳坐标差、叉积及距离平方，整数大坐标使用 Point<i128>。每条边枚举对踵点，平行对边的两个端点都检查。

```c++
template <class T> T diameter2(const vector<Point<T>> &p) {
    int n = p.size();
    if (n < 2) return 0;
    if (n == 2) return dis2(p[0], p[1]);
    T ans = 0;
    int j = 1;
    for (int i = 0; i < n; i++) {
        int k = (i + 1) % n;
        auto h = [&](int z) {
            return cross(p[k] - p[i], p[z] - p[i]);
        };
        while (h((j + 1) % n) > h(j)) j = (j + 1) % n;
        ans = max(ans, max(dis2(p[i], p[j]), dis2(p[k], p[j])));
        if (h((j + 1) % n) == h(j)) ans = max(ans, max(dis2(p[i], p[(j + 1) % n]), dis2(p[k], p[(j + 1) % n])));
    }
    return ans;
}
```

#### 旋转卡壳：凸包最大面积三角形

对严格凸包的所有顶点三角形，返回最大**两倍面积**，$\mathcal{O}(n^{2})$ 时间、$\mathcal{O}(1)$ 额外空间；不足三个点返回 0。输入为 `convexHull(p,1)` 的结果。固定前两个顶点后，第三个顶点的面积先增后减，指针可单调移动。T 须容纳坐标差和叉积；浮点结果可能有误差。

```c++
template <class T> T maxTriangle2(const vector<Point<T>> &p) {
    int n = p.size();
    T ans = 0;
    for (int i = 0; i + 2 < n; i++) {
        int k = i + 2;
        for (int j = i + 1; j + 1 < n; j++) {
            if (k <= j) k = j + 1;
            auto s = [&](int z) {
                return cross(p[j] - p[i], p[z] - p[i]);
            };
            while (k + 1 < n && s(k + 1) >= s(k)) k++;
            ans = max(ans, s(k));
        }
    }
    return ans;
}
```

#### 旋转卡壳：最小宽度与最小面积外接矩形

分别返回最窄平行支撑线间距，以及 `(最小面积, 四个逆时针角点)`，**各自 $\mathcal{O}(n)$** 时间、$\mathcal{O}(1)$ 额外空间。依赖 Pd、cross/dot/dis。输入严格逆时针凸包；最优矩形至少一条边平行于凸包某边。点/线段返回宽度和面积 0；矩形角点允许重复，空集角点为空。

`j` 维护法向最远点，`l/r` 维护边方向投影最小/最大点。初始化扫描一次，其后沿凸包单调移动。比较使用精确大小，不能让 EPS 人为阻止指针前进；近共线图形需注意精度。

```c++
ld minWidth(const vector<Pd> &p) {
    int n = p.size();
    if (n < 3) return 0;
    ld ans = numeric_limits<ld>::infinity();
    int j = 1;
    for (int i = 0; i < n; i++) {
        Pd v = p[(i + 1) % n] - p[i];
        auto h = [&](int z) {
            return cross(v, p[z] - p[i]);
        };
        while (h((j + 1) % n) > h(j)) j = (j + 1) % n;
        ans = min(ans, h(j) / dis(p[i], p[(i + 1) % n]));
    }
    return ans;
}
pair<ld, vector<Pd>> minRect(const vector<Pd> &p) {
    int n = p.size();
    if (!n) return {0, {}};
    if (n == 1) return {0, vector<Pd>(4, p[0])};
    if (n == 2) return {0, {p[0], p[1], p[1], p[0]}};
    int j = 0, l = 0, r = 0;
    Pd u = p[1] - p[0];
    for (int k = 1; k < n; k++) {
        if (cross(u, p[k] - p[0]) > cross(u, p[j] - p[0])) j = k;
        if (dot(u, p[k] - p[0]) < dot(u, p[l] - p[0])) l = k;
        if (dot(u, p[k] - p[0]) > dot(u, p[r] - p[0])) r = k;
    }
    ld best = numeric_limits<ld>::infinity();
    vector<Pd> box;
    for (int i = 0; i < n; i++) {
        Pd a = p[i], v = p[(i + 1) % n] - a;
        auto x = [&](int k) {
            return dot(v, p[k] - a);
        };
        auto y = [&](int k) {
            return cross(v, p[k] - a);
        };
        while (y((j + 1) % n) > y(j)) j = (j + 1) % n;
        while (x((l + 1) % n) < x(l)) l = (l + 1) % n;
        while (x((r + 1) % n) > x(r)) r = (r + 1) % n;
        ld d = dot(v, v), s = (x(r) - x(l)) * y(j) / d;
        if (s < best) {
            best = s;
            Pd h = Pd{-v.y, v.x} * (y(j) / d), b = a + v * (x(l) / d), c = a + v * (x(r) / d);
            box = {b, c, c + h, b + h};
        }
    }
    return {best, box};
}
```

#### 平面最近点对（分治）

<a id="geo-closest"></a>

求最小欧氏距离及一组点对，时间 $\mathcal{O}(n \log n)$、空间 $\mathcal{O}(n)$。依赖 Pd、dot；允许重复点，重复点距离为 0。少于两个点返回 `(infinity, {}, {})`。函数复制输入，递归将每段归并为 y 有序，跨中线只比较纵坐标差小于当前答案的点。最后一次开方；坐标及距离平方须为有限可表示值。

```c++
tuple<ld, Pd, Pd> closestPair(vector<Pd> p) {
    int n = p.size();
    ld best = numeric_limits<ld>::infinity();
    Pd a, b;
    if (n < 2) return {best, a, b};
    sort(p.begin(), p.end());
    for (int i = 1; i < n; i++)
        if (p[i] == p[i - 1]) return {0, p[i], p[i]};
    vector<Pd> tmp(n);
    auto upd = [&](Pd u, Pd v) {
        ld d = dot(u - v, u - v);
        if (d < best) best = d, a = u, b = v;
    };
    auto byY = [](Pd u, Pd v) {
        return u.y != v.y ? u.y < v.y : u.x < v.x;
    };
    function<void(int, int)> go = [&](int l, int r) {
        if (r - l <= 3) {
            for (int i = l; i < r; i++)
                for (int j = l; j < i; j++) upd(p[i], p[j]);
            sort(p.begin() + l, p.begin() + r, byY);
            return;
        }
        int m = (l + r) / 2;
        ld x = p[m].x;
        go(l, m);
        go(m, r);
        merge(p.begin() + l, p.begin() + m, p.begin() + m, p.begin() + r, tmp.begin() + l, byY);
        copy(tmp.begin() + l, tmp.begin() + r, p.begin() + l);
        int k = 0;
        for (int i = l; i < r; i++)
            if ((p[i].x - x) * (p[i].x - x) < best) {
                for (int j = k - 1; j >= 0 && (p[i].y - tmp[j].y) * (p[i].y - tmp[j].y) < best; j--) upd(p[i], tmp[j]);
                tmp[k++] = p[i];
            }
    };
    go(0, n);
    return {sqrtl(best), a, b};
}
```

#### 闵可夫斯基和

求两个凸集的逐点和，$\mathcal{O}(n+m)$ 时间和空间。输入均为 convexHull(p,1) 的输出：逆时针、无重复点、无共线中间点；支持空集、单点、线段。内部按边方向归并，输出逆时针、从字典序最小点开始，**不再重复末点**，可直接用于 inConvex/diameter2。整数运算按 T 进行，预留坐标相加及叉积位宽。原函数名 mincowski 保留。

```c++
template <class T> vector<Point<T>> mincowski(vector<Point<T>> a, vector<Point<T>> b) {
    if (a.empty() || b.empty()) return {};
    auto norm = [](vector<Point<T>> &p) {
        auto it =
            min_element(p.begin(), p.end(), [](Point<T> u, Point<T> v) { return u.y != v.y ? u.y < v.y : u.x < v.x; });
        std::rotate(p.begin(), it, p.end());
    };
    norm(a);
    norm(b);
    if (a.size() == 1 || b.size() == 1) {
        if (a.size() != 1) swap(a, b);
        for (auto &p : b) p += a[0];
        std::rotate(b.begin(), min_element(b.begin(), b.end()), b.end());
        return b;
    }
    int n = a.size(), m = b.size(), i = 0, j = 0;
    auto half = [](Point<T> p) {
        return p.y < 0 || (p.y == 0 && p.x < 0);
    };
    vector<Point<T>> q{a[0] + b[0]};
    while (i < n || j < m) {
        Point<T> u = a[(i + 1) % n] - a[i % n], v = b[(j + 1) % m] - b[j % m], w;
        if (i == n) w = v, j++;
        else if (j == m) w = u, i++;
        else if (half(u) == half(v) && cross(u, v) == 0) w = u + v, i++, j++;
        else if (half(u) != half(v) ? half(u) < half(v) : cross(u, v) > 0) w = u, i++;
        else w = v, j++;
        q.push_back(q.back() + w);
    }
    q.pop_back();
    std::rotate(q.begin(), min_element(q.begin(), q.end()), q.end());
    return q;
}
```

#### 两凸集最短距离

<a id="geo-convex-distance"></a>

返回两个非空凸集的欧氏距离，$\mathcal{O}(n+m)$ 时间和空间；相交、包含或相切返回 0。输入均为 `convexHull(p,1)` 的浮点结果，支持点、线段和多边形，不重复末点。利用 `A-B` 到原点的距离，依赖 mincowski、inConvex、projSeg；EPS 只用于最终包含/边界判断，近相切数据仍需按题目尺度调整。

```c++
ld convexDist(vector<Pd> a, vector<Pd> b) {
    /* assert(!a.empty() && !b.empty()) */;
    for (auto &p : b) p = -p;
    // 中心对称不改变顶点的逆时针顺序
    auto c = mincowski(a, b);
    if (inConvex(Pd{}, c)) return 0;
    if (c.size() == 1) return dis(c[0], Pd{});
    ld ans = numeric_limits<ld>::infinity();
    for (int i = 0; i < (int)c.size(); i++) ans = min(ans, projSeg(Pd{}, Ld{c[i], c[(i + 1) % c.size()]}).second);
    return ans;
}
```

#### 半平面交

计算有向直线左侧半平面的交，$\mathcal{O}(n \log n)$ 时间、$\mathcal{O}(n)$ 空间。输入为浮点非退化直线，须保证交集有界（必要时按题意添加足够大的包围盒）；仅输出有正面积的多边形，空集、线段、单点及无界结果不作区分。排序使用精确方向序，近乎平行的输入仍受浮点误差影响。

```c++
// lineInter：求直线交点
template <class T> vector<Point<T>> halfcut(vector<Line<T>> lines) {
    auto half = [](Point<T> d) {
        return d.y > 0 || (d.y == 0 && d.x >= 0);
    };
    sort(lines.begin(), lines.end(), [&](Line<T> a, Line<T> b) {
        auto u = a.b - a.a, v = b.b - b.a;
        if (half(u) != half(v)) return half(u) > half(v);
        return cross(u, v) > 0; // 精确方向序，不能在排序比较器里引入 EPS
    });
    vector<Line<T>> a;
    for (auto l : lines) { // 同方向只保留最严格的一条，避免重复交点和无效断言
        if (!a.empty() && cross(l.b - l.a, a.back().b - a.back().a) == 0 &&
            dot(l.b - l.a, a.back().b - a.back().a) > 0) {
            if (cross(a.back().b - a.back().a, l.a - a.back().a) > 0) a.back() = l;
        } else a.push_back(l);
    }
    auto out = [](Line<T> l, Point<T> p) {
        return cross(l.b - l.a, p - l.a) < -EPS;
    };
    deque<Line<T>> q;
    for (auto l : a) {
        while (q.size() > 1 && out(l, lineInter(q[q.size() - 2], q.back()))) q.pop_back();
        while (q.size() > 1 && out(l, lineInter(q[0], q[1]))) q.pop_front();
        if (!q.empty() && cross(q.back().b - q.back().a, l.b - l.a) == 0) return {}; // 不能围成正面积有界交
        q.push_back(l);
    }
    while (q.size() > 2 && out(q.front(), lineInter(q[q.size() - 2], q.back()))) q.pop_back();
    while (q.size() > 2 && out(q.back(), lineInter(q[0], q[1]))) q.pop_front();
    if (q.size() < 3 || cross(q.back().b - q.back().a, q.front().b - q.front().a) == 0) return {};
    vector<Point<T>> ans;
    for (int i = 0; i < (int)(q.size()); ++i) ans.push_back(lineInter(q[i], q[(i + 1) % q.size()]));
    T area2 = 0;
    for (int i = 0; i < (int)(ans.size()); ++i) area2 += cross(ans[i], ans[(i + 1) % ans.size()]);
    if (abs(area2) < EPS) return {};
    return ans;
}
```

### Delaunay 三角剖分（整数点）

<a id="geo-delaunay"></a>

`delaunay` 返回去重后的字典序点集 `p` 及逆时针三角形下标 `f`；全共线时 `f` 为空。使用 Guibas-Stolfi 分治与 Quad-Edge，时间 $\mathcal{O}(n \log n)$、空间 $\mathcal{O}(n)$。共圆时保留任一合法对角线；重复点只保留一个，因此返回下标不是原输入下标。

`inCircleDet(a,b,c,d)` 是精确圆内谓词：`abc` 逆时针时，正/零/负分别表示 d 在外接圆内/上/外。所有计算在相减前扩宽到 i128。为保证四次行列式不溢出，要求 `max(maxX-minX,maxY-minY)<=1e9`；坐标本身可带大平移。

```c++
template <class T> i128 inCircleDet(Point<T> a, Point<T> b, Point<T> c, Point<T> d) {
    i128 ax = i128(a.x) - d.x, ay = i128(a.y) - d.y;
    i128 bx = i128(b.x) - d.x, by = i128(b.y) - d.y;
    i128 cx = i128(c.x) - d.x, cy = i128(c.y) - d.y;
    return (ax * ax + ay * ay) * (bx * cy - by * cx) + (bx * bx + by * by) * (cx * ay - cy * ax) +
           (cx * cx + cy * cy) * (ax * by - ay * bx);
}
struct Delaunay {
    vector<Point<ll>> p;
    vector<array<int, 3>> f;
};
Delaunay delaunay(vector<Point<ll>> a) {
    sort(a.begin(), a.end());
    a.erase(unique(a.begin(), a.end()), a.end());
    Delaunay ans;
    ans.p = a;
    int n = a.size();
    if (n < 3) return ans;
    i128 xl = a[0].x, xr = xl, yl = a[0].y, yr = yl;
    for (auto p : a) xl = min(xl, i128(p.x)), xr = max(xr, i128(p.x)), yl = min(yl, i128(p.y)), yr = max(yr, i128(p.y));
    /* assert(max(xr-xl,yr-yl)<=1000000000) */;
    struct E {
        int o = -1;
        E *rot = nullptr, *next = nullptr;
        bool dead = false;
        E *rev() {
            return rot->rot;
        }
        int dest() {
            return rev()->o;
        }
        E *oPrev() {
            return rot->next->rot;
        }
        E *lNext() {
            return rot->rev()->next->rot;
        }
    };
    deque<array<E, 4>> pool;
    vector<E *> all;
    auto makeEdge = [&](int u, int v) {
        pool.push_back({});
        auto &q = pool.back();
        for (int i = 0; i < 4; i++) q[i].rot = &q[(i + 1) % 4];
        q[0].next = &q[0];
        q[1].next = &q[3];
        q[2].next = &q[2];
        q[3].next = &q[1];
        q[0].o = u;
        q[2].o = v;
        all.push_back(&q[0]);
        return &q[0];
    };
    auto splice = [](E *x, E *y) {
        swap(x->next->rot->next, y->next->rot->next);
        swap(x->next, y->next);
    };
    auto del = [&](E *e) {
        splice(e, e->oPrev());
        splice(e->rev(), e->rev()->oPrev());
        E *q = e;
        for (int i = 0; i < 4; i++, q = q->rot) q->dead = true;
    };
    auto connect = [&](E *x, E *y) {
        E *e = makeEdge(x->dest(), y->o);
        splice(e, x->lNext());
        splice(e->rev(), y);
        return e;
    };
    auto side = [&](int x, E *e) {
        return orient(a[e->o], a[e->dest()], a[x]);
    };
    auto inside = [&](int x, int y, int z, int w) {
        i128 d = inCircleDet(a[x], a[y], a[z], a[w]), s = orient(a[x], a[y], a[z]);
        return s > 0 ? d > 0 : s < 0 ? d < 0 : false;
    };
    function<pair<E *, E *>(int, int)> build = [&](int l, int r) -> pair<E *, E *> {
        if (r - l == 2) {
            E *e = makeEdge(l, l + 1);
            return {e, e->rev()};
        }
        if (r - l == 3) {
            E *x = makeEdge(l, l + 1), *y = makeEdge(l + 1, l + 2);
            splice(x->rev(), y);
            i128 s = orient(a[l], a[l + 1], a[l + 2]);
            if (s > 0) {
                connect(y, x);
                return {x, y->rev()};
            }
            if (s < 0) {
                E *z = connect(y, x);
                return {z->rev(), z};
            }
            return {x, y->rev()};
        }
        int m = (l + r) / 2;
        auto [ldo, ldi] = build(l, m);
        auto [rdi, rdo] = build(m, r);
        while (true)
            if (side(rdi->o, ldi) > 0) ldi = ldi->lNext();
            else if (side(ldi->o, rdi) < 0) rdi = rdi->rev()->next;
            else break;
        E *base = connect(rdi->rev(), ldi);
        if (ldi->o == ldo->o) ldo = base->rev();
        if (rdi->o == rdo->o) rdo = base;
        while (true) {
            E *lc = base->rev()->next;
            if (side(lc->dest(), base) < 0)
                while (inside(base->dest(), base->o, lc->dest(), lc->next->dest())) {
                    E *t = lc->next;
                    del(lc);
                    lc = t;
                }
            E *rc = base->oPrev();
            if (side(rc->dest(), base) < 0)
                while (inside(base->dest(), base->o, rc->dest(), rc->oPrev()->dest())) {
                    E *t = rc->oPrev();
                    del(rc);
                    rc = t;
                }
            bool lv = side(lc->dest(), base) < 0, rv = side(rc->dest(), base) < 0;
            if (!lv && !rv) break;
            if (!lv || (rv && inside(lc->dest(), lc->o, rc->o, rc->dest()))) base = connect(rc, base->rev());
            else base = connect(base->rev(), lc->rev());
        }
        return {ldo, rdo};
    };
    build(0, n);
    vector<vector<int>> g(n);
    set<pair<int, int>> es;
    for (E *e : all)
        if (!e->dead) {
            int u = e->o, v = e->dest();
            if (u > v) swap(u, v);
            es.insert({u, v});
        }
    for (auto [u, v] : es) g[u].push_back(v), g[v].push_back(u);
    auto angleLess = [&](int o, int x, int y) {
        Point<i128> u{i128(a[x].x) - a[o].x, i128(a[x].y) - a[o].y};
        Point<i128> v{i128(a[y].x) - a[o].x, i128(a[y].y) - a[o].y};
        auto half = [](Point<i128> p) {
            return p.y < 0 || (p.y == 0 && p.x < 0);
        };
        if (half(u) != half(v)) return half(u) < half(v);
        i128 z = cross(u, v);
        return z ? z > 0 : dot(u, u) < dot(v, v);
    };
    map<pair<int, int>, int> pos;
    for (int i = 0; i < n; i++) {
        sort(g[i].begin(), g[i].end(), [&](int x, int y) { return angleLess(i, x, y); });
        for (int j = 0; j < (int)g[i].size(); j++) pos[{i, g[i][j]}] = j;
    }
    set<pair<int, int>> used;
    for (int s = 0; s < n; s++)
        for (int t : g[s])
            if (!used.count({s, t})) {
                vector<int> f;
                int u = s, v = t;
                do {
                    used.insert({u, v});
                    f.push_back(u);
                    int k = pos[{v, u}];
                    int w = g[v][(k - 1 + (int)g[v].size()) % g[v].size()];
                    u = v;
                    v = w;
                } while (u != s || v != t);
                i128 z = 0;
                for (int i = 1; i + 1 < (int)f.size(); i++) z += orient(a[f[0]], a[f[i]], a[f[i + 1]]);
                if (z > 0 && f.size() == 3) ans.f.push_back({f[0], f[1], f[2]});
            }
    return ans;
}
```

### 空间多边形

#### 正N棱锥体积公式

棱锥通用体积公式 $V=\dfrac{1}{3}Sh$ ，当其恰好是棱长为 $l$ 的正 $n$ 棱锥时，有公式 $\displaystyle V=\frac{l^3\cdot n}{12\tan \frac{\pi}{n}}\cdot\sqrt{1-\frac{1}{4\cdot \sin^2\frac{\pi}{n}}}$。

```c++
ld V(ld l, int n) { // 所有底边和侧棱都为 l，非退化时仅 n=3,4,5
    /* assert(l >= 0 && n >= 3 && n <= 5) */;
    return l * l * l * n / (12 * tan(PI / n)) * sqrt(1 - 1 / (4 * sin(PI / n) * sin(PI / n)));
}
```

此特式要求底边与侧棱全部等长，非退化时仅 n=3、4、5；n=6 高为 0，n>6 不存在。一般正棱锥应使用底面积×高/3。

#### 四面体边长体积公式

<a id="tetrahedron-edge-volume"></a>

四面体中 $U,V,W$ 构成一个顶点处的三条边，$u,v,w$ 分别与它们相对，则体积

$$\mathcal V=\frac1{12}\sqrt{4u^2v^2w^2
-\sum_{\rm cyc}u^2(v^2+w^2-U^2)^2
+\prod_{\rm cyc}(v^2+w^2-U^2)},$$

其中循环同步置换 $(u,U),(v,V),(w,W)$；根号内为负表示这些边长不能构成非退化实四面体。


来源：[星尘幻想公式表](<../Wiki/上交-星辰幻想.md#s-7-1-23>)；由数学章按主题迁入，点坐标版见上方。边长还须满足各面三角形的合法性，根号非负本身不是完整的可实现性判据。

#### 通用空间三角形相交

<a id="geo-triangle3"></a>

`triInterFull` 判断两个非退化闭三角形是否有公共点，包含共面重叠、边/点接触和非共面穿越，$\mathcal{O}(1)$。`inTriFull` 包含边界；`segTriFull` 判断闭线段与闭三角形相交。依赖 P3/L3、crossEx/dot/len/sign 和三维 `segInter`。绝对 EPS 会影响近共面分类，极端尺度需按题意调整。

旧 `inTri/segTri/triInter` 的“严格内部”语义保持不变；新题通常优先使用本节完整版本。

```c++
bool inTriFull(P3 p, const array<P3, 3> &t) { // 点在闭三角形上
    P3 n = crossEx(t[1] - t[0], t[2] - t[0]);
    if (sign(len(n)) == 0 || sign(dot(n, p - t[0])) != 0) return false;
    for (int i = 0; i < 3; i++)
        if (sign(dot(crossEx(t[(i + 1) % 3] - t[i], p - t[i]), n)) < 0) return false;
    return true;
}
bool segTriFull(L3 l, const array<P3, 3> &t) { // 闭线段与闭三角形
    P3 n = crossEx(t[1] - t[0], t[2] - t[0]);
    /* assert(sign(len(n))!=0) */;
    ld a = dot(n, l.a - t[0]), b = dot(n, l.b - t[0]);
    if (sign(a) == 0 && sign(b) == 0) {
        if (inTriFull(l.a, t) || inTriFull(l.b, t)) return true;
        for (int i = 0; i < 3; i++)
            if (segInter(l, {t[i], t[(i + 1) % 3]})) return true;
        return false;
    }
    if (sign(a) * sign(b) > 0) return false;
    P3 p = l.a + (l.b - l.a) * (a / (a - b));
    return inTriFull(p, t);
}
bool triInterFull(const array<P3, 3> &a, const array<P3, 3> &b) {
    /* assert(sign(len(crossEx(a[1]-a[0],a[2]-a[0])))!=0) */;
    /* assert(sign(len(crossEx(b[1]-b[0],b[2]-b[0])))!=0) */;
    for (int i = 0; i < 3; i++) {
        if (segTriFull({a[i], a[(i + 1) % 3]}, b)) return true;
        if (segTriFull({b[i], b[(i + 1) % 3]}, a)) return true;
    }
    return false;
}
```

### 空间球

<a id="geo-sphere"></a>

`Sphere3` 表示闭球/球面，`Circle3` 表示空间圆（圆心、单位法向、半径）。`lineSphere` 返回直线与球面的 0/1/2 个交点；`planeSphere` 返回平面与球面是否相交及交圆，相切时交圆半径为 0；`sphereInter` 返回 0 无交、1 相切、2 相交圆、3 两球面重合。均为 $\mathcal{O}(1)$，要求半径非负、直线和平面非退化。

```c++
struct Sphere3 {
    P3 o;
    ld r;
    Sphere3(P3 o_ = {}, ld r_ = 0) : o(o_), r(r_) {}
    bool contains(P3 p) const {
        return r >= 0 && dis(o, p) <= r + EPS;
    }
};
struct Circle3 {
    P3 o, n;
    ld r;
};

tuple<int, P3, P3> lineSphere(L3 l, Sphere3 s) {
    auto [d, p] = projLine(s.o, l);
    if (sign(d - s.r) > 0) return {0, {}, {}};
    if (sign(d - s.r) == 0) return {1, p, p};
    P3 v = unit(l.b - l.a) * sqrtl(max(0.L, s.r * s.r - d * d));
    return {2, p + v, p - v};
}
pair<bool, Circle3> planeSphere(Plane p, Sphere3 s) {
    auto [d, o] = projPlane(s.o, p);
    if (sign(d - s.r) > 0) return {false, {}};
    return {true, {o, unit(getVec(p)), sqrtl(max(0.L, s.r * s.r - d * d))}};
}
pair<int, Circle3> sphereInter(Sphere3 a, Sphere3 b) {
    P3 v = b.o - a.o;
    ld d = len(v);
    if (sign(d) == 0) return {sign(a.r - b.r) == 0 ? 3 : 0, {}};
    if (sign(d - a.r - b.r) > 0 || sign(d - abs(a.r - b.r)) < 0) return {0, {}};
    ld x = (a.r * a.r - b.r * b.r + d * d) / (2 * d);
    ld h2 = max(0.L, a.r * a.r - x * x);
    Circle3 c{a.o + v * (x / d), v / d, sqrtl(h2)};
    return {sign(h2) == 0 ? 1 : 2, c};
}
```

`minSphere` 求覆盖所有点的最小闭球，随机增量期望 $\mathcal{O}(n)$，最坏 $\mathcal{O}(n^{4})$，空间 $\mathcal{O}(n)$。空集返回原点零半径；重复、共线、共面点可用。`sphereBound` 枚举至多四个边界点的全部候选支撑集，专门处理退化；通常只作为 `minSphere` 的内部依赖。随机种子可传入以复现对拍。

```c++
Sphere3 sphereBound(const vector<P3> &p) { // p.size()<=4
    if (p.empty()) return {};
    Sphere3 ans{{}, numeric_limits<ld>::infinity()};
    auto take = [&](Sphere3 s) {
        if (s.r < 0 || s.r >= ans.r) return;
        for (auto q : p)
            if (!s.contains(q)) return;
        ans = s;
    };
    for (auto a : p) take({a, 0});
    for (int i = 0; i < (int)p.size(); i++)
        for (int j = 0; j < i; j++) take({(p[i] + p[j]) / 2, dis(p[i], p[j]) / 2});
    for (int i = 0; i < (int)p.size(); i++)
        for (int j = 0; j < i; j++)
            for (int k = 0; k < j; k++) {
                P3 a = p[i], u = p[j] - a, v = p[k] - a, n = crossEx(u, v);
                ld z = 2 * dot(n, n);
                if (sign(z) != 0) {
                    P3 o = a + (crossEx(v, n) * dot(u, u) + crossEx(n, u) * dot(v, v)) / z;
                    take({o, dis(a, o)});
                }
            }
    if (p.size() == 4) {
        P3 a = p[0], u = p[1] - a, v = p[2] - a, w = p[3] - a;
        ld z = 2 * dot(u, crossEx(v, w));
        if (sign(z) != 0) {
            P3 o = a + (crossEx(v, w) * dot(u, u) + crossEx(w, u) * dot(v, v) + crossEx(u, v) * dot(w, w)) / z;
            take({o, dis(o, a)});
        }
    }
    return ans;
}
Sphere3 minSphere(vector<P3> p, ull seed = chrono::steady_clock::now().time_since_epoch().count()) {
    mt19937_64 rng(seed);
    shuffle(p.begin(), p.end(), rng);
    Sphere3 s{{}, -1};
    for (int i = 0; i < (int)p.size(); i++)
        if (!s.contains(p[i])) {
            s = {p[i], 0};
            for (int j = 0; j < i; j++)
                if (!s.contains(p[j])) {
                    s = sphereBound({p[i], p[j]});
                    for (int k = 0; k < j; k++)
                        if (!s.contains(p[k])) {
                            s = sphereBound({p[i], p[j], p[k]});
                            for (int l = 0; l < k; l++)
                                if (!s.contains(p[l])) s = sphereBound({p[i], p[j], p[k], p[l]});
                        }
                }
        }
    return s.r < 0 ? Sphere3{} : s;
}
```

### 空间旋转与闭合多面体

<a id="geo-solid-extra"></a>

`rotateAxis` 用 Rodrigues 公式将向量 p 绕非零轴向量 u 旋转 rad，遵循右手定则，$\mathcal{O}(1)$。绕过点 o 的轴旋转点 p，使用 `o+rotateAxis(p-o,u,rad)`。依赖 P3、unit、crossEx、dot。

`polyVolume` 对闭合三角网格求体积，$\mathcal{O}(f)$ 时间、$\mathcal{O}(1)$ 额外空间；每个三角面的顶点从外侧看为逆时针，边必须成对出现，整体闭合、不自交。支持凹多面体；空面集返回 0。只取最终和的绝对值，**不能对每个四面体分别取绝对值**。输入索引合法，嵌套空腔的法向需指向材料外部。

```c++
P3 rotateAxis(P3 p, P3 u, ld rad) {
    u = unit(u);
    return p * cosl(rad) + crossEx(u, p) * sinl(rad) + u * (dot(u, p) * (1 - cosl(rad)));
}
ld polyVolume(const vector<P3> &p, const vector<array<int, 3>> &f) {
    if (f.empty()) return 0;
    P3 o = p[f[0][0]];
    ld s = 0;
    for (auto t : f) s += dot(p[t[0]] - o, crossEx(p[t[1]] - o, p[t[2]] - o));
    return abs(s) / 6;
}
```

### 三维凸包

<a id="geo-hull3"></a>

`convexHull3` 返回所有点的满三维凸包，面均为从外侧看逆时针的三角形；时间 $\mathcal{O}(n^{2} \log n)$，空间 $\mathcal{O}(n)$，适合中等点数。算法先精确去重，再增量维护可见面和地平线。输入不足四个非共面点时 `f` 为空；共面点集应投影到二维后调用二维凸包。允许凸包面上有四点及以上，此时同一平面会被三角剖分。

`inHull3` 返回 0 外、1 内、2 边界，$\mathcal{O}(\text{面数})$；`hullArea3` 求表面积。`polyVolume(h.p,h.f)` 可直接求体积。依赖 P3、crossEx/dot/len、EPS；输入必须有限。绝对 EPS 参与可见面判断，近共面或极端尺度数据应改用题目允许的整数扩宽谓词或调整阈值。

```c++
struct Hull3 {
    vector<P3> p;
    vector<array<int, 3>> f;
};
ld vol6(P3 a, P3 b, P3 c, P3 d) {
    return dot(crossEx(b - a, c - a), d - a);
}
Hull3 convexHull3(vector<P3> p) {
    sort(p.begin(), p.end(), [](P3 a, P3 b) {
        if (a.x != b.x) return a.x < b.x;
        if (a.y != b.y) return a.y < b.y;
        return a.z < b.z;
    });
    p.erase(unique(p.begin(), p.end(), [](P3 a, P3 b) { return a.x == b.x && a.y == b.y && a.z == b.z; }), p.end());
    if (p.size() < 4) return {p, {}};
    int j = 1;
    while (j < (int)p.size() && sign(dis(p[0], p[j])) == 0) j++;
    if (j == (int)p.size()) return {p, {}};
    swap(p[1], p[j]);
    j = 2;
    while (j < (int)p.size() && sign(len(crossEx(p[1] - p[0], p[j] - p[0]))) == 0) j++;
    if (j == (int)p.size()) return {p, {}};
    swap(p[2], p[j]);
    j = 3;
    while (j < (int)p.size() && sign(vol6(p[0], p[1], p[2], p[j])) == 0) j++;
    if (j == (int)p.size()) return {p, {}};
    swap(p[3], p[j]);
    P3 in = (p[0] + p[1] + p[2] + p[3]) / 4;
    vector<array<int, 3>> f;
    auto add = [&](int a, int b, int c) {
        if (vol6(p[a], p[b], p[c], in) > 0) swap(b, c);
        f.push_back({a, b, c});
    };
    add(0, 1, 2);
    add(0, 3, 1);
    add(0, 2, 3);
    add(1, 3, 2);
    for (int i = 4; i < (int)p.size(); i++) {
        vector<array<int, 3>> keep;
        map<pair<int, int>, bool> edge;
        auto ae = [&](int a, int b) {
            auto it = edge.find({b, a});
            if (it == edge.end()) edge[{a, b}] = 1;
            else edge.erase(it);
        };
        for (auto t : f)
            if (vol6(p[t[0]], p[t[1]], p[t[2]], p[i]) > EPS) ae(t[0], t[1]), ae(t[1], t[2]), ae(t[2], t[0]);
            else keep.push_back(t);
        if (edge.empty()) continue;
        f.swap(keep);
        for (auto e : edge) add(e.first.first, e.first.second, i);
    }
    return {p, f};
}
int inHull3(P3 q, const Hull3 &h) {
    if (h.f.empty()) return 0;
    bool bd = false;
    for (auto t : h.f) {
        int s = sign(vol6(h.p[t[0]], h.p[t[1]], h.p[t[2]], q));
        if (s > 0) return 0;
        if (s == 0) bd = true;
    }
    return bd ? 2 : 1;
}
ld hullArea3(const Hull3 &h) {
    ld s = 0;
    for (auto t : h.f) s += len(crossEx(h.p[t[1]] - h.p[t[0]], h.p[t[2]] - h.p[t[0]])) / 2;
    return s;
}
```
