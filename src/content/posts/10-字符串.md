### 字符串模式匹配算法 KMP

> 应用：
> 
> 1. 在字符串中查找子串；
> 2. 最小周期：字符串长度-整个字符串的 $\tt border$ ；
> 3. 最小循环节：区别于周期，当字符串长度 $n \bmod (n - nxt[n]) = 0$ 时，等于最小周期，否则为 $n$ 。

以最坏 $\mathcal O(N+M)$ 的时间计算 $t$ 在 $s$ 中出现的全部位置。

```c++
vector<int> kmp(const string &s, const string &t) { // 返回模式串出现的 1-based 起点，允许重叠
    int n = s.size(), m = t.size();
    vector<int> pi(m), ans; // pi[i]：t[0..i] 的最长真 border 长度
    if (m == 0) {
        ans.resize(n + 1);
        iota(ans.begin(), ans.end(), 1);
        return ans;
    }
    for (int i = 1, j = 0; i < m; ++i) {
        while (j && t[i] != t[j]) j = pi[j - 1];
        if (t[i] == t[j]) ++j;
        pi[i] = j;
    }
    for (int i = 0, j = 0; i < n; ++i) {
        while (j && s[i] != t[j]) j = pi[j - 1];
        if (s[i] == t[j]) ++j;
        if (j == m) {
            ans.push_back(i - m + 2);
            j = pi[j - 1];
        }
    }
    return ans;
}
```

### Z函数（扩展 KMP）

获取字符串 $s$ 和 $s[i,n-1]$ （即以 $s[i]$ 开头的后缀）的最长公共前缀（LCP）的长度，总复杂度 $\mathcal O(N)$。

```c++
// zFunc：计算 Z 数组
vector<int> zFunc(string s) {
    int n = s.size();
    vector<int> z(n);
    if (n == 0) return z; // 空串没有任何后缀
    z[0] = n;
    for (int i = 1, j = 1; i < n; i++) {
        z[i] = max(0, min(j + z[j] - i, z[i - j]));
        while (i + z[i] < n && s[z[i]] == s[i + z[i]]) z[i]++;
        if (i + z[i] > j + z[j]) j = i;
    }
    return z;
}
```

### 字符串哈希

#### 双哈希封装

<a id="double-hash"></a>

直接用 ll 保存两个模数下的哈希值，无额外模运算类依赖。

构造 $\mathcal{O}(n)$，get/substring/modify 均为 $\mathcal{O}(1)$，空间 $\mathcal{O}(n)$。substring 使用 0-based 闭区间；modify 只返回假设替换一个字符后的整串哈希，不修改对象。哈希有碰撞概率，不作为严格判等证明；需要确定性匹配时用 KMP/后缀结构。幂表自动扩展，构造对象后不要手动缩短幂表。

```c++
const int N = 1 << 21;
static const int mod1 = 1E9 + 7, base1 = 127;
static const int mod2 = 1E9 + 9, base2 = 131;
vector<ll> val1;
vector<ll> val2;
void init(int n = N) {
    val1.resize(n + 1), val2.resize(n + 2);
    val1[0] = 1, val2[0] = 1;
    for (int i = 1; i <= n; i++) {
        val1[i] = val1[i - 1] * base1 % mod1;
        val2[i] = val2[i - 1] * base2 % mod2;
    }
}
struct String {
    vector<ll> hash1; // hash1[i]：前 i 个字符的第一模哈希
    vector<ll> hash2;
    string s;

    String(string s_) : s(s_), hash1{0}, hash2{0} {
        if (val1.size() <= s.size()) init(s.size());
        for (unsigned char it : s) {
            hash1.push_back((hash1.back() * base1 + it) % mod1);
            hash2.push_back((hash2.back() * base2 + it) % mod2);
        }
    }
    pair<ll, ll> get() { // 输出整串的哈希值
        return {hash1.back(), hash2.back()};
    }
    pair<ll, ll> substring(int l, int r) { // 输出子串的哈希值
        if (l > r) swap(l, r);
        /* assert(0 <= l && r < int(s.size())) */;
        ll ans1 = (hash1[r + 1] - hash1[l] * val1[r - l + 1] % mod1 + mod1) % mod1;
        ll ans2 = (hash2[r + 1] - hash2[l] * val2[r - l + 1] % mod2 + mod2) % mod2;
        return {ans1, ans2};
    }
    pair<ll, ll> modify(int idx, char x) { // 修改 idx 位为 x
        int n = s.size() - 1;
        /* assert(0 <= idx && idx <= n) */;
        ll d = ll((unsigned char)x) - ll((unsigned char)s[idx]);
        ll ans1 = (hash1.back() + val1[n - idx] * d % mod1 + mod1) % mod1;
        ll ans2 = (hash2.back() + val2[n - idx] * d % mod2 + mod2) % mod2;
        return {ans1, ans2};
    }
};
```

#### 前后缀去重

`sample please ease` 去重后得到 `samplease`。

```c++
string compress(vector<string> in) { // 前后缀压缩
    size_t mx = 0;
    for (const auto &s : in) mx = max(mx, s.size());
    if (val1.size() <= mx) init(mx);
    vector<ll> hash1{1};
    vector<ll> hash2{1};
    string ans = "#";
    for (auto s : in) {
        s = "#" + s;
        int st = 0;
        ll chk1 = 0;
        ll chk2 = 0;
        for (int j = 1; j < s.size() && j < ans.size(); j++) {
            chk1 = (chk1 * base1 + (unsigned char)s[j]) % mod1;
            chk2 = (chk2 * base2 + (unsigned char)s[j]) % mod2;
            if ((hash1.back() == (hash1[ans.size() - 1 - j] * val1[j] + chk1) % mod1) &&
                (hash2.back() == (hash2[ans.size() - 1 - j] * val2[j] + chk2) % mod2))
                st = j;
        }
        for (int j = st + 1; j < s.size(); j++) {
            ans += s[j];
            hash1.push_back((hash1.back() * base1 + (unsigned char)s[j]) % mod1);
            hash2.push_back((hash2.back() * base2 + (unsigned char)s[j]) % mod2);
        }
    }
    return ans.substr(1);
}
```

### 马拉车

$\mathcal O(N)$ 时间求出字符串的最长回文子串。

```c++
vector<int> manacher(const string &s) { // p[i]-1 为变换串中心 i 对应的原串回文长度
    vector<int> t{-2, -1}; // 使用负整数作哨兵，不与任意字节字符冲突
    for (unsigned char c : s) {
        t.push_back(c);
        t.push_back(-1);
    }
    int m = t.size();
    t.push_back(-3);
    vector<int> p(m);
    int mid = 0, r = 0; // r 为最右回文的右侧开边界
    for (int i = 1; i < m; ++i) {
        p[i] = i < r ? min(p[2 * mid - i], r - i) : 1;
        while (t[i - p[i]] == t[i + p[i]]) ++p[i];
        if (i + p[i] > r) {
            r = i + p[i];
            mid = i;
        }
    }
    return p; // 最长回文长度：*max_element(p.begin(),p.end()) - 1
}
```

### 字典树 trie

<a id="character-trie"></a>

整数按位存储与异或查询见[数据结构：二进制 Trie](<08 - 数据结构.md#binary-trie>)。

#### 基础封装

```c++
struct Trie { // 62 种字母数字；节点动态初始化，局部构造和 clear 均安全
    vector<array<int, 62>> ch{array<int, 62>{}};
    vector<int> cnt{0}; // cnt[u]：以该节点字符串为前缀的插入次数
    static int id(char c) {
        if ('a' <= c && c <= 'z') return c - 'a';
        if ('A' <= c && c <= 'Z') return c - 'A' + 26;
        if ('0' <= c && c <= '9') return c - '0' + 52;
        throw invalid_argument("Trie requires letters or digits");
    }
    void clear() {
        ch.assign(1, {});
        cnt.assign(1, 0);
    }
    void insert(const string &s) {
        int u = 0;
        ++cnt[0];
        for (char c : s) {
            int v = id(c);
            if (!ch[u][v]) {
                ch[u][v] = ch.size();
                ch.push_back({});
                cnt.push_back(0);
            }
            u = ch[u][v];
            ++cnt[u];
        }
    }
    int query(const string &s) const { // 前缀计数；空串返回全部插入次数，不是整串相等计数
        int u = 0;
        for (char c : s) {
            u = ch[u][id(c)];
            if (!u) return 0;
        }
        return cnt[u];
    }
};
```

### 后缀数组 SA

倍增 + 计数排序实现，时间复杂度 $\mathcal O(N\log N)$，空间复杂度 $\mathcal O(N)$。支持空串；sa 为后缀起点，rk 为其排名，lc[i] 是相邻后缀 sa[i]、sa[i+1] 的 LCP。

```c++
struct SuffixArray {
    int n;
    vector<int> sa, rk, lc;
    SuffixArray(const string &s) {
        n = s.length();
        sa.resize(n);
        if (n == 0) return;
        lc.resize(n - 1);
        rk.resize(n);
        iota(sa.begin(), sa.end(), 0);
        sort(sa.begin(), sa.end(), [&](int a, int b) { return s[a] < s[b]; });
        rk[sa[0]] = 0;
        for (int i = 1; i < n; ++i) rk[sa[i]] = rk[sa[i - 1]] + (s[sa[i]] != s[sa[i - 1]]);
        int k = 1;
        vector<int> tmp, cnt(n);
        tmp.reserve(n);
        while (rk[sa[n - 1]] < n - 1) {
            tmp.clear();
            for (int i = 0; i < k; ++i) tmp.push_back(n - k + i);
            for (auto i : sa)
                if (i >= k) tmp.push_back(i - k);
            fill(cnt.begin(), cnt.end(), 0);
            for (int i = 0; i < n; ++i) ++cnt[rk[i]];
            for (int i = 1; i < n; ++i) cnt[i] += cnt[i - 1];
            for (int i = n - 1; i >= 0; --i) sa[--cnt[rk[tmp[i]]]] = tmp[i];
            swap(rk, tmp);
            rk[sa[0]] = 0;
            for (int i = 1; i < n; ++i)
                rk[sa[i]] = rk[sa[i - 1]] +
                            (tmp[sa[i - 1]] < tmp[sa[i]] || sa[i - 1] + k == n || tmp[sa[i - 1] + k] < tmp[sa[i] + k]);
            k *= 2;
        }
        for (int i = 0, j = 0; i < n; ++i) {
            if (rk[i] == 0) {
                j = 0;
                continue;
            }
            for (j -= j > 0; i + j < n && sa[rk[i] - 1] + j < n && s[i + j] == s[sa[rk[i] - 1] + j];) ++j;
            lc[rk[i] - 1] = j;
        }
    }
};
```

### AC 自动机

设模式总长度 L、节点数 V、文本长度 m、字符集 $\sigma =26$。插入 $\mathcal{O}(L)$，构建 $\mathcal{O}(\sigma V)$，每次查询 $\mathcal{O}(m+V)$，空间 $\mathcal{O}(\sigma V)$。query 返回出现过的模式串数量（含重复插入的重数），不是所有匹配位置数；空模式不支持，空文本返回 0。

```c++
struct ACAutomaton { // 小写字母；插入全部模式串后 build；build 后不再插入
    vector<array<int, 26>> ch{array<int, 26>{}};
    vector<int> fail{0}, cnt{0}, ord; // cnt 为模式串重数，ord 为 BFS 顺序
    bool built = false;
    void clear() {
        ch.assign(1, {});
        fail.assign(1, 0);
        cnt.assign(1, 0);
        ord.clear();
        built = false;
    }
    void insert(const string &s) {
        /* assert(!built && !s.empty()) */;
        int u = 0;
        for (char c : s) {
            int x = c - 'a'; /* assert(0 <= x && x < 26) */;
            if (!ch[u][x]) {
                ch[u][x] = ch.size();
                ch.push_back({});
                fail.push_back(0);
                cnt.push_back(0);
            }
            u = ch[u][x];
        }
        ++cnt[u];
    }
    void build() {
        if (built) return;
        queue<int> q;
        for (int v : ch[0])
            if (v) q.push(v);
        while (!q.empty()) {
            int u = q.front();
            q.pop();
            ord.push_back(u);
            for (int c = 0; c < 26; ++c) {
                int v = ch[u][c];
                if (v) {
                    fail[v] = ch[fail[u]][c];
                    q.push(v);
                } else ch[u][c] = ch[fail[u]][c];
            }
        }
        built = true;
    }
    ll query(const string &s) const { // 文本中出现过的模式串数量；重复插入分别计数
        /* assert(built) */;
        vector<bool> seen(ch.size());
        int u = 0;
        for (char c : s) {
            int x = c - 'a'; /* assert(0 <= x && x < 26) */;
            u = ch[u][x];
            seen[u] = true;
        }
        for (int i = int(ord.size()) - 1; i >= 0; --i)
            if (seen[ord[i]]) seen[fail[ord[i]]] = true;
        ll ans = 0;
        for (int v : ord)
            if (seen[v]) ans += cnt[v];
        return ans; // 不破坏 cnt，可重复查询不同文本
    }
};
```

### 回文自动机 PAM（回文树）

> 应用：
> 
> 1. 本质不同的回文串个数：`tot - 1`（两个根不计入） ；
> 2. 回文子串出现次数：构建后调用 `count()`；`cnt[last]` 是当前前缀末尾的回文子串数量。

对于一个字符串 $s$，它的本质不同回文子串个数最多只有 $|s|$ 个，那么，构造 $s$ 的回文树的时间复杂度是 $\mathcal O(|s|)$ 。

```c++
// tot：已分配的最大节点编号
struct PalindromeAutomaton {
    vector<array<int, 26>> tr;
    vector<int> fail, len, cnt, occ; // cnt：该点的回文后缀数；occ：作为最长回文后缀出现的次数
    int tot = 1, last = 0; // 0 为偶根，1 为奇根，有效回文节点从 2 开始
    string s;
    PalindromeAutomaton(const string &s) : tr(2), fail{1, 0}, len{0, -1}, cnt(2), occ(2), s(s) {}
    int get_fail(int u, int i) const { // 沿 fail 找到能被 s[i] 扩展的最长回文
        while (i - len[u] - 1 < 0 || s[i - len[u] - 1] != s[i]) u = fail[u];
        return u;
    }
    void insert(char c, int i) { // 按 i=0..s.size()-1 顺序调用，c 必须是 s[i]
        /* assert(0 <= i && i < int(s.size()) && c == s[i] && 'a' <= c && c <= 'z') */;
        int x = c - 'a', u = get_fail(last, i);
        if (!tr[u][x]) {
            int v = ++tot, f = tr[get_fail(fail[u], i)][x];
            tr.push_back({});
            fail.push_back(f);
            len.push_back(len[u] + 2);
            cnt.push_back(cnt[f] + 1);
            occ.push_back(0);
            tr[u][x] = v;
        }
        last = tr[u][x];
        ++occ[last];
    }
    vector<int> count() const { // 每个回文串的总出现次数；不破坏原计数
        auto a = occ;
        for (int u = tot; u >= 2; --u) a[fail[u]] += a[u];
        return a;
    }
};
```

### 后缀自动机 SAM

固定 26 字母转移数组，构建时间和空间 $\mathcal{O}(26n)$，count 的时间 $\mathcal{O}(n)$。从根 1 开始连续调用 `last=extend(last,c-'a')`；这份是单串 SAM，不支持从任意历史状态分叉插入。不同子串数为 $\sum_{u\ge2}(\mathrm{len}[u]-\mathrm{len}[\mathrm{link}[u]])$，u≥2。

```c++
// tot：已分配的最大节点编号
struct SuffixAutomaton { // 单串 SAM，小写字母；根为 1，0 为无状态
    struct node {
        int len = 0, link = 0;
        array<int, 26> nxt{};
        int siz = 0;
    };
    vector<node> t = vector<node>(2);
    int tot = 1, last = 1;
    int extend(int p, int c) { // 必须沿同一个字符串顺序扩展：p=last，c=字符-'a'
        /* assert(p == last && 0 <= c && c < 26) */;
        int u = ++tot;
        t.push_back(node{});
        t[u].len = t[p].len + 1;
        t[u].siz = 1;
        while (p && !t[p].nxt[c]) {
            t[p].nxt[c] = u;
            p = t[p].link;
        }
        if (!p) t[u].link = 1;
        else {
            int q = t[p].nxt[c];
            if (t[p].len + 1 == t[q].len) t[u].link = q;
            else {
                int v = ++tot;
                node clone = t[q];
                t.push_back(clone);
                t[v].len = t[p].len + 1;
                t[v].siz = 0;
                while (p && t[p].nxt[c] == q) {
                    t[p].nxt[c] = v;
                    p = t[p].link;
                }
                t[q].link = t[u].link = v;
            }
        }
        return last = u;
    }
    vector<int> count() const { // 各状态 endpos 大小，即对应子串出现次数
        vector<int> a(t.size()), cnt(t[last].len + 1), ord(tot);
        for (int u = 1; u <= tot; ++u) {
            ++cnt[t[u].len];
            a[u] = t[u].siz;
        }
        for (int i = 1; i < int(cnt.size()); ++i) cnt[i] += cnt[i - 1];
        for (int u = 1; u <= tot; ++u) ord[--cnt[t[u].len]] = u;
        for (int i = int(ord.size()) - 1; i > 0; --i) a[t[ord[i]].link] += a[ord[i]];
        return a;
    }
};
```

### 子序列自动机

下方保存每种元素出现的位置并逐个二分。自动离散化版预处理 $\mathcal{O}(n \log n)$，查询 $\mathcal{O}(m(\log \sigma +\log n))$；朴素版预处理 $\mathcal{O}(n+\sigma)$，查询 $\mathcal{O}(m \log n)$，空间均为 $\mathcal{O}(n+\sigma)$。空询问为真。

#### 自动离散化、自动类型匹配封装

```c++
template <class T> struct SequenceAutomaton {
    vector<T> alls;
    vector<vector<int>> ver;

    template <class Seq> SequenceAutomaton(const Seq &in) { // C++14 成员函数模板
        for (auto &i : in) alls.push_back(i);
        sort(alls.begin(), alls.end());
        alls.erase(unique(alls.begin(), alls.end()), alls.end());

        ver.resize(alls.size() + 1);
        for (int i = 0; i < in.size(); i++) ver[get(in[i])].push_back(i + 1);
    }
    bool count(T x) const {
        return binary_search(alls.begin(), alls.end(), x);
    }
    int get(T x) const {
        return lower_bound(alls.begin(), alls.end(), x) - alls.begin();
    }
    template <class Seq> bool contains(const Seq &in) const {
        int at = 0;
        for (auto &i : in) {
            if (!count(i)) return false;

            auto j = get(i);
            auto it = lower_bound(ver[j].begin(), ver[j].end(), at + 1);
            if (it == ver[j].end()) return false;
            at = *it;
        }
        return true;
    }
};
```

#### 朴素封装

主串元素必须在 [0,size]；查询遇到范围外元素直接返回 false。

```c++
struct SequenceAutomaton {
    vector<vector<int>> ver;

    SequenceAutomaton(const vector<int> &in, int size) : ver(size + 1) {
        for (int i = 0; i < in.size(); i++) {
            /* assert(0 <= in[i] && in[i] <= size) */;
            ver[in[i]].push_back(i + 1);
        }
    }
    bool contains(const vector<int> &in) const {
        int at = 0;
        for (auto &i : in) {
            if (i < 0 || i >= int(ver.size())) return false;
            auto it = lower_bound(ver[i].begin(), ver[i].end(), at + 1);
            if (it == ver[i].end()) return false;
            at = *it;
        }
        return true;
    }
};
```
